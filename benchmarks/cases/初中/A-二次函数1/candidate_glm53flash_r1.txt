MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
A = (0, 3)
B = (-1, 0)
D = (2, 3)
E = (Root(f, 2, 4), 0)
C = B + D - A
O = (0, 0)
polyABCD = Polygon(A, B, C, D)
M = (A + C) / 2
l = Line(E, M)
F = Intersect(l, f, 2)
t = Slider(-2/5, 3, 0.05)
P = (t, f(t))
polyPFE = Polygon(P, F, E)
g(x) = 3x + 5f(x) - 9
dg = Derivative(g)
tMax = Root(dg, 0, 3)
polyPAE = Polygon(P, A, E)
segAE = Segment(A, E)
perpA = PerpendicularLine(A, segAE)
k(x) = f(x) - x - 3
tRtA = Root(k, 0.5, 3)
RtA = (tRtA, f(tRtA))
cAE = Circle(Midpoint(A, E), A)
h(x) = x(x - 3) + (f(x) - 3) f(x)
tRtP = Root(h, 1, 2)
RtP = (tRtP, f(tRtP))
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
SetLabelMode(E, 1)
ShowLabel(E, true)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFilling(polyABCD, 0.2)
SetColor(polyABCD, "Blue")
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(l, "Green")
SetLineStyle(l, 2)
SetColor(F, "Black")
SetPointSize(F, 4)
SetLabelMode(F, 1)
ShowLabel(F, true)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPFE, 0.3)
SetColor(polyPFE, "Red")
SetValue(t, tMax)
SetFilling(polyPAE, 0.15)
SetColor(polyPAE, "Orange")
SetColor(segAE, "Black")
SetColor(perpA, "Gray")
SetLineStyle(perpA, 2)
SetColor(RtA, "Orange")
SetPointSize(RtA, 5)
ShowLabel(RtA, true)
SetLabelMode(RtA, 1)
SetColor(cAE, "Gray")
SetLineStyle(cAE, 2)
SetColor(RtP, "Orange")
SetPointSize(RtP, 5)
ShowLabel(RtP, true)
SetLabelMode(RtP, 1)
