MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2*x + 3
A = (0, 3)
B = (-1, 0)
D = (2, 3)
C = B + D - A
E = (Root(f, 2, 4), 0)
polyABCD = Polygon(A, B, C, D)
M = Midpoint(A, C)
l = Line(E, M)
F = Intersect(l, f, 2)
t = Slider(-2/5, 3, 0.05)
P = (t, f(t))
polyPAE = Polygon(P, A, E)
polyPFE = Polygon(P, F, E)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(E, true)
SetColor(E, "Black")
SetPointSize(E, 4)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFilling(polyABCD, 0.2)
SetColor(polyABCD, "Blue")
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(l, "Green")
SetLineStyle(l, 2)
SetFixed(F, true)
SetColor(F, "Black")
SetPointSize(F, 4)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPAE, 0.1)
SetColor(polyPAE, "Orange")
SetFilling(polyPFE, 0.3)
SetColor(polyPFE, "Red")
