MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
SetColor(f, "Black")
A = (0, 3)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (-1, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
D = (2, 3)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
C = (1, 0)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
M = Midpoint(A, C)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
E = (Root(f, 2, 4), 0)
SetColor(E, "Black")
SetPointSize(E, 4)
ShowLabel(E, true)
SetLabelMode(E, 1)
l = Line(E, M)
SetColor(l, "Blue")
F = Intersect(f, l, -1, 0)
SetColor(F, "Black")
SetPointSize(F, 4)
ShowLabel(F, true)
SetLabelMode(F, 1)
polyABCD = Polygon(A, B, C, D)
SetColor(polyABCD, "Gray")
t = Slider(-0.4, 3, 0.05)
P = (t, f(t))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
triPFE = Polygon(P, F, E)
SetColor(triPFE, "Green")
SetFilling(triPFE, 0.2)
triPAE = Polygon(P, A, E)
SetColor(triPAE, "Orange")
SetFilling(triPAE, 0.2)
