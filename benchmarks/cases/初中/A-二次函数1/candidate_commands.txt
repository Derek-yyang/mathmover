MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
// 1. 定义已知点并计算解析式
A = (0, 3)
B = (-1, 0)
D = (2, 3)
// 抛物线解析式推导：过 A(0,3) 和 D(2,3) 对称轴为 x=1，设 y=a(x-0)(x-2)+3，代入 B(-1,0) 得 0=a(-1)(-3)+3 => a=-1
f(x) = -x^2 + 2x + 3
// 2. 构造平行四边形 ABCD
// 根据 AD 向量 (2,0)，确定 C 点坐标为 B + (2,0) = (1, 0)
C = (1, 0)
poly_ABCD = Polygon(A, B, C, D)
SetColor(poly_ABCD, "Blue")
SetFilling(poly_ABCD, 0.2)
// 3. 计算点 E 和直线 l
// E 为抛物线与 x 轴另一交点，-x^2 + 2x + 3 = 0 得 x=3 或 x=-1 (B点)，故 E=(3, 0)
E = (3, 0)
// 直线 l 过点 E 且平分平行四边形面积，必过 ABCD 中心 M
M = ((0+1)/2, (3+0)/2) // M = (0.5, 1.5)
// 直线 l 方程：y - 0 = ((1.5-0)/(0.5-3))*(x - 3) => y = -0.6x + 1.8
l = Line(E, M)
SetLineStyle(l, "Dashed")
SetColor(l, "Gray")
// 4. 计算交点 F
// 联立 -x^2 + 2x + 3 = -0.6x + 1.8 => x^2 - 2.6x - 1.2 = 0
// 解得 x = 3 (E点) 或 x = -0.4
x_f = -0.4
y_f = -0.6 * x_f + 1.8
F = (x_f, y_f)
// 5. 设置动点 P
// 点 P 在直线 l 上方的抛物线上运动，对应 x 范围为 [-0.4, 3]
t = Slider(-0.4, 3, 0.01)
P = (t, f(t))
SetColor(P, "Red")
SetPointSize(P, 5)
// 6. 构造动态三角形（对应小问 2 和 3）
poly_PFE = Polygon(P, F, E)
SetColor(poly_PFE, "Orange")
SetFilling(poly_PFE, 0.3)
poly_PAE = Polygon(P, A, E)
SetColor(poly_PAE, "Green")
SetLineStyle(poly_PAE, "Dashed")
SetFilling(poly_PAE, 0.1)
// 7. 静态点样式设置
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
ShowLabel(P, true)
SetLabelMode(P, 1)
// 限制缩放
Execute({"SetOptions(\"Zooming\", false)"})
