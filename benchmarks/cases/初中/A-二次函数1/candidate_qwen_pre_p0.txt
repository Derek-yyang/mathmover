MODE: 2D
// 1. 定义已知定点
A = (0, 3)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
B = (-1, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
D = (2, 3)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
// 2. 推导并定义抛物线
// 代入 A(0,3) -> c=3
// 代入 B(-1,0) -> a-b+3=0
// 代入 D(2,3) -> 4a+2b+3=3 -> 2a+b=0 -> b=-2a
// a - (-2a) + 3 = 0 -> 3a = -3 -> a = -1, b = 2
f(x) = -x^2 + 2x + 3
SetColor(f, "Blue")
SetLineStyle(f, "Solid")
// 3. 定义平行四边形 ABCD 的顶点 C
// 向量 AB = (-1, -3), C = D + AB = (2-1, 3-3) = (1, 0)
C = (1, 0)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
// 4. 构建平行四边形区域
polyABCD = Polygon(A, B, C, D)
SetColor(polyABCD, "LightGray")
SetFilling(polyABCD, 0.2)
SetLineStyle(polyABCD, "Solid")
// 5. 定义点 E (抛物线与 x 轴另一交点)
// -x^2 + 2x + 3 = 0 -> x^2 - 2x - 3 = 0 -> (x-3)(x+1)=0 -> x=3 (E), x=-1 (B)
E = (3, 0)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
// 6. 定义直线 l (过 E 且平分平行四边形面积 -> 过中心 M)
// 中心 M 为 BD 中点: ((-1+2)/2, (0+3)/2) = (0.5, 1.5)
M = (0.5, 1.5)
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Gray")
SetPointSize(M, 2)
// 直线 l 过 E(3,0) 和 M(0.5, 1.5)
// k = (1.5 - 0) / (0.5 - 3) = 1.5 / -2.5 = -0.6
// y - 0 = -0.6(x - 3) -> y = -0.6x + 1.8
line_l: y = -0.6x + 1.8
SetColor(line_l, "Red")
SetLineStyle(line_l, "Dashed")
// 7. 定义点 F (直线 l 与抛物线的另一交点)
// -x^2 + 2x + 3 = -0.6x + 1.8 -> x^2 - 2.6x - 1.2 = 0 -> 5x^2 - 13x - 6 = 0
// (5x + 2)(x - 3) = 0 -> x = -0.4 (F), x = 3 (E)
x_F = -0.4
y_F = -0.6 * x_F + 1.8
F = (x_F, y_F)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
// 8. 定义动点 P
// P 在直线 l 上方抛物线上，横坐标 t 范围为 (x_F, x_E) 即 (-0.4, 3)
t = Slider(-0.4, 3, 0.01)
P = (t, f(t))
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
// 9. 辅助图形：三角形 PFE (对应第2问) 和 三角形 PAE (对应第3问)
// 为了清晰展示，只描边不填充，或者浅色填充
polyPFE = Polygon(P, F, E)
SetColor(polyPFE, "Green")
SetFilling(polyPFE, 0.1)
SetLineStyle(polyPFE, "Solid")
polyPAE = Polygon(P, A, E)
SetColor(polyPAE, "Orange")
SetFilling(polyPAE, 0.1)
SetLineStyle(polyPAE, "Dashed")
// 10. 视图设置
SetAxesVisible(true, true)
SetGridVisible(true)
// 调整视图范围以包含所有关键点
ZoomIn(1)
