MODE: 2D
s = sqrt(5)
O = (0, 0)
B = (-s, 0)
C = (s, 0)
D = (s, 2*s)
A = (-s, 2*s)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
poly1 = Polygon(A, B, C, D)
SetColor(poly1, "Black")
SetFilling(poly1, 0)
pathE = Circle(O, 2)
SetLineStyle(pathE, 2)
SetColor(pathE, "Gray")
t = Slider(0.4, 2.7, 0.01)
E = (2*cos(t), 2*sin(t))
SetColor(E, "Red")
ShowLabel(E, true)
SetLabelMode(E, 1)
F = Rotate(E, 90°, D)
SetColor(F, "Blue")
ShowLabel(F, true)
SetLabelMode(F, 1)
segOE = Segment(O, E)
SetColor(segOE, "Red")
segDE = Segment(D, E)
segDF = Segment(D, F)
angEDF = Angle(E, D, F)
segOF = Segment(O, F)
SetColor(segOF, "Blue")
segAE = Segment(A, E)
segCF = Segment(C, F)
SetLineStyle(segAE, 2)
SetLineStyle(segCF, 2)
