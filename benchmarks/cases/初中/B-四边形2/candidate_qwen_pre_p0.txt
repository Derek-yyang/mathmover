MODE: 2D
s = 2sqrt(5)
B = (0, 0)
C = (s, 0)
D = (s, s)
A = (0, s)
poly1 = Polygon(A, B, C, D)
SetColor(poly1, "lightgray")
SetFilling(poly1, 0.1)
O = Midpoint(B, C)
cE = Circle(O, 2)
SetLineStyle(cE, 2)
SetColor(cE, "gray")
t = Slider(0, pi, 0.01)
E = (x(O) + 2cos(t), y(O) + 2sin(t))
SetColor(E, "red")
ShowLabel(E, true)
SetLabelMode(E, 1)
F = Rotate(E, 90°, D)
SetColor(F, "blue")
ShowLabel(F, true)
SetLabelMode(F, 1)
Segment(D, E)
Segment(D, F)
Segment(A, E)
Segment(C, F)
Segment(O, F)
SetColor(Segment(O, F), "black")
SetLineWidth(Segment(O, F), 2)
SetFixed(A, true)
SetFixed(B, true)
SetFixed(C, true)
SetFixed(D, true)
SetFixed(O, true)
ShowLabel(A, true)
ShowLabel(B, true)
ShowLabel(C, true)
ShowLabel(D, true)
ShowLabel(O, true)
SetLabelMode(A, 1)
SetLabelMode(B, 1)
SetLabelMode(C, 1)
SetLabelMode(D, 1)
SetLabelMode(O, 1)
