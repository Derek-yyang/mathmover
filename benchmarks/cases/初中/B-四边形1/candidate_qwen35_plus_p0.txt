MODE: 2D
A = (0, 0)
B = (160, 0)
C = (160, 120)
D = (0, 120)
SetFixed(A, true)
SetFixed(B, true)
SetFixed(C, true)
SetFixed(D, true)
ShowLabel(A, true)
ShowLabel(B, true)
ShowLabel(C, true)
ShowLabel(D, true)
SetLabelMode(A, 1)
SetLabelMode(B, 1)
SetLabelMode(C, 1)
SetLabelMode(D, 1)
rect = Polygon(A, B, C, D)
SetColor(rect, "black")
SetFilling(rect, 0)
E = Midpoint(B, C)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
F = Point(Segment(A, B))
SetColor(F, "red")
ShowLabel(F, true)
SetLabelMode(F, 1)
dist_EF = Distance(E, F)
k = 4800 / (dist_EF^2)
F_rot = Rotate(F, -90deg, E)
G = Dilate(F_rot, k, E)
SetColor(G, "blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
seg_DG = Segment(D, G)
seg_AG = Segment(A, G)
seg_EF = Segment(E, F)
seg_EG = Segment(E, G)
SetColor(seg_EF, "gray")
SetColor(seg_EG, "gray")
SetLineStyle(seg_EF, 2)
SetLineStyle(seg_EG, 2)
poly_GEF = Polygon(G, E, F)
SetColor(poly_GEF, "gray")
SetFilling(poly_GEF, 0.1)
SetLineStyle(poly_GEF, 2)
