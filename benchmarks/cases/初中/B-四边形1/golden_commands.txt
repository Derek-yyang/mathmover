MODE: 2D
A = (0, 0)
B = (160, 0)
C = (160, 120)
D = (0, 120)
SetFixed(A, true)
SetFixed(B, true)
SetFixed(C, true)
SetFixed(D, true)
ShowLabel(A, true)
ShowLabel(B, true)
ShowLabel(C, true)
ShowLabel(D, true)
SetLabelMode(A, 1)
SetLabelMode(B, 1)
SetLabelMode(C, 1)
SetLabelMode(D, 1)
rect = Polygon(A, B, C, D)
E = Midpoint(B, C)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
t = Slider(0, 1, 0.01)
F = Point(Segment(A, B), t)
SetColor(F, "Red")
ShowLabel(F, true)
SetLabelMode(F, 1)
distSq = (x(F) - x(E))^2 + (y(F) - y(E))^2
G = E + (4800 / distSq) * (y(F) - y(E), x(E) - x(F))
SetColor(G, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
triGEF = Polygon(G, E, F)
SetColor(triGEF, "Green")
SetFilling(triGEF, 0.3)
triADG = Polygon(A, D, G)
SetColor(triADG, "Orange")
SetFilling(triADG, 0.3)
dg = Segment(D, G)
ag = Segment(A, G)
angGEF = Angle(F, E, G)
SetFixed(angGEF, true)
