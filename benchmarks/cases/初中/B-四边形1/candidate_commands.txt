SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (160, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (160, 120)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (0, 120)
SetFixed(D, true)
ShowLabel(D, true)

E = Midpoint(B, C)
ShowLabel(E, true)

ShowLabel(F, true)

G = E + 4800 / ((x(F) - x(E))^2 + (y(F) - y(E))^2) * (y(F) - y(E), -(x(F) - x(E)))
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(G, "Blue")
