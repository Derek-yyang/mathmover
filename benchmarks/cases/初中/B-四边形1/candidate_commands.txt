MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (160, 0)
C = (160, 120)
D = (0, 120)
E = (160, 60)
t = Slider(0, 160, 0.1)
F = Point(Segment(A, B), t / 160)
vEF = Vector(E, F)
b = Length(vEF)
a = 4800 / b
dir = (-y(vEF), x(vEF)) / b
G = E + a * dir
segDG = Segment(D, G)
distDG = Distance(D, G)
funcDG = distDG
minPoint = Extremum(funcDG, 0, 160)
SetVisible(minPoint, false)
t_min = x(minPoint)
F_min = (t_min, 0)
G_min = E + 4800 / Length(Vector(E, F_min)) * (-y(Vector(E, F_min)), x(Vector(E, F_min))) / Length(Vector(E, F_min))
polyADG = Polygon(A, D, G)
polyGEF = Polygon(G, E, F)
lineAG = Segment(A, G)
lineDG = Segment(D, G)
lineEG = Segment(E, G)
lineFG = Segment(F, G)
areaADG = Area(A, D, G_min)

ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
ShowLabel(t, false)
SetColor(F, "Red")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(G, "Red")
