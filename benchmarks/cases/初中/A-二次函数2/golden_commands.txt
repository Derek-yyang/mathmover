MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f_line(x) = x + 2
# 5/2 = a(1/4) + b/2 + 6  => a + 2b = -14
# 6 = 16a + 4b + 6      => 4a + b = 0 => b = -4a
# a + 2(-4a) = -14 => -7a = -14 => a = 2, b = -8
f_parab(x) = 2x^2 - 8x + 6
f1 = Function(f_line, -2, 5)
f2 = Function(f_parab, 0.5, 4.5)
SetColor(f1, "Blue")
SetColor(f2, "Green")
A = (0.5, 2.5)
B = (4, 6)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
t = Slider(0, 1, 0.01)
P = (1 - t) * A + t * B
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetPointSize(P, 5)
# D = (x_p, 0)
# C = (x_p, f_parab(x_p))
D = (x(P), 0)
C = (x(P), f_parab(x(P)))
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Black")
SetPointSize(D, 3)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 3)
seg_PC = Segment(P, C)
SetColor(seg_PC, "Magenta")
SetLineStyle(seg_PC, "Dashed")
seg_PD = Segment(P, D)
SetColor(seg_PD, "Gray")
SetLineStyle(seg_PD, "Dashed")
seg_AB = Segment(A, B)
SetColor(seg_AB, "Blue")
SetLineStyle(seg_AB, "Solid")
poly_PAC = Polygon(P, A, C)
SetColor(poly_PAC, "Orange")
SetFilling(poly_PAC, 0.25)
SetLineStyle(poly_PAC, "Solid")
