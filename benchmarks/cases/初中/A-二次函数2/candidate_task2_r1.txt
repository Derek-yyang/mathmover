MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2*x^2 - 8*x + 6
g(x) = x + 2
A = (1/2, 5/2)
B = (4, g(4))
E = (Root(g, -3, -1), 0)
O = (0, 0)
t = Slider(1/2, 4, 0.05)
P = (t, g(t))
D = (t, 0)
C = (t, f(t))
segPD = Segment(P, D)
segPC = Segment(P, C)
triPAC = Polygon(P, A, C)
SetColor(f, "Black")
SetColor(g, "Blue")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
SetLabelMode(E, 1)
ShowLabel(E, true)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPD, "Red")
SetLineStyle(segPD, 2)
SetColor(segPC, "Red")
SetLineStyle(segPC, 2)
SetFilling(triPAC, 0.3)
SetColor(triPAC, "Blue")
