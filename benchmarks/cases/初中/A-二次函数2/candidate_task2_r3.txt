MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = x + 2
f(x) = 2x^2 - 8x + 6
A = (1/2, 5/2)
B = (4, 6)
E = (Root(g, -3, -1), 0)
O = (0, 0)
t = Slider(0.5, 4, 0.05)
P = (t, g(t))
C = (t, f(t))
D = (t, 0)
segPC = Segment(P, C)
segPD = Segment(P, D)
triPAC = Polygon(P, A, C)
SetColor(g, "Blue")
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetPointSize(E, 4)
SetColor(E, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segPC, "Red")
SetColor(segPD, "Gray")
SetLineStyle(segPD, 2)
SetColor(triPAC, "Blue")
SetFilling(triPAC, 0.3)
