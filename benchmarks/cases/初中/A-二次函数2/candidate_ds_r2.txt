MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2x^2 - 8x + 6
l: y = x + 2
A = (1/2, 5/2)
B = (4, 6)
t = Slider(0, 1, 0.01)
P = A + t*(B - A)
D = (x(P), 0)
C = (x(P), f(x(P)))
segPC = Segment(P, C)
triPAC = Polygon(P, A, C)
g(x) = (x + 2) - f(x)
gmax = Element(Extremum(g), 1)
P_max = (x(gmax), x(gmax) + 2)
C_max = (x(P_max), f(x(P_max)))
segPC_max = Segment(P_max, C_max)
h1(x) = 4x^2 - 16x + 7
root1 = Root(h1, 2, 4)
P_rt1 = (x(root1), x(root1) + 2)
h2(x) = 2x^2 - 7x + 3
root2 = Root(h2, 2, 4)
P_rt2 = (x(root2), x(root2) + 2)
SetColor(f, "Black")
SetColor(l, "Green")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Black")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Blue")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPC, "Red")
SetLineStyle(segPC, 2)
SetFilling(triPAC, 0.2)
SetColor(triPAC, "Orange")
SetColor(gmax, "Black")
SetColor(P_max, "Black")
SetFixed(P_max, true)
ShowLabel(P_max, true)
SetLabelMode(P_max, 1)
SetColor(C_max, "Black")
SetFixed(C_max, true)
ShowLabel(C_max, true)
SetLabelMode(C_max, 1)
SetColor(segPC_max, "Red")
SetLineStyle(segPC_max, 2)
SetColor(P_rt1, "Black")
SetFixed(P_rt1, true)
ShowLabel(P_rt1, true)
SetLabelMode(P_rt1, 1)
SetColor(P_rt2, "Black")
SetFixed(P_rt2, true)
ShowLabel(P_rt2, true)
SetLabelMode(P_rt2, 1)
