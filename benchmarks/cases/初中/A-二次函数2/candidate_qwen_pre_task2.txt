MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = x + 2
SetColor(g, "Black")
f(x) = 2x^2 - 8x + 6
SetColor(f, "Black")
A = (1/2, 5/2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (4, g(4))
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
E = (Root(g, -5, 0), 0)
SetColor(E, "Black")
SetPointSize(E, 4)
SetLabelMode(E, 1)
ShowLabel(E, true)
t = Slider(1/2, 4, 0.05)
P = (t, g(t))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
C = (t, f(t))
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (t, 0)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
segPD = Segment(P, D)
SetColor(segPD, "Red")
triPAC = Polygon(P, A, C)
SetFilling(triPAC, 0.3)
SetColor(triPAC, "Blue")
