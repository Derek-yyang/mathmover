MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = x + 2
f(x) = 2x^2 - 8x + 6
A = (1/2, 5/2)
B = (4, 6)
E = Intersect(g, xAxis)
O = (0, 0)
t = Slider(0.5, 4, 0.05)
P = (t, t + 2)
perpPC = PerpendicularLine(P, xAxis)
D = Intersect(perpPC, xAxis)
C = (t, f(t))
segPC = Segment(P, C)
triAPC = Polygon(A, P, C)
SetColor(g, "Blue")
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(perpPC, "Gray")
SetLineStyle(perpPC, 2)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPC, "Red")
SetLineStyle(segPC, 2)
SetFilling(triAPC, 0.2)
SetColor(triAPC, "Red")
