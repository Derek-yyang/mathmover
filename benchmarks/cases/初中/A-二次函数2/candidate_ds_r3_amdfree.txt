MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2x^2 - 3x + 5
A = (1/2, 5/2)
B = (4, 7)
l = Line(A, B)
t1 = Slider(0, 1, 0.01)
P = A + t1 * (B - A)
D = (x(P), 0)
C = (x(P), f(x(P)))
Segment(C, D)
g = Segment(A, B)
triPAC = Polygon(P, A, C)
valPC = Length(Segment(P, C))
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l, "Green")
SetLineStyle(l, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Blue")
SetColor(Segment(C, D), "Red")
SetLineThickness(Segment(C, D), 2)
SetColor(g, "Gray")
SetLineStyle(g, 3)
SetFilling(triPAC, 0.2)
SetColor(triPAC, "Red")
ShowLabel(valPC, true)
SetLabelMode(valPC, 1)
