MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = x + 2
f(x) = 2x^(2) - 8x + 6
O = (0, 0)
A = (1/2, 5/2)
B = (4, g(4))
E = (Root(g, -4, 0), 0)
segAB = Segment(A, B)
t = Slider(0.5, 4, 0.01)
P = (t, g(t))
D = (t, 0)
C = (t, f(t))
segPC = Segment(P, C)
polyPAC = Polygon(A, P, C)
SetColor(g, "Green")
SetColor(f, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(segAB, "Blue")
SetValue(t, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Black")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPC, "Purple")
SetLineStyle(segPC, 2)
SetFilling(polyPAC, 0.3)
SetColor(polyPAC, "Red")
