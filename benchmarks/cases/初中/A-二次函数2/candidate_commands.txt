MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = x + 2
f(x) = 2x^2 - 8x + 6
m = g(4)
A = (1/2, 5/2)
B = (4, m)
O = (0, 0)
E = (Root(g, -3, -1), 0)
segAB = Segment(A, B)
t = Slider(0.5, 4, 0.05)
P = (t, g(t))
D = (t, 0)
C = (t, f(t))
segPC = Segment(P, C)
triPAC = Polygon(P, A, C)
h(x) = g(x) - f(x)
maxPC = Extremum(h)
SetColor(g, "Green")
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(E, "Black")
SetPointSize(E, 4)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(segAB, "Green")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Black")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Black")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPC, "Red")
SetFilling(triPAC, 0.3)
SetColor(triPAC, "Red")
SetColor(h, "Blue")
SetColor(maxPC, "Blue")
SetPointSize(maxPC, 4)
ShowLabel(maxPC, true)
SetLabelMode(maxPC, 1)
