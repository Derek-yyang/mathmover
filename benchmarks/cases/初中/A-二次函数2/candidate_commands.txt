MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2x^2 - 8x + 6
A = (0.5, 2.5)
B = (4, 6)
l = Line(A, B)
t = Slider(0.5, 4, 0.05)
P = (t, t + 2)
C = (t, f(t))
D = (t, 0)
segPC = Segment(P, C)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(C, "Green")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segPC, "Red")
SetLineStyle(segPC, 1)
