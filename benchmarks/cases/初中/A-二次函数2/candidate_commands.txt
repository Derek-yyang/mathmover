MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = x + 2
g(x) = (1/2)*x^2 - 4*x + 6
A = (1/2, 5/2)
B = (4, 6)
t = Slider(0.5, 4, 0.01)
P = (t, f(t))
C = (t, g(t))
D = (t, 0)
segmentPC = Segment(P, C)
segmentPD = Segment(P, D)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segmentPC, "Blue")
SetLineStyle(segmentPC, 1)
SetColor(segmentPD, "Blue")
SetLineStyle(segmentPD, 2)
