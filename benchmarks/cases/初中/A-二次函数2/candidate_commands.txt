MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2x^2 - 8x + 6
lineAB: y = x + 2
A = (1/2, 5/2)
B = (4, 6)
segAB = Segment(A, B)
E = Intersect(lineAB, xAxis)
t = Slider(0.5, 4, 0.01)
P = (t, t + 2)
D = (t, 0)
C = (t, f(t))
segPC = Segment(P, C)
triPAC = Polygon(P, A, C)
g(x) = x + 2 - f(x)
maxG = Extremum(g)
Pmax = (x(maxG), x(maxG) + 2)
h1(x) = (x - 1/2)^2 + (x - 1/2)*(f(x) - 5/2)
h2(x) = (f(x) - x - 2)*(f(x) - 5/2)
x1 = Root(h1, 2, 3.9)
P_rtA = (x(x1), x(x1) + 2)
x2 = Root(h2, 3, 3.9)
P_rtC = (x(x2), x(x2) + 2)
SetColor(f, "Black")
SetLineThickness(f, 3)
SetColor(lineAB, "Blue")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(segAB, "Gray")
SetLineStyle(segAB, 3)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(D, "Black")
SetPointSize(D, 3)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(C, "Purple")
SetPointSize(C, 3)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segPC, "Red")
SetLineThickness(segPC, 2)
SetFilling(triPAC, 0.2)
SetColor(triPAC, "Orange")
SetColor(g, "Green")
SetLineStyle(g, 3)
SetColor(maxG, "Magenta")
SetPointSize(maxG, 4)
ShowLabel(maxG, true)
SetLabelMode(maxG, 1)
SetColor(Pmax, "Magenta")
SetPointSize(Pmax, 4)
ShowLabel(Pmax, true)
SetLabelMode(Pmax, 1)
SetColor(P_rtA, "Green")
SetPointSize(P_rtA, 4)
ShowLabel(P_rtA, true)
SetLabelMode(P_rtA, 1)
SetColor(P_rtC, "Green")
SetPointSize(P_rtC, 4)
ShowLabel(P_rtC, true)
SetLabelMode(P_rtC, 1)
