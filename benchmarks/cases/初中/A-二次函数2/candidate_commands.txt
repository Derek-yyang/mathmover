MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = 2x^2 - 8x + 6
SetColor(f, "Black")
lineAB(x) = x + 2
SetColor(lineAB, "Blue")
A = (1/2, 5/2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (4, lineAB(4))
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
E = (Root(lineAB, -3, -1), 0)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
t = Slider(1/2, 4, 0.05)
P = (t, lineAB(t))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
C = (t, f(t))
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (t, 0)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
segPC = Segment(P, C)
SetColor(segPC, "Red")
SetThickness(segPC, 3)
segCD = Segment(C, D)
SetColor(segCD, "Gray")
triPAC = Polygon(P, A, C)
SetFilling(triPAC, 0.2)
SetColor(triPAC, "Green")
