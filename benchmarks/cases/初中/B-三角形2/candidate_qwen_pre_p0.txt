MODE: 2D
A = (0, sqrt(3))
B = (-1, 0)
C = (1, 0)
poly1 = Polygon(A, B, C)
SetFilling(poly1, 0.1)
E = Midpoint(A, B)
F = Midpoint(A, C)
G = Midpoint(B, C)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
seg_EF = Segment(E, F)
SetLineStyle(seg_EF, 2)
t = Slider(0, 1, 0.01)
P = Point(seg_EF, t)
SetColor(P, "red")
ShowLabel(P, true)
SetLabelMode(P, 1)
seg_BP = Segment(B, P)
seg_PG = Segment(P, G)
SetColor(seg_BP, "blue")
SetColor(seg_PG, "blue")
