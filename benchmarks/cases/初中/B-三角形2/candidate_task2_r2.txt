MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
B = (0, 0)
C = (2, 0)
A = (1, sqrt(3))
E = Midpoint(A, B)
F = Midpoint(A, C)
G = Midpoint(B, C)
segEF = Segment(E, F)
t = Slider(0, 1, 0.01)
P = Point(segEF, t)
lineEF = Line(E, F)
B_prime = Reflect(B, lineEF)
segBprimeG = Segment(B_prime, G)
P_opt = Intersect(segBprimeG, segEF)
polyABC = Polygon(A, B, C)
polyBPG = Polygon(B, P, G)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetLineStyle(lineEF, 2)
SetColor(lineEF, "Gray")
SetColor(B_prime, "Blue")
ShowLabel(B_prime, true)
SetLabelMode(B_prime, 1)
SetLineStyle(segBprimeG, 2)
SetColor(segBprimeG, "Gray")
SetColor(P_opt, "Blue")
ShowLabel(P_opt, true)
SetLabelMode(P_opt, 1)
SetFilling(polyABC, 0.1)
SetFilling(polyBPG, 0.2)
