MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
B = (0, 0)
C = (2, 0)
A = (1, sqrt(3))
triABC = Polygon(A, B, C)
E = Midpoint(A, B)
F = Midpoint(A, C)
G = Midpoint(B, C)
segEF = Segment(E, F)
lineEF = Line(E, F)
t = Slider(0, 1, 0.01)
P = Point(segEF, t)
triBPG = Polygon(B, P, G)
perimBPG = Distance(B, P) + Distance(P, G) + Distance(G, B)
G_prime = Reflect(G, lineEF)
lineBG_prime = Line(B, G_prime)
P_min = Intersect(lineEF, lineBG_prime)
minPerimBPG = Distance(B, G_prime) + Distance(G, B)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFilling(triABC, 0.05)
SetColor(E, "Blue")
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(F, "Blue")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(G, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
SetLineStyle(lineEF, 2)
SetColor(lineEF, "Gray")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triBPG, 0.2)
SetColor(G_prime, "Blue")
ShowLabel(G_prime, true)
SetLabelMode(G_prime, 1)
SetLineStyle(lineBG_prime, 2)
SetColor(lineBG_prime, "Gray")
SetColor(P_min, "Blue")
ShowLabel(P_min, true)
