MODE: 2D
SetAxesVisible(false, false)
SetGridVisible(false)
A = (0, sqrt(3))
B = (-1, 0)
C = (1, 0)

F = Midpoint(A, C)
G = Midpoint(B, C)
t = Slider(0, 1, 0.01)
P = Point(Segment(E, F), t)
triABC = Polygon(A, B, C)
segBP = Segment(B, P)
segPG = Segment(P, G)
segEF = Segment(E, F)
segBG = Segment(B, G)
segGC = Segment(G, C)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(E, "Blue")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(G, "Blue")

ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triABC, 0.1)
SetLineStyle(segEF, 2)
SetColor(segEF, "Gray")
SetLineStyle(segBG, 2)
SetColor(segBG, "Gray")
