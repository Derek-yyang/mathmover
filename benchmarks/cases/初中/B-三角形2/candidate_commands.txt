MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
B = (0, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (2, 0)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
A = (1, sqrt(3))
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
E = Midpoint(A, B)
ShowLabel(E, true)
SetLabelMode(E, 1)
F = Midpoint(A, C)
ShowLabel(F, true)
SetLabelMode(F, 1)
G = Midpoint(B, C)
ShowLabel(G, true)
SetLabelMode(G, 1)
pathEF = Segment(E, F)
t = Slider(0, 1, 0.01)
P = Point(pathEF, t)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
polyABC = Polygon(A, B, C)
SetFilling(polyABC, 0.05)
polyBPG = Polygon(B, P, G)
SetFilling(polyBPG, 0.1)
segBP = Segment(B, P)
segGP = Segment(G, P)
