MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 4)
B = (0, 0)
C = (4, 0)
D = (4, 4)
E = (1, 0)
polyABCD = Polygon(A, B, C, D)
segAB = Segment(A, B)
t = Slider(0, 1, 0.01)
F = Point(segAB, t)
G = E + ((x(F) - x(E)) / 2 + (y(F) - y(E)) * sqrt(3) / 2, -(x(F) - x(E)) * sqrt(3) / 2 + (y(F) - y(E)) / 2)
triEFG = Polygon(E, F, G)
segCG = Segment(C, G)
locusG = Segment((1/2, sqrt(3) / 2), (1/2 + 2 * sqrt(3), 2 + sqrt(3) / 2))
perpC = PerpendicularLine(C, locusG)
H = Intersect(locusG, perpC)
segCH = Segment(C, H)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFilling(polyABCD, 0.1)
SetColor(F, "Red")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(G, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
SetFilling(triEFG, 0.25)
SetLineStyle(locusG, 2)
SetColor(locusG, "Gray")
SetLineStyle(perpC, 2)
SetColor(perpC, "Gray")
SetColor(H, "Blue")
ShowLabel(H, true)
SetLabelMode(H, 1)
