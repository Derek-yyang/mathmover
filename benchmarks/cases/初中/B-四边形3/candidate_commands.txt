MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 4)
B = (0, 0)
C = (4, 0)
D = (4, 4)
E = (1, 0)
polyABCD = Polygon(A, B, C, D)
t = Slider(0, 1, 0.01)
F = Point(Segment(A, B), t)
G = (E + F) / 2 + (sqrt(3) / 2) * (y(F) - y(E), x(E) - x(F))
polyEFG = Polygon(E, F, G)
segCG = Segment(C, G)
locusG = Locus(G, F)
lenCG = Length(segCG)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFilling(polyABCD, 0.05)
SetValue(t, 0.45)
SetColor(F, "Red")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(G, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
SetFilling(polyEFG, 0.2)
SetLineStyle(locusG, 2)
SetColor(locusG, "Gray")
