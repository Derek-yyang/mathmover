MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 4)
B = (0, 0)
C = (4, 0)
D = (4, 4)
polyABCD = Polygon(A, B, C, D)
E = (1, 0)
segAB = Segment(A, B)
t = Slider(0, 1, 0.01)
F = Point(segAB, t)
G = Rotate(F, -60°, E)
polyEFG = Polygon(E, F, G)
segCG = Segment(C, G)
locB = Rotate(B, -60°, E)
locA = Rotate(A, -60°, E)
locG = Segment(locB, locA)
perpC = PerpendicularLine(C, locG)
Gmin = Intersect(locG, perpC)
cgNow = Distance(C, G)
minCG = Distance(C, Gmin)
txtNow = Text("CG = " + cgNow, (4.4, 3.8))
txtMin = Text("CG_min = " + minCG, (4.4, 3.2))
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFilling(polyABCD, 0.1)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetLineThickness(segAB, 4)
SetColor(F, "Red")
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(G, "Blue")
ShowLabel(G, true)
SetLabelMode(G, 1)
SetFilling(polyEFG, 0.25)
SetColor(segCG, "Red")
SetLineThickness(segCG, 3)
SetColor(locB, "Gray")
ShowLabel(locB, false)
SetColor(locA, "Gray")
ShowLabel(locA, false)
SetColor(locG, "Gray")
SetLineStyle(locG, 2)
SetColor(perpC, "Gray")
SetLineStyle(perpC, 2)
SetColor(Gmin, "Blue")
ShowLabel(Gmin, true)
SetLabelMode(Gmin, 1)
