SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
O = (0, 0)
lineBC = Line(B, C)
axis = Line((1 / 2, 0), (1 / 2, 1))

hLine = Line((0, f(m)), (1, f(m)))
N = Intersect(vLine, lineBC)
M = Intersect(hLine, axis)
segPN = Segment(P, N)
segPM = Segment(P, M)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)

SetLabelMode(B, 1)

ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(lineBC, "Blue")
SetColor(axis, "Gray")
SetLineStyle(axis, 2)
SetColor(P, "Red")

SetLabelMode(P, 1)
SetColor(vLine, "Red")
SetLineStyle(vLine, 2)

SetColor(N, "Black")
