MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
symAxis = Line((1/2, -1), (1/2, 7))
BC = Segment(B, C)
m = Slider(1/2, 3, 0.01)
P = (m, f(m))
linePM = Line(P, (m+1, f(m)))
M = Intersect(linePM, symAxis)
segPM = Segment(P, M)
linePN = Line(P, (m, f(m)-1))
N = Intersect(linePN, BC)
segPN = Segment(P, N)
SetColor(f, "Black")
SetLineThickness(f, 3)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(symAxis, "Black")
SetLineStyle(symAxis, 2)
SetColor(BC, "Gray")
SetLineThickness(BC, 2)
SetValue(m, (1+sqrt(5))/2)
SetColor(P, "Red")
SetPointSize(P, 4)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(linePM, 1, false)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPM, "Red")
SetLineThickness(segPM, 3)
SetVisibleInView(linePN, 1, false)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segPN, "Red")
SetLineThickness(segPN, 3)
