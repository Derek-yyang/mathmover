MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
A = (-2, 0)
B = (3, 0)
C = (0, 6)
axisSym = Line(1, 0, x)
lineBC = Line(B, C)
t = Slider(0.5, 3, 0.01)
P = (t, f(t))
N = (t, 0)
segPN = Segment(P, N)
M = (0, f(t))
segPM = Segment(P, M)
eq1 = PM == (1/2) * PN
E = (1, 4)
NlineBC = Intersect(lineBC, Line(P, (t, 0)))
polyAPBC = Polygon(A, P, B, C)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(axisSym, "Black")
SetLineStyle(axisSym, 2)
SetColor(lineBC, "Black")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetLineStyle(N, 1)
SetColor(N, "Red")
SetPointSize(N, 3)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segPN, "Red")
SetLineStyle(segPN, 2)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPM, "Red")
SetLineStyle(segPM, 2)
SetConditionToShowObject(eq1, false)
SetColor(E, "Black")
SetPointSize(E, 3)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetColor(NlineBC, "Blue")
ShowLabel(NlineBC, true)
SetLabelMode(NlineBC, 1)
SetFilling(polyAPBC, 0.1)
SetColor(polyAPBC, "Blue")
SetConditionToShowObject(polyAPBC, false)
