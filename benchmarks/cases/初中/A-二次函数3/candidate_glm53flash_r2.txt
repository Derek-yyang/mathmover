N = Intersect(linePN, lineBC)
segPN = Segment(P, N)
linePM = PerpendicularLine(P, yAxis)
M = Intersect(linePM, axisSym)
segPM = Segment(P, M)
lenPM = Distance(P, M)
lenPN = Distance(P, N)
