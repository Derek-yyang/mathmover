SetFixed(B, true)

SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)

SetColor(lineBC, "Gray")
SetLineStyle(lineBC, 1)
SetColor(axis_sym, "Gray")
SetLineStyle(axis_sym, 1)

SetLabelMode(N, 1)

SetColor(M, "Red")

SetLabelMode(M, 1)
