segPM = Segment(P, M)

ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(axisSym, "Gray")
SetLineStyle(axisSym, 2)
SetColor(lineBC, "Blue")
SetColor(P, "Red")

SetLabelMode(P, 1)

SetLabelMode(N, 1)
SetColor(M, "Green")

SetColor(segPN, "Red")
