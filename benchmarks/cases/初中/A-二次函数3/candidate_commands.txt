MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
O = (0, 0)
lineBC = Line(B, C)
segmentBC = Segment(B, C)
symAxis = Line((1/2, 0), (1/2, 1))
xLine = Line(O, (1, 0))
yLine = Line(O, (0, 1))
m = Slider(1/2, 3, 0.01)
P = (m, f(m))
linePN = PerpendicularLine(P, xLine)
N = Intersect(linePN, lineBC)
linePM = PerpendicularLine(P, yLine)
M = Intersect(linePM, symAxis)
segmentPN = Segment(P, N)
segmentPM = Segment(P, M)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Black")
SetPointSize(O, 4)
SetColor(lineBC, "Black")
SetVisibleInView(lineBC, false)
SetColor(segmentBC, "Black")
SetColor(symAxis, "Blue")
SetLineStyle(symAxis, 2)
SetColor(xLine, "Gray")
SetColor(yLine, "Gray")
SetVisibleInView(xLine, false)
SetVisibleInView(yLine, false)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetValue(m, (1 + sqrt(5)) / 2)
SetColor(N, "Black")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetPointSize(N, 4)
SetColor(M, "Black")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetPointSize(M, 4)
SetColor(segmentPN, "Red")
SetColor(segmentPM, "Red")
SetVisibleInView(linePN, false)
SetVisibleInView(linePM, false)
