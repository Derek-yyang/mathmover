MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
lBC = Line(B, C)
sym = x = 1/2
t = Slider(0.5, 3, 0.01)
P = (t, f(t))
N = Intersect(lBC, Line(P, (t, 0)))
M = (1/2, f(t))
segPN = Segment(P, N)
segPM = Segment(P, M)
Polygon(P, N, (t, 0))
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(lBC, "Blue")
SetLineStyle(lBC, 1)
SetColor(sym, "Gray")
SetLineStyle(sym, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPN, "Red")
SetLineStyle(segPN, 1)
SetColor(segPM, "Red")
SetLineStyle(segPM, 1)
SetFilling(%, 0.3)
SetColor(%, "Blue")
