MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
SetColor(f, "Black")
A = (-2, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, 6)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
lineBC = Line(B, C)
SetColor(lineBC, "Black")
axisSym = Line((1/2, 0), (1/2, 1))
SetColor(axisSym, "Black")
SetLineStyle(axisSym, 2)
m = Slider(1/2, 3, 0.05)
P = (m, f(m))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
lineX = Line((m, 0), (m, 1))
SetColor(lineX, "Gray")
SetLineStyle(lineX, 2)
N = Intersect(lineBC, lineX)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
lineY = Line((0, f(m)), (1, f(m)))
SetColor(lineY, "Gray")
SetLineStyle(lineY, 2)
M = Intersect(axisSym, lineY)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
segPN = Segment(P, N)
SetColor(segPN, "Red")
segPM = Segment(P, M)
SetColor(segPM, "Red")
