MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
SetColor(f, "Black")
A = (-2, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, 6)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
lineBC = Line(B, C)
SetColor(lineBC, "Green")
SetLineStyle(lineBC, 2)
segSym = Segment((0.5, -1), (0.5, 7))
SetColor(segSym, "Gray")
SetLineStyle(segSym, 2)
m = Slider(0.5, 3, 0.05)
P = (m, f(m))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
segVert = Segment((m, 0), P)
SetColor(segVert, "Red")
N = Intersect(lineBC, segVert)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
segHoriz = Segment((0, f(m)), P)
SetColor(segHoriz, "Blue")
M = Intersect(segSym, segHoriz)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
segPN = Segment(P, N)
SetColor(segPN, "Red")
segPM = Segment(P, M)
SetColor(segPM, "Blue")
