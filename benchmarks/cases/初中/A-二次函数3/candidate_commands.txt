MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
O = (0, 0)
segBC = Segment(C, B)
V = Vertex(f)
axis = PerpendicularLine(V, xAxis)
m = Slider(1/2, 3, 0.01)
P = (m, f(m))
linePM = PerpendicularLine(P, yAxis)
linePN = PerpendicularLine(P, xAxis)
M = Intersect(linePM, axis)
N = Intersect(linePN, segBC)
segPM = Segment(P, M)
segPN = Segment(P, N)
E = Intersect(linePM, yAxis)
segEM = Segment(E, M)
H = Intersect(linePN, xAxis)
segNH = Segment(N, H)
SetColor(f, "Black")
SetPointSize(A, 4)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetPointSize(B, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetPointSize(C, 4)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetPointSize(O, 4)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(segBC, "Green")
SetColor(V, "Black")
SetColor(axis, "Gray")
SetLineStyle(axis, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(linePM, 1, false)
SetVisibleInView(linePN, 1, false)
SetPointSize(M, 4)
SetColor(M, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetPointSize(N, 4)
SetColor(N, "Blue")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segPM, "Blue")
SetColor(segPN, "Blue")
ShowLabel(E, false)
SetColor(segEM, "Blue")
ShowLabel(H, false)
SetColor(segNH, "Blue")
