MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (-2, 0)
B = (3, 0)
C = (0, 6)
f(x) = -x^2 + x + 6
axisSym = Line((0.5, 0), (0.5, 10))
m = Slider(0.5, 3, 0.01)
P = (m, f(m))
N = (m, -2*m + 6)
M = (0.5, f(m))
linePN = Segment(P, N)
linePM = Segment(P, M)
lineBC = Line(B, C)
val_m = 2
P0 = (val_m, f(val_m))
N0 = (val_m, -2*val_m + 6)
M0 = (0.5, f(val_m))
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(f, "Black")
SetColor(axisSym, "Gray")
SetLineStyle(axisSym, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(linePN, "Red")
SetLineStyle(linePN, 2)
SetColor(linePM, "Red")
SetLineStyle(linePM, 2)
SetColor(lineBC, "Black")
SetLineStyle(lineBC, 3)
SetValue(m, val_m)
SetColor(P0, "Red")
ShowLabel(P0, true)
SetLabelMode(P0, 1)
SetColor(N0, "Red")
ShowLabel(N0, true)
SetLabelMode(N0, 1)
SetColor(M0, "Red")
ShowLabel(M0, true)
SetLabelMode(M0, 1)
