MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^(2) + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
V = Vertex(f)
axisSym = PerpendicularLine(V, xAxis)
m = Slider(0.5, 3, 0.05)
P = (m, f(m))
perpPN = PerpendicularLine(P, xAxis)
BC = Segment(B, C)
N = Intersect(perpPN, BC)
segPN = Segment(P, N)
perpPM = PerpendicularLine(P, yAxis)
M = Intersect(perpPM, axisSym)
segPM = Segment(P, M)
lenPM = Distance(P, M)
lenPN = Distance(P, N)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(V, false)
SetColor(axisSym, "Gray")
SetLineStyle(axisSym, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(perpPN, "Green")
SetLineStyle(perpPN, 1)
SetColor(BC, "Black")
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segPN, "Green")
SetColor(perpPM, "Green")
SetLineStyle(perpPM, 1)
SetColor(M, "Black")

ShowLabel(M, true)

SetColor(segPM, "Blue")
