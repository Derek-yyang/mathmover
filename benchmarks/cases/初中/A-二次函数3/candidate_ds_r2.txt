MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
lineBC = Line(B, C)
axis_sym = Line((1/2, 0), (1/2, 7))
m = Slider(1/2, 3, 0.05)
P = (m, f(m))
linePM = Line(P, (0, f(m)))
M = Intersect(linePM, axis_sym)
linePN = Line(P, (m, 0))
N = Intersect(linePN, lineBC)
segPM = Segment(M, P)
segPN = Segment(P, N)
triPMN = Polygon(M, P, N)
m_sol = (1 + sqrt(5)) / 2
P_sol = (m_sol, f(m_sol))
txt_m = Text("m = " + ((1 + sqrt(5)) / 2), (m_sol, f(m_sol) - 1))
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(lineBC, "Blue")
SetColor(axis_sym, "Gray")
SetLineStyle(axis_sym, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePM, "Red")
SetLineStyle(linePM, 3)
SetColor(M, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(linePN, "Red")
SetLineStyle(linePN, 3)
SetColor(N, "Blue")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segPM, "Red")
SetLineThickness(segPM, 3)
SetColor(segPN, "Blue")
SetLineThickness(segPN, 3)
SetFilling(triPMN, 0.2)
SetColor(triPMN, "Purple")
SetColor(P_sol, "Green")
SetPointSize(P_sol, 5)
ShowLabel(P_sol, true)
SetLabelMode(P_sol, 1)
SetColor(txt_m, "Green")
