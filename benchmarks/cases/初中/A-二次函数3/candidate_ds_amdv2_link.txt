MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (-2, 0)
B = (3, 0)
C = (0, 6)
f(x) = -x^2 + x + 6
lineBC = Line(B, C)
axisSym = Line((0.5, -5), (0.5, 10))
m = Slider(0.5, 3, 0.01)
P = (m, f(m))
N = (m, 0)
N = Intersect(PerpendicularLine(P, xAxis), lineBC)
M = Intersect(PerpendicularLine(P, yAxis), axisSym)
segPM = Segment(P, M)
segPN = Segment(P, N)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(f, "Black")
SetColor(lineBC, "Gray")
SetLineStyle(lineBC, 1)
SetColor(axisSym, "Gray")
SetLineStyle(axisSym, 3)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(N, false)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPM, "Red")
SetLineStyle(segPM, 1)
SetColor(segPN, "Red")
SetLineStyle(segPN, 1)
