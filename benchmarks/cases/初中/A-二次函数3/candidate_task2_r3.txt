MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
lineBC = Line(B, C)
axis = Line((1/2, 0), (1/2, 1))
m = Slider(0.5, 3, 0.01)
P = (m, f(m))
l1 = Line(P, (P.x, 0))
N = Intersect(l1, lineBC)
l2 = Line(P, (0, P.y))
M = Intersect(l2, axis)
segPN = Segment(P, N)
segPM = Segment(P, M)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(lineBC, "Black")
SetColor(axis, "Blue")
SetLineStyle(axis, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(l1, "Gray")
SetLineStyle(l1, 3)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(l2, "Gray")
SetLineStyle(l2, 3)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPN, "Red")
SetColor(segPM, "Red")
