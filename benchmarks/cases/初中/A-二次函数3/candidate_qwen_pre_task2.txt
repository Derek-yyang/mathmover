MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
SetColor(f, "Black")
A = (-2, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, 6)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
lineBC = Line(B, C)
SetColor(lineBC, "Blue")
axis = Line((1/2, 0), (1/2, 1))
SetColor(axis, "Gray")
SetLineStyle(axis, 2)
m = Slider(1/2, 3, 0.05)
P = (m, f(m))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
v = Line(P, (m, 0))
SetVisibleInView(v, false, false)
h = Line(P, (0, f(m)))
SetVisibleInView(h, false, false)
N = Intersect(v, lineBC)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
M = Intersect(h, axis)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
segPN = Segment(P, N)
SetColor(segPN, "Black")
segPM = Segment(P, M)
SetColor(segPM, "Black")
