MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
V = Extremum(f)
lineBC = Line(B, C)
axis = Line((1/2, 0), (1/2, 1))
m = Slider(0.5, 3, 0.05)
P = (m, f(m))
v = PerpendicularLine(P, xAxis)
h = PerpendicularLine(P, yAxis)
N = Intersect(v, lineBC)
M = Intersect(h, axis)
segPN = Segment(P, N)
segPM = Segment(P, M)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(V, true)
ShowLabel(V, true)
SetLabelMode(V, 1)
SetColor(lineBC, "Blue")
SetColor(axis, "Gray")
SetLineStyle(axis, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPN, "Red")
SetColor(segPM, "Red")
