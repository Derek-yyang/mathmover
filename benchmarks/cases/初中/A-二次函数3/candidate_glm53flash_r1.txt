MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 6
A = (-2, 0)
B = (3, 0)
C = (0, 6)
segBC = Segment(B, C)
symAxis = PerpendicularBisector(A, B)
m = Slider(1/2, 3, 0.05)
P = (m, f(m))
linePN = PerpendicularLine(P, xAxis)
N = Intersect(linePN, segBC)
linePM = PerpendicularLine(P, yAxis)
M = Intersect(linePM, symAxis)
segPN = Segment(P, N)
segPM = Segment(P, M)
footX = (m, 0)
segPNext = Segment(N, footX)
footY = (0, f(m))
segPMext = Segment(M, footY)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(segBC, "Green")
SetColor(symAxis, "Gray")
SetLineStyle(symAxis, 2)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePN, "Gray")
SetLineStyle(linePN, 2)
SetColor(N, "Black")
SetPointSize(N, 4)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(linePM, "Gray")
SetLineStyle(linePM, 2)
SetColor(M, "Black")
SetPointSize(M, 4)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segPN, "Blue")
SetColor(segPM, "Magenta")
SetColor(segPNext, "Blue")
SetColor(segPMext, "Magenta")
