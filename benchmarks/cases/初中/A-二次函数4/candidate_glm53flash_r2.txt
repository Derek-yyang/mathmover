MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = -x^2 + x - 1
f(x) = -x^2 + 3x + 4
k = Slider(-3, 3, 0.1)
l = Line((0, 2), (1, k + 2))
A = (-2 / k, 0)
B = (0, 2)
C = Intersect(l, f, 1)
D = Intersect(l, f, 2)
E1 = (-1, 0)
E2 = (4, 0)
t = Slider(-1, 4, 0.05)
P = (t, f(t))
triPCD = Polygon(P, C, D)
SetColor(g, "Gray")
SetColor(f, "Black")
SetValue(k, -0.5)
SetColor(l, "Green")
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(E1, true)
ShowLabel(E1, true)
SetLabelMode(E1, 1)
SetFixed(E2, true)
ShowLabel(E2, true)
SetLabelMode(E2, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triPCD, 0.3)
SetColor(triPCD, "Red")
