MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
// 1. 定义抛物线解析式
// 根据题目：与 y = -x^2 + x - 1 形状相同且开口向下，a = -1
// 经过 (-1, 0) 和 (4, 0)，解析式为 f(x) = -(x+1)(x-4) = -x^2 + 3x + 4
f(x) = -x^2 + 3*x + 4
SetColor(f, "Black")
// 2. 定义参数 k 和直线 g(x) = kx + 2
// k 设置为滑块，默认值为 2 (对应第2小问)
k = Slider(-5, 5, 0.1)
SetValue(k, 2)
g(x) = k*x + 2
SetColor(g, "Blue")
SetLineStyle(g, 2)
// 3. 计算直线与坐标轴的交点 A, B
// A 是直线与 x 轴交点：kx + 2 = 0 => x = -2/k
A = (-2 / k, 0)
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(A, true)
// B 是直线与 y 轴交点：(0, 2)
B = (0, 2)
SetColor(B, "Black")
SetPointSize(B, 4)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(B, true)
// 4. 定义线段 AB (第3问涉及)
segAB = Segment(A, B)
SetColor(segAB, "Green")
SetThickness(segAB, 5)
// 5. 计算直线与抛物线的交点 C, D
// 解方程 -x^2 + 3x + 4 = kx + 2 => x^2 + (k-3)x - 2 = 0
// 根据求根公式得到交点横坐标
x_c = (3 - k - sqrt((k - 3)^2 + 8)) / 2
x_d = (3 - k + sqrt((k - 3)^2 + 8)) / 2
C = (x_c, g(x_c))
D = (x_d, g(x_d))
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(C, true)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(D, true)
// 6. 定义动点 P (第2问：在直线之上的抛物线上)
// 使用主从滑块联动：t 的范围依赖于交点 C 和 D 的横坐标
t = Slider(x_c, x_d, 0.01)
P = (t, f(t))
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFixed(P, true)
// 7. 构建三角形 PCD 并设置填充
triPCD = Polygon(P, C, D)
SetColor(triPCD, "Red")
SetFilling(triPCD, 0.3)
// 8. 标记坐标原点及抛物线零点
O = (0, 0)
SetColor(O, "Black")
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(O, true)
X1 = (-1, 0)
SetColor(X1, "Gray")
SetPointSize(X1, 3)
ShowLabel(X1, true)
SetLabelMode(X1, 1)
SetFixed(X1, true)
X2 = (4, 0)
SetColor(X2, "Gray")
SetPointSize(X2, 3)
ShowLabel(X2, true)
SetLabelMode(X2, 1)
SetFixed(X2, true)
