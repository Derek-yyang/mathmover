MODE: 2D
k = Slider(-5, 5, 0.1)
f(x) = -x^2 + 3x + 4
g(x) = k*x + 2
A = (-2/k, 0)
B = (0, 2)
xC = (3 - k - sqrt((k - 3)^2 + 8)) / 2
xD = (3 - k + sqrt((k - 3)^2 + 8)) / 2
C = (xC, g(xC))
D = (xD, g(xD))
t = Slider(xC, xD, 0.1)
P = (t, f(t))
poly = Polygon(P, C, D)
segAB = Segment(A, B)
SetColor(f, "Black")
SetLineThickness(f, 3)
SetColor(g, "Black")
SetLineStyle(g, "Solid")
SetColor(A, "Black")
SetPointSize(A, 4)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
SetPointSize(P, 4)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(poly, "Blue")
SetFilling(poly, 0.3)
SetColor(segAB, "Black")
SetLineStyle(segAB, "Solid")
SetAxesVisible(true, true)
SetGridVisible(true)
