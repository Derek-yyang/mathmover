MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
k = Slider(-3, 3, 0.05)
f_line(x) = k*x + 2
B = (0, 2)
X1 = (Root(f, -2, 0), 0)
X2 = (Root(f, 3, 5), 0)
A = (Root(f_line, -10, 10), 0)
C = Intersect(f_line, f, 1)
D = Intersect(f_line, f, 2)
segAB = Segment(A, B)
t = Slider(-1, 4, 0.05)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetColor(f_line, "Green")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetFixed(X1, true)
ShowLabel(X1, true)
SetLabelMode(X1, 1)
SetColor(X1, "Black")
SetPointSize(X1, 4)
SetFixed(X2, true)
ShowLabel(X2, true)
SetLabelMode(X2, 1)
SetColor(X2, "Black")
SetPointSize(X2, 4)
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segAB, "Green")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(polyPCD, "Red")
