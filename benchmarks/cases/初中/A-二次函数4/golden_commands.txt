MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
k = Slider(-3, 3, 0.05)
SetValue(k, 2)
g(x) = k * x + 2
# x_c = ((3 - k) - sqrt((k - 3)^2 + 8)) / 2
# x_d = ((3 - k) + sqrt((k - 3)^2 + 8)) / 2
x_c = ((3 - k) - sqrt((k - 3)^2 + 8)) / 2
x_d = ((3 - k) + sqrt((k - 3)^2 + 8)) / 2
C = (x_c, f(x_c))
D = (x_d, f(x_d))
A = (-2 / k, 0)
B = (0, 2)
t = Slider(x_c, x_d, 0.01)
SetValue(t, 0.5)
P = (t, f(t))
poly_PCD = Polygon(P, C, D)
SetColor(f, "Blue")
SetColor(g, "Green")
SetColor(poly_PCD, "Orange")
SetFilling(poly_PCD, 0.3)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
Segment_CD = Segment(C, D)
SetLineStyle(Segment_CD, "Dashed")
SetColor(Segment_CD, "Gray")
Segment_PC = Segment(P, C)
SetLineStyle(Segment_PC, "Dashed")
SetColor(Segment_PC, "Gray")
Segment_PD = Segment(P, D)
SetLineStyle(Segment_PD, "Dashed")
SetColor(Segment_PD, "Gray")
