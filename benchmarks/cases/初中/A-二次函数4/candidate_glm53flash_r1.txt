MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
q(x) = -x^2 + x - 1
E = (-1, 0)
F = (4, 0)
k = Slider(-4, 4, 0.05)
g(x) = k*x + 2
A = (Root(g, -20, 20), 0)
B = (0, 2)
C = Element(Intersect(f, g), 1)
D = Element(Intersect(f, g), 2)
segmentAB = Segment(A, B)
t = Slider(-1, 4, 0.05)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetColor(q, "Gray")
SetLineStyle(q, 2)
SetFixed(E, true)
ShowLabel(E, true)
SetLabelMode(E, 1)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetValue(k, 2)
SetColor(g, "Green")
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segmentAB, "Blue")
SetLineStyle(segmentAB, 2)
SetValue(t, 0.5)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
