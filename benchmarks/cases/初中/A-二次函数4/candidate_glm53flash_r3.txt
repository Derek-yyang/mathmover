MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
f0(x) = -x^2 + x - 1
O = (0, 0)
G = (Root(f, -2, 0), 0)
H = (Root(f, 3, 5), 0)
k = Slider(-3, 3, 0.05)
g(x) = k*x + 2
A = Intersect(g, xAxis, 1)
B = Intersect(g, yAxis, 1)
C = Intersect(g, f, 1)
D = Intersect(g, f, 2)
lineAB = Segment(A, B)
t = Slider(-1, 2, 0.05)
P = (t, f(t))
triPCD = Polygon(P, C, D)
areaPCD = Area(triPCD)
SetColor(f, "Black")
SetColor(f0, "Gray")
SetLineStyle(f0, 2)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(G, "Black")
SetPointSize(G, 4)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(H, "Black")
SetPointSize(H, 4)
SetFixed(H, true)
ShowLabel(H, true)
SetLabelMode(H, 1)
SetValue(k, 2)
SetColor(g, "Green")
SetColor(A, "Black")
SetPointSize(A, 4)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(lineAB, "Blue")
SetLineStyle(lineAB, 2)
SetValue(t, 0.5)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triPCD, 0.3)
SetColor(triPCD, "Red")
