MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
k = Slider(-3, 3, 0.1, -0.4)
g(x) = k*x + 2
h(x) = f(x) - g(x)
O = (0, 0)
N1 = (-1, 0)
N2 = (4, 0)
B = (0, 2)
A = (-2/k, 0)
C = (Root(h, -100, 0), g(Root(h, -100, 0)))
D = (Root(h, 0, 100), g(Root(h, 0, 100)))
segAB = Segment(A, B)
t = Slider(-1, 4, 0.05)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetColor(g, "Green")
SetLineStyle(g, 2)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(N1, true)
ShowLabel(N1, true)
SetLabelMode(N1, 1)
SetFixed(N2, true)
ShowLabel(N2, true)
SetLabelMode(N2, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segAB, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
