MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
k = Slider(-3, 3, 0.1)
SetValue(k, -0.5)
f(x) = -x^2 + 3x + 4
SetColor(f, "Black")
SetLineStyle(f, "Solid")
g(x) = k * x + 2
SetColor(g, "Blue")
B = (0, 2)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
A = (-2 / k, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
x_C = (3 - k - sqrt((k - 3)^2 + 8)) / 2
x_D = (3 - k + sqrt((k - 3)^2 + 8)) / 2
C = (x_C, k * x_C + 2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
D = (x_D, k * x_D + 2)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
t = Slider(x_C, x_D, 0.1)
P = (t, f(t))
SetFixed(P, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
poly_PCD = Polygon(P, C, D)
SetColor(poly_PCD, "Green")
SetFilling(poly_PCD, 0.3)
SetLineStyle(poly_PCD, "Solid")
seg_AB = Segment(A, B)
SetColor(seg_AB, "Blue")
SetLineStyle(seg_AB, "Solid")
ZoomIn(1, 1)
