MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
SetColor(f, "Black")
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
M = (-1, 0)
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
N = (4, 0)
SetFixed(N, true)
ShowLabel(N, true)
SetLabelMode(N, 1)
B = (0, 2)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
k = Slider(-3, -0.3, 0.05)
line = Line(B, (1, k + 2))
SetColor(line, "Green")
SetLineStyle(line, 2)
A = Intersect(line, xAxis, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetLabelMode(A, 1)
ShowLabel(A, true)
C = Intersect(line, f, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
D = Intersect(line, f, 2)
SetColor(D, "Black")
SetPointSize(D, 4)
SetLabelMode(D, 1)
ShowLabel(D, true)
segAB = Segment(A, B)
SetColor(segAB, "Blue")
t = Slider(-1, 4, 0.05)
P = (t, f(t))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
polyPCD = Polygon(P, C, D)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
