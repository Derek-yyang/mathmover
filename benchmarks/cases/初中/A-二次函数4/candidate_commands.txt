MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x
g(x) = 2x + 2
A = (4, 0)
B = (-1, 0)
C = Intersect(f, g, 1)
D = Intersect(f, g, 2)
t = Slider(-1, 4, 0.05)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetColor(g, "Green")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
SetColor(D, "Black")
SetPointSize(D, 4)
SetLabelMode(D, 1)
ShowLabel(D, true)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.25)
SetColor(polyPCD, "Red")
