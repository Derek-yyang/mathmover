MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
X1 = (Root(f, -2, 0), 0)
X2 = (Root(f, 3, 5), 0)
B = (0, 2)
k = Slider(-3, 3, 0.1)
lineK = k*x + 2
A = Intersect(lineK, xAxis)
C = Intersect(lineK, f, 1)
D = Intersect(lineK, f, 2)
segAB = Segment(A, B)
t = Slider(-1, 2, 0.05)
P = (t, f(t))
triPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetColor(X1, "Black")
SetPointSize(X1, 4)
SetFixed(X1, true)
ShowLabel(X1, true)
SetLabelMode(X1, 1)
SetColor(X2, "Black")
SetPointSize(X2, 4)
SetFixed(X2, true)
ShowLabel(X2, true)
SetLabelMode(X2, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
SetValue(k, 2)
SetColor(lineK, "Green")
SetLineStyle(lineK, 2)
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segAB, "Blue")
SetThickness(segAB, 3)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triPCD, 0.3)
SetColor(triPCD, "Red")
