MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
k = Slider(-5, 5, 0.1)
g(x) = k*x + 2
B = (0, 2)
A = (-2/k, 0)
R1 = Root(f, -2, 0)
R2 = Root(f, 3, 5)
X1 = (R1, 0)
X2 = (R2, 0)
pts = Intersect(f, g)
C = Element(pts, 1)
D = Element(pts, 2)
segAB = Segment(A, B)
t = Slider(-1.5, 5, 0.01)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetFixed(k, false)
ShowLabel(k, true)
SetLabelMode(k, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(X1, true)
ShowLabel(X1, true)
SetLabelMode(X1, 1)
SetFixed(X2, true)
ShowLabel(X2, true)
SetLabelMode(X2, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
SetColor(D, "Black")
SetPointSize(D, 4)
SetLabelMode(D, 1)
ShowLabel(D, true)
SetColor(segAB, "Blue")
SetFixed(t, false)
ShowLabel(t, true)
SetLabelMode(t, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
