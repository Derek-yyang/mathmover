MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + x + 2
g(x) = a*x^2 + b*x + c
A = (4, 0)
B = (-1, 0)
O = (0, 0)
C = Intersect(Line(B, A), f, 1)
a = 1
b = -3
c = -4
h(x) = If(x >= -1, If(x <= 4, g(x)))
l = Line(B, A)
k = Slider(-5, 5, 0.1)
line_k = Line((0, k*0), (1, k*1))
ptB = Intersect(line_k, h, 1)
ptCpt = Intersect(line_k, h, 2)
ptD = Intersect(line_k, h, 3)
t = Slider(-5, 5, 0.05)
P = (t, h(t))
polyPCD = Polygon(P, ptCpt, ptD)
maxArea = Maximum(Area(polyPCD), t)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
SetColor(h, "Red")
SetLineStyle(h, 3)
SetColor(l, "Green")
SetLineStyle(l, 2)
SetValue(k, 2)
SetColor(line_k, "Blue")
SetColor(ptB, "Black")
SetPointSize(ptB, 4)
SetLabelMode(ptB, 1)
ShowLabel(ptB, true)
SetColor(ptCpt, "Black")
SetPointSize(ptCpt, 4)
SetLabelMode(ptCpt, 1)
ShowLabel(ptCpt, true)
SetColor(ptD, "Black")
SetPointSize(ptD, 4)
SetLabelMode(ptD, 1)
ShowLabel(ptD, true)
SetValue(t, 0.5)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
