MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
g(x) = -x^2 + x
f(x) = -x^2 + 2x + 3
A = (4, 0)
B = (-1, 0)
k = Slider(-3, 5, 0.1)
lineL(x) = k*x + 2
C = Intersect(lineL, f, 1)
D = Intersect(lineL, f, 2)
k2 = 2
lineL2(x) = k2*x + 2
C2 = Intersect(lineL2, f, 1)
D2 = Intersect(lineL2, f, 2)
t = Slider(0, 3, 0.05)
P = (t, f(t))
polyPCD = Polygon(P, C2, D2)
SetColor(g, "Black")
SetColor(f, "Red")
SetLineStyle(f, 3)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(lineL, "Blue")
SetColor(C, "Black")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(lineL2, "Green")
SetLineStyle(lineL2, 2)
SetColor(C2, "Black")
ShowLabel(C2, true)
SetLabelMode(C2, 1)
SetColor(D2, "Black")
ShowLabel(D2, true)
SetLabelMode(D2, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Yellow")
