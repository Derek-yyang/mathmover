MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
SetColor(f, "Black")
k = Slider(-2, 2, 0.05)
g(x) = kx + 2
SetColor(g, "Blue")
B = (0, 2)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
A = Intersect(g, xAxis)
SetColor(A, "Black")
SetPointSize(A, 4)
SetLabelMode(A, 1)
ShowLabel(A, true)
C = Intersect(f, g, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
D = Intersect(f, g, 2)
SetColor(D, "Black")
SetPointSize(D, 4)
SetLabelMode(D, 1)
ShowLabel(D, true)
segAB = Segment(A, B)
SetColor(segAB, "Green")
t = Slider(-1, 4, 0.05)
P = (t, f(t))
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
polyPCD = Polygon(P, C, D)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
