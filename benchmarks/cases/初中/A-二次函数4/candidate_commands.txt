MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
k = Slider(-6, 6, 0.1)
f(x) = -x^2 + 3*x + 4
X1 = (-1, 0)
X2 = (4, 0)
l: y = k*x + 2
A = Intersect(l, xAxis)
B = Intersect(l, yAxis)
segAB = Segment(A, B)
C = Intersect(l, f, 1)
D = Intersect(l, f, 2)
t = Slider(x(C), x(D), 0.05)
P = (t, f(t))
triPCD = Polygon(P, C, D)
SetValue(k, 2)
SetColor(f, "Black")
SetFixed(X1, true)
ShowLabel(X1, true)
SetLabelMode(X1, 1)
SetFixed(X2, true)
ShowLabel(X2, true)
SetLabelMode(X2, 1)
SetColor(l, "Green")
SetColor(A, "Black")
SetPointSize(A, 4)
SetLabelMode(A, 1)
ShowLabel(A, true)
SetColor(B, "Black")
SetPointSize(B, 4)
SetLabelMode(B, 1)
ShowLabel(B, true)
SetColor(segAB, "Orange")
SetColor(C, "Black")
SetPointSize(C, 4)
SetLabelMode(C, 1)
ShowLabel(C, true)
SetColor(D, "Black")
SetPointSize(D, 4)
SetLabelMode(D, 1)
ShowLabel(D, true)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(triPCD, 0.3)
SetColor(triPCD, "Red")
