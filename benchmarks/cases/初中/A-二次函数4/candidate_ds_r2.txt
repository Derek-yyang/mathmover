MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 3x + 4
Z1 = (-1, 0)
Z2 = (4, 0)
k = Slider(-5, 5, 0.1)
g(x) = k*x + 2
A = If(k != 0, (-2/k, 0), undefined)
B = (0, 2)
h(x) = f(x) - g(x)
C = Root(h, -5, 0)
D = Root(h, 0, 5)
segAB = Segment(A, B)
t = Slider(-1, 2, 0.01)
P = (t, f(t))
polyPCD = Polygon(P, C, D)
SetColor(f, "Black")
SetLineThickness(f, 3)
SetFixed(Z1, true)
ShowLabel(Z1, true)
SetLabelMode(Z1, 1)
SetFixed(Z2, true)
ShowLabel(Z2, true)
SetLabelMode(Z2, 1)
SetValue(k, 2)
SetColor(g, "Green")
SetLineStyle(g, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(segAB, "Blue")
SetLineStyle(segAB, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyPCD, 0.3)
SetColor(polyPCD, "Red")
