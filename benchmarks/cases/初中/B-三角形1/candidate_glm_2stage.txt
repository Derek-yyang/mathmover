MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
M = Midpoint(B, C)
D = Rotate(B, pi/2, M)
P = 2*D - A
locusP = Locus(P, C)
polyABC = Polygon(A, B, C)
polyBCD = Polygon(B, C, D)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetValue(t, 0.25)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(M, "Blue")
ShowLabel(M, false)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetLineStyle(locusP, 2)
SetColor(locusP, "Gray")
SetFilling(polyABC, 0.2)
SetFilling(polyBCD, 0.2)
