MODE: 2D
A = (0, 0)
B = (4, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
t = Slider(0, 1, 0.01)
C = Point(Circle(A, 2), t)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, 255, 0, 0)
M = Midpoint(B, C)
D = M + 0.5 * (-(y(B) - y(C)), x(B) - x(C))
ShowLabel(D, true)
SetLabelMode(D, 1)
P = 2 * D - A
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, 0, 0, 255)
poly_ABC = Polygon(A, B, C)
poly_BCD = Polygon(B, C, D)
seg_AD = Segment(A, D)
seg_DP = Segment(D, P)
seg_PB = Segment(P, B)
seg_AC = Segment(A, C)
seg_CB = Segment(C, B)
seg_DB = Segment(D, B)
seg_DC = Segment(D, C)
seg_AB = Segment(A, B)
SetFilling(poly_ABC, 0.05)
SetFilling(poly_BCD, 0.05)
SetColor(poly_ABC, 100, 100, 255)
SetColor(poly_BCD, 255, 150, 100)
