MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
D = Midpoint(B, C) + 0.5 * (y(C) - y(B), -(x(C) - x(B)))
P = 2 * D - A
polyBCD = Polygon(B, C, D)
polyABC = Polygon(A, B, C)
segAP = Segment(A, P)
segPB = Segment(P, B)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyBCD, 0.2)
SetFilling(polyABC, 0.1)
