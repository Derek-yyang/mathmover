MODE: 2D
A = (0, 0)
B = (4, 0)
SetFixed(A, true)
SetFixed(B, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
ShowLabel(B, true)
SetLabelMode(B, 1)
t = Slider(0, 2pi, 0.01)
C = (2cos(t), 2sin(t))
SetColor(C, "blue")
ShowLabel(C, true)
SetLabelMode(C, 1)
D = Dilate(Rotate(C, -45°, B), 1/sqrt(2), B)
ShowLabel(D, true)
SetLabelMode(D, 1)
P = 2*D
SetColor(P, "red")
ShowLabel(P, true)
SetLabelMode(P, 1)
Segment(A, D)
Segment(D, P)
Segment(P, B)
Segment(B, C)
Segment(C, D)
Segment(D, B)
Circle_Locus = Circle((4, 4), 2*sqrt(2))
SetColor(Circle_Locus, "gray")
SetLineStyle(Circle_Locus, 2)
