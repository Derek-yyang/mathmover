MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
t = Slider(0, 2*pi, 0.01)
C = A + 2 * (cos(t), sin(t))
M = Midpoint(B, C)
vBC = B - C
D = M + 0.5 * (-(y(vBC)), x(vBC))
P = 2 * D - A
polyBCD = Polygon(B, C, D)
segAD = Segment(A, D)
segBP = Segment(B, P)
pathP = Locus(P, t)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyBCD, 0.2)
SetLineStyle(pathP, 2)
SetColor(pathP, "Gray")
