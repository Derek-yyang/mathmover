MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
M = Midpoint(B, C)
D = M + (sqrt(2) / 2) * (-(y(B) - y(C)), x(B) - x(C))
P = 2 * D - A
triBCD = Polygon(B, C, D)
triABC = Polygon(A, B, C)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetFilling(triBCD, 0.3)
SetFilling(triABC, 0.1)
