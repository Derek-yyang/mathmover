MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
M = Midpoint(B, C)
D = M + 0.5 * Rotate(C - B, -pi / 2)
P = 2 * D - A
pathP = Circle((8, 0), 2 * sqrt(2))
polyABC = Polygon(A, B, C)
polyBCD = Polygon(B, C, D)
segAB = Segment(A, B)
segAC = Segment(A, C)
segBC = Segment(B, C)
segCD = Segment(C, D)
segDB = Segment(D, B)
segAD = Segment(A, D)
segPD = Segment(P, D)
segPB = Segment(P, B)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetLineStyle(pathP, 2)
SetColor(pathP, "Gray")
SetFilling(polyABC, 0.05)
SetFilling(polyBCD, 0.05)
