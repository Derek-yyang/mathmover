MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (4, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
pathC = Circle(A, 2)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
M = Midpoint(B, C)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
D = M + 0.5 * (y(C) - y(B), -(x(C) - x(B)))
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Blue")
P = 2 * D - A
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
polyABC = Polygon(A, B, C)
SetFilling(polyABC, 0.05)
polyBCD = Polygon(B, C, D)
SetFilling(polyBCD, 0.05)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
