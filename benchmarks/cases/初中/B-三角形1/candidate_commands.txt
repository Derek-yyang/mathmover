MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
D = Midpoint(B, C) + 0.5 * (-(y(B) - y(C)), x(B) - x(C))
P = 2 * D - A
polyABC = Polygon(A, B, C)
polyBCD = Polygon(B, C, D)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetFilling(polyABC, 0.05)
SetFilling(polyBCD, 0.05)
