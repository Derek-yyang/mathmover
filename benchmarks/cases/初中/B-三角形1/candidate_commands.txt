MODE: 2D
A = (0, 0)
B = (4, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
t = Slider(0, 2pi, 0.01)
C = (2cos(t), 2sin(t))
SetColor(C, "red")
ShowLabel(C, true)
SetLabelMode(C, 1)
M = Midpoint(B, C)
D = M + 0.5 * Rotate(C - B, -90°)
ShowLabel(D, true)
SetLabelMode(D, 1)
P = Dilate(D, 2, A)
SetColor(P, "blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
c_traj = Circle((4, 4), 2sqrt(2))
SetLineStyle(c_traj, 2)
SetColor(c_traj, "gray")
