MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 1, 0.01)
C = Point(pathC, t)
M = Midpoint(B, C)
D = M + 0.5 * (y(C) - y(B), x(B) - x(C))
P = 2 * D - A
polyBCD = Polygon(B, C, D)
segCA = Segment(C, A)
segAB = Segment(A, B)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
angCDB = Angle(C, D, B)
locusP = Locus(P, C)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(M, "Blue")
ShowLabel(M, false)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetFilling(polyBCD, 0.1)
SetLineStyle(locusP, 2)
SetColor(locusP, "Gray")
