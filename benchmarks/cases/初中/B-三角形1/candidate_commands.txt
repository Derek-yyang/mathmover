MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 0)
B = (4, 0)
pathC = Circle(A, 2)
t = Slider(0, 2*pi, 0.01)
C = Point(pathC, t)
M = Midpoint(B, C)
D = M + 0.5 * (-(y(B) - y(C)), x(B) - x(C))
P = 2 * D - A
pathP = Locus(P, t)
segAB = Segment(A, B)
segAC = Segment(A, C)
segBC = Segment(B, C)
segBD = Segment(B, D)
segCD = Segment(C, D)
segAD = Segment(A, D)
segDP = Segment(D, P)
segPB = Segment(P, B)
triABC = Polygon(A, B, C)
triBCD = Polygon(B, C, D)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetLineStyle(pathC, 2)
SetColor(pathC, "Gray")
SetColor(C, "Red")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetConditionToShowObject(M, false)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetLineStyle(pathP, 2)
SetColor(pathP, "Gray")
SetFilling(triABC, 0.2)
SetFilling(triBCD, 0.2)
