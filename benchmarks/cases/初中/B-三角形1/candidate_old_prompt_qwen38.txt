D = Rotate(C, pi/2, M)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetPointColor(D, 255, 0, 0)
P = Reflect(A, D)
ShowLabel(P, true)
SetLabelMode(P, 1)
