MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
A = (Root(f, -2, 0), 0)
B = (Root(f, 2, 4), 0)
C = (0, 3)
Q = (0, 1)
lineQB = Line(Q, B)
t = Slider(-2, 3, 0.01)
M = (t, f(t))
lineMQ = Line(M, Q)
lineMB = Line(M, B)
angleMBQ = Angle(M, B, Q)
tanAngle = tan(angleMBQ)
tanVal = Slider(-10, 20, 0.01)
polyMBQ = Polygon(M, B, Q)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(lineQB, "Black")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(lineMQ, "Blue")
SetLineStyle(lineMQ, 2)
SetColor(lineMB, "Green")
SetColor(angleMBQ, "Purple")
ShowLabel(angleMBQ, true)
SetLabelMode(angleMBQ, 1)
SetColor(tanAngle, "Red")
ShowLabel(tanAngle, true)
SetLabelMode(tanAngle, 1)
SetColor(tanVal, "Blue")
SetFilling(polyMBQ, 0.2)
SetColor(polyMBQ, "Orange")
