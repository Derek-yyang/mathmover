MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
A = (Root(f, -3, 0), 0)
B = (Root(f, 2, 4), 0)
C = (0, f(0))
Q = (0, 1)
t = Slider(-1, 3, 0.05)
M = (t, f(t))
segBM = Segment(B, M)
segBQ = Segment(B, Q)
lineBQ = Line(B, Q)
angMBQ = Angle(M, B, Q)
line45a = Rotate(lineBQ, 45°, B)
line45b = Rotate(lineBQ, -45°, B)
M1 = Intersect(f, line45a, 1)
M2 = Intersect(f, line45b, 1)
lineHalfA = Rotate(lineBQ, atan(1/2), B)
lineHalfB = Rotate(lineBQ, -atan(1/2), B)
M3 = Intersect(f, lineHalfA, 1)
M4 = Intersect(f, lineHalfB, 1)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Black")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(segBM, "Red")
SetColor(segBQ, "Black")
SetColor(lineBQ, "Green")
SetLineStyle(lineBQ, 2)
SetColor(angMBQ, "Green")
ShowLabel(angMBQ, true)
SetLabelMode(angMBQ, 2)
SetColor(line45a, "Blue")
SetLineStyle(line45a, 2)
SetColor(line45b, "Blue")
SetLineStyle(line45b, 2)
SetColor(M1, "Blue")
SetPointSize(M1, 4)
ShowLabel(M1, true)
SetLabelMode(M1, 1)
SetColor(M2, "Blue")
SetPointSize(M2, 4)
ShowLabel(M2, true)
SetLabelMode(M2, 1)
SetColor(lineHalfA, "Orange")
SetLineStyle(lineHalfA, 2)
SetColor(lineHalfB, "Orange")
SetLineStyle(lineHalfB, 2)
SetColor(M3, "Orange")
SetPointSize(M3, 4)
ShowLabel(M3, true)
SetLabelMode(M3, 1)
SetColor(M4, "Orange")
SetPointSize(M4, 4)
ShowLabel(M4, true)
SetLabelMode(M4, 1)
