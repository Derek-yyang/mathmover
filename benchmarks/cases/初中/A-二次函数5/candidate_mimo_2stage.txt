MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
A = (Root(f, -1, 0), 0)
B = (Root(f, 2, 4), 0)
C = (0, f(0))
Q = (0, 1)
t = Slider(-1, 3, 0.01)
M = (t, f(t))
Segment(M, B)
Segment(B, Q)
Angle(M, B, Q)
SetColor(f, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetFilling(Angle(M, B, Q), 0.3)
SetColor(Angle(M, B, Q), "Green")
