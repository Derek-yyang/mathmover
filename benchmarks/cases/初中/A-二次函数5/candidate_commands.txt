MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
SetColor(f, "Black")
A = (Root(f, -2, 0), 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (Root(f, 2, 4), 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, f(0))
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
Q = (0, 1)
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
lineQB = Line(Q, B)
SetColor(lineQB, "Black")
t = Slider(-2, 4, 0.05)
M = (t, f(t))
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
lineMB = Line(M, B)
SetColor(lineMB, "Black")
