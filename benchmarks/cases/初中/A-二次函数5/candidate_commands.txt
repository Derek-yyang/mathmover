SetAxesVisible(true, true)
SetGridVisible(true)
f(x) = -x^2 + 2x + 3
SetColor(f, "Black")
A = (Root(f, -2, 0), 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
B = (Root(f, 2, 4), 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetPointSize(B, 4)
C = (0, 3)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Black")
SetPointSize(C, 4)
Q = (0, 1)
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Black")
SetPointSize(Q, 4)
lineQB = Line(Q, B)
SetColor(lineQB, "Blue")
SetLineStyle(lineQB, 2)

M = (t, f(t))
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
segmentMB = Segment(M, B)
SetColor(segmentMB, "Red")
angMBQ = Angle(M, B, Q)
