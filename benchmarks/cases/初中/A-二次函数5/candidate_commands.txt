SetColor(f, "Black")

ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(Q, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(M, "Red")

SetLabelMode(M, 1)
SetColor(l1, "Blue")

SetColor(l2, "Green")

SetColor(angle1, "Red")
