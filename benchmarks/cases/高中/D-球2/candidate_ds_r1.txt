MODE: 3D
A = (0, 0, 0)
B = (2, 0, 0)
C = B + (0, 2*sqrt(3), 0)
P = A + (0, 0, 2)
sAB = Segment(A, B)
sAC = Segment(A, C)
sBC = Segment(B, C)
sPA = Segment(P, A)
sPB = Segment(P, B)

l = PerpendicularLine(M, planeABC)
pbAP = PlaneBisector(A, P)
O = Intersect(l, pbAP)
sphereO = Sphere(O, A)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(M, -1, false)
SetVisibleInView(l, -1, false)
SetVisibleInView(pbAP, -1, false)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
