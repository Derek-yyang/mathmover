MODE: 3D
B = (-3/2, 0, 0)
C = (3/2, 0, 0)
O = Midpoint(B, C)
P = (1/2, 0, 2)
circleO = Circle(O, B)
A = Point(circleO)
planeABC = Plane(A, B, C)
planePBC = Plane(P, B, C)
sAB = Segment(A, B)
sAC = Segment(A, C)
sBC = Segment(B, C)
sPA = Segment(P, A)
sPB = Segment(P, B)
sPC = Segment(P, C)
lineSO = PerpendicularLine(O, planeABC)
planePBisector = PlaneBisector(sPB)
S = Intersect(lineSO, planePBisector)
spherePABC = Sphere(S, B)
SetLineStyle(circleO, 2)
SetColor(circleO, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(planePBC, -1, false)
SetVisibleInView(lineSO, -1, false)
SetVisibleInView(planePBisector, -1, false)
SetColor(spherePABC, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetFixed(S, true)
ShowLabel(S, true)
SetLabelMode(S, 1)
SetColor(S, "Blue")
