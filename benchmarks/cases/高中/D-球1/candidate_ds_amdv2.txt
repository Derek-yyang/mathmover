MODE: 3D
O = (0, 0, 0)
B = (-1.5, 0, 0)
C = (1.5, 0, 0)
A = Point(Circle(O, B))
P = (0, 1, sqrt(7))
circABC = Circle(O, B)
triPBC = Polygon(P, B, C)
triABC = Polygon(A, B, C)
triPAB = Polygon(P, A, B)
triPCA = Polygon(P, C, A)
O1 = Midpoint(P, O)
sphere1 = Sphere(O1, O1)
radius = Distance(O1, P)
text1 = Text("外接球表面积 = " + (4 * pi * radius^2), (0, 0, 3))
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetColor(circABC, "Blue")
SetLineStyle(circABC, 3)
SetFilling(triPBC, 0.3)
SetColor(triPBC, "Orange")
SetFilling(triABC, 0.3)
SetColor(triABC, "Orange")
SetFilling(triPAB, 0.3)
SetColor(triPAB, "Orange")
SetFilling(triPCA, 0.3)
SetColor(triPCA, "Orange")
SetFixed(O1, true)
ShowLabel(O1, true)
SetLabelMode(O1, 1)
SetColor(O1, "Red")
SetFilling(sphere1, 0.1)
SetColor(sphere1, "Green")
ShowLabel(text1, true)
