MODE: 3D
B = (-3/2, 0, 0)
C = (3/2, 0, 0)
O = Midpoint(B, C)
P = (1/2, 0, 2)
sBC = Segment(B, C)
sPB = Segment(P, B)
sPC = Segment(P, C)
cO = Circle(O, B)
A = Point(cO)
sAB = Segment(A, B)
sAC = Segment(A, C)
sPA = Segment(P, A)
pl1 = PerpendicularBisector(sPB)
pl2 = PerpendicularBisector(sPC)
pl3 = PerpendicularBisector(sBC)
O_prime = Intersect(pl1, pl2, pl3)
S = Sphere(O_prime, B)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetLineStyle(cO, 2)
SetColor(cO, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetVisibleInView(pl1, -1, false)
SetVisibleInView(pl2, -1, false)
SetVisibleInView(pl3, -1, false)
SetFixed(O_prime, true)
ShowLabel(O_prime, true)
SetLabelMode(O_prime, 1)
SetColor(O_prime, "Blue")
SetFilling(S, 0.2)
