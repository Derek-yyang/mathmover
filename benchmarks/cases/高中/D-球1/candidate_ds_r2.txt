MODE: 3D
O = (0, 0, 0)
B = (-1.5, 0, 0)
C = (1.5, 0, 0)
A = (0, 1.5, 0)
P = (0, 0, 2)
sAB = Segment(A, B)
sAC = Segment(A, C)
sPB = Segment(P, B)
sPC = Segment(P, C)
sPA = Segment(P, A)
sBC = Segment(B, C)
circleO = Circle(O, A)
triABC = Polygon(A, B, C)
triPBC = Polygon(P, B, C)
triPAB = Polygon(P, A, B)
triPAC = Polygon(P, A, C)
planeABC = Plane(A, B, C)
O1 = Midpoint(B, C)
n1 = Vector(O1, P)
n2 = Vector(O1, A)
m1 = Cross(n1, n2)
O2 = Midpoint(P, B)
m2 = (0, 0, 0)
O3 = Midpoint(P, C)
p1 = PerpendicularPlane(O2, sPB)
p2 = PerpendicularPlane(O3, sPC)
p3 = PerpendicularPlane(O1, sBC)
O_center = Intersect(p1, p2, p3)
R_sphere = Distance(O_center, P)
sph = Sphere(O_center, R_sphere)
t = Text("外接球的表面积", O_center)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetColor(sBC, "Red")
SetLineStyle(sBC, 2)
SetColor(circleO, "Red")
SetLineStyle(circleO, 2)
SetFilling(triABC, 0.2)
SetFilling(triPBC, 0.2)
SetFilling(triPAB, 0.2)
SetFilling(triPAC, 0.2)
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(m1, -1, false)
SetVisibleInView(m2, -1, false)
SetVisibleInView(O3, -1, false)
SetVisibleInView(p1, -1, false)
SetVisibleInView(p2, -1, false)
SetVisibleInView(p3, -1, false)
ShowLabel(O_center, true)
SetLabelMode(O_center, 1)
SetColor(O_center, "Red")
SetColor(sph, "Green")
SetFilling(sph, 0.1)
SetVisibleInView(t, -1, false)
