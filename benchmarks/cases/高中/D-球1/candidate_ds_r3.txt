MODE: 3D
B = (0, 0, 0)
C = (3, 0, 0)
O = Midpoint(B, C)
pathA = Circle(O, B)
A = Point(pathA)
P = (1.5, sqrt(23)/2, 2.5)
sPB = Segment(P, B)
sPC = Segment(P, C)
sPA = Segment(P, A)
sAB = Segment(A, B)
sAC = Segment(A, C)
sBC = Segment(B, C)
planeABC = Plane(B, C, A)
planePBC = Plane(P, B, C)
polyABC = Polygon(A, B, C)
polyPBC = Polygon(P, B, C)
centerS = Midpoint(O, P)
sphereS = Sphere(centerS, Distance(centerS, P))
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetLineStyle(pathA, 2)
SetColor(pathA, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetConditionToShowObject(A, A != B && A != C)
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetLineStyle(sPB, 2)
SetColor(sPB, "Red")
SetLineStyle(sPC, 2)
SetColor(sPC, "Red")
SetLineStyle(sPA, 2)
SetColor(sPA, "Red")
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(planePBC, -1, false)
SetFilling(polyABC, 0.2)
SetVisibleInView(polyABC, -1, false)
SetFilling(polyPBC, 0.2)
SetVisibleInView(polyPBC, -1, false)
SetVisibleInView(centerS, -1, false)
SetFixed(centerS, true)
ShowLabel(centerS, true)
SetLabelMode(centerS, 1)
SetColor(centerS, "Blue")
ShowLabel(sphereS, true)
SetLabelMode(sphereS, 1)
SetColor(sphereS, "Blue")
