MODE: 3D
O = (1.5, 0, 0)
B = (0, 0, 0)
C = (3, 0, 0)
sBC = Segment(B, C)
t = Slider(0, 2*pi, 0.01)
A = Point(O + (1.5*cos(t), 1.5*sin(t), 0))
P = (0.5, 1, 2)
sPB = Segment(P, B)
sPC = Segment(P, C)
sPA = Segment(P, A)
sAB = Segment(A, B)
sAC = Segment(A, C)
planeABC = Plane(A, B, C)
planePBC = Plane(P, B, C)
D = Midpoint(B, C)
h1 = PerpendicularLine(D, planePBC)
M = (2.5, 0.5, 1)
h2 = PerpendicularLine(M, planeABC)
R_center = Intersect(h1, h2)
sphere = Sphere(R_center, A)
base_circle = Circle(O, 1.5)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetLineStyle(sBC, 2)
SetColor(sBC, "Red")
SetValue(t, 0.5)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(planePBC, -1, false)
SetVisibleInView(D, -1, false)
SetVisibleInView(h1, -1, false)
SetVisibleInView(M, -1, false)
SetVisibleInView(h2, -1, false)
SetVisibleInView(R_center, -1, false)
SetColor(sphere, "Blue")
SetFilling(sphere, 0.15)
SetColor(base_circle, "Red")
SetLineStyle(base_circle, 2)
