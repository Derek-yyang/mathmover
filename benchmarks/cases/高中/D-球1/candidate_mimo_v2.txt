MODE: 3D
B = (0, 0, 0)
C = (3, 0, 0)
O = Midpoint(B, C)
circleO = Circle(O, 1.5, yAxis)
t = Slider(0, 2*pi, 0.01)
A = Point(circleO, t)
planeABC = Plane(A, B, C)
planePBC = Plane(B, C, B + (0, 0, 1))
P = Intersect(planePBC, Sphere(B, 2*sqrt(2)), 1)
s1 = Segment(B, C)
s2 = Segment(B, A)
s3 = Segment(C, A)
s4 = Segment(P, B)
s5 = Segment(P, C)
s6 = Segment(P, A)
spherePABC = Sphere(A, B, C, P)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetVisibleInView(circleO, -1, false)
SetFixed(t, true)
ShowLabel(t, false)
SetFixed(A, false)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetVisibleInView(planeABC, -1, false)
SetVisibleInView(planePBC, -1, false)
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetVisibleInView(spherePABC, -1, true)
SetFilling(spherePABC, 0.1)
SetColor(spherePABC, "LightBlue")
ShowLabel(spherePABC, false)
