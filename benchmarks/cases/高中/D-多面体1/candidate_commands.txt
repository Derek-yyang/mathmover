MODE: 3D
A = (0, 0, 0)
B = (2, 0, 0)
C = (2, 2*sqrt(2), 0)
P = (xP, yP, zP)
O = Midpoint(B, C)
D = Midpoint(B, P)
E = Midpoint(A, P)
planeADO = Plane(A, D, O)
lineAO = Line(A, O)
sAB = Segment(A, B)
sBC = Segment(B, C)
sCA = Segment(C, A)
sPB = Segment(P, B)
sPC = Segment(P, C)
sPA = Segment(P, A)
t = Slider(0, 1, 0.01)
xP = 1
yP = sqrt(2)
zP = 2 + t
lineBF_perpAO = PerpendicularLine(B, lineAO)
F = Intersect(lineBF_perpAO, sCA)
sEF = Segment(E, F)
linePO = Line(P, O)
lineOF = Line(O, F)
anglePOF = Angle(P, O, F)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Blue")
SetVisibleInView(planeADO, -1, false)
SetVisibleInView(lineAO, -1, false)
SetFixed(P, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetDynamicColor(P, "Red")
SetFilling(planeADO, 0.2)
SetVisibleInView(lineBF_perpAO, -1, false)
SetFixed(F, false)

SetLabelMode(F, 1)
SetColor(F, "Red")
SetDynamicColor(F, "Red")
SetLineStyle(sEF, 2)
SetColor(sEF, "Red")

SetVisibleInView(lineOF, -1, false)
SetVisibleInView(anglePOF, -1, false)
