MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Parabola((1, 0), Line((-1, 0), (-1, 1)))
F = (1, 0)
m = Slider(0.2, 5, 0.01)
l = Line(F, F + (1, m))
A = Intersect(l, C, 1)
B = Intersect(l, C, 2)
Pab = Midpoint(A, B)
l2 = PerpendicularLine(Pab, l)
M = Intersect(l2, C, 1)
N = Intersect(l2, C, 2)
circ = Circle(A, M, B)
SetColor(C, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetPointSize(F, 4)
SetValue(m, 1)
SetColor(l, "Green")
SetLineThickness(l, 3)
ShowLabel(A, true)
SetLabelMode(A, 1)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetVisibleInView(Pab, 1, false)
SetColor(l2, "Blue")
SetLabelStyle(l2, 3)
SetCaption(l2, "l'")
ShowLabel(l2, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Red")
SetColor(circ, "Purple")
SetLineStyle(circ, 2)
ShowLabel(circ, false)
