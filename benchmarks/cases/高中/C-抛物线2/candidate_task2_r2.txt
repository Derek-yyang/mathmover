MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
directrix_line = Line((-1, 0), (-1, 1))
C_curve = Parabola(F, directrix_line)
k = Slider(0.3, 3, 0.01)
line_l = Line(F, F + (1, k))
A = Intersect(line_l, C_curve, 1)
B = Intersect(line_l, C_curve, 2)
P = Midpoint(A, B)
line_lprime = PerpendicularLine(P, line_l)
M = Intersect(line_lprime, C_curve, 1)
N = Intersect(line_lprime, C_curve, 2)
circAMBN = Circle(A, M, B)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetVisibleInView(directrix_line, 1, false)
SetColor(C_curve, "Black")
SetColor(line_l, "Blue")
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetVisibleInView(P, 1, false)
SetColor(line_lprime, "Green")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circAMBN, "Purple")
ShowLabel(circAMBN, false)
