MODE: 2D
SetAxesVisible(true, true)
F = (1, 0)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
SetVisibleInView(F, 1, true)
d: x = -1
SetVisibleInView(d, 1, true)
SetLineStyle(d, 2)
SetColor(d, "Gray")
C = Parabola(F, d)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
m = Slider(-5, 5, 0.1)
l = Line(F, F + (1, m))
ShowLabel(l, true)
SetVisibleInView(l, 1, true)
tempA = Intersect(l, C, 1)
tempB = Intersect(l, C, 2)
A = If(y(tempA) > y(tempB), tempA, tempB)
B = If(y(tempA) < y(tempB), tempA, tempB)
SetVisibleInView(tempA, 1, false)
SetVisibleInView(tempB, 1, false)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Blue")
l_prime = PerpendicularBisector(A, B)
ShowLabel(l_prime, true)
SetVisibleInView(l_prime, 1, true)
tempM = Intersect(l_prime, C, 1)
tempN = Intersect(l_prime, C, 2)
M = If(y(tempM) > y(tempN), tempM, tempN)
N = If(y(tempM) < y(tempN), tempM, tempN)
SetVisibleInView(tempM, 1, false)
SetVisibleInView(tempN, 1, false)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Red")
circleAMB = Circle(A, M, B)
SetLineStyle(circleAMB, 2)
SetColor(circleAMB, "DarkGreen")
SetVisibleInView(circleAMB, 1, true)
