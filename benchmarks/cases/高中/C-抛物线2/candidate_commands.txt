MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Parabola((1, 0), Line((-1, 0), (-1, 1)))
SetColor(C, "Black")
F = Focus(C)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 5)
directrix = Directrix(C)
SetColor(directrix, "Gray")
SetLineStyle(directrix, 2)
SetThickness(directrix, 1)
ShowLabel(directrix, false)
k = Slider(-2, 2, 0.01)
SetValue(k, 1)
line_l = Line(F, F + (1, k))
SetColor(line_l, "Blue")
ShowLabel(line_l, true)
SetLabelMode(line_l, 1)
A = Intersect(line_l, C, 1)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetPointSize(A, 5)
B = Intersect(line_l, C, 2)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Red")
SetPointSize(B, 5)
P = Midpoint(A, B)
ShowLabel(P, false)
SetVisibleInView(P, 1, false)
line_lprime = PerpendicularLine(line_l, P)
SetColor(line_lprime, "DarkGreen")
ShowLabel(line_lprime, true)
SetLabelMode(line_lprime, 1)
M = Intersect(line_lprime, C, 1)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "DarkBlue")
SetPointSize(M, 5)
N = Intersect(line_lprime, C, 2)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "DarkBlue")
SetPointSize(N, 5)
circleAMBN = Circle(A, M, B)
SetColor(circleAMBN, "Orange")
SetThickness(circleAMBN, 2)
ShowLabel(circleAMBN, false)
quadAMBN = Polygon(A, M, B, N)
SetColor(quadAMBN, "LightBlue")
SetFilling(quadAMBN, 0.25)
ShowLabel(quadAMBN, false)
