MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
d = Line((-1, 0), (-1, 1))
C = Parabola(F, d)
k = Slider(-5, 5, 0.1)
P1 = (2, k)
l = Line(F, P1)
A = Intersect(l, C, 1)
B = Intersect(l, C, 2)
P = Midpoint(A, B)
l_prime = PerpendicularLine(P, l)
M = Intersect(l_prime, C, 1)
N = Intersect(l_prime, C, 2)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(d, "Gray")
SetLineStyle(d, 2)
ShowLabel(d, false)
SetColor(C, "Black")
SetValue(k, sqrt(2))
SetFixed(k, true)
ShowLabel(k, true)
SetLabelMode(k, 1)
SetVisibleInView(P1, 1, false)
SetColor(l, "Blue")
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(P, "DarkGreen")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetPointSize(P, 4)
SetColor(l_prime, "Purple")
SetColor(M, "Orange")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Orange")
ShowLabel(N, true)
SetLabelMode(N, 1)
