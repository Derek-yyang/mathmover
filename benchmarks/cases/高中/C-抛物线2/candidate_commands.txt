MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Parabola((1, 0), x = -1)
F = (1, 0)
l_line = Line(F, (0, 0))
A = Intersect(C_curve, l_line, 1)
B = Intersect(C_curve, l_line, 2)
l_perp = PerpendicularLine(Midpoint(A, B), l_line)
M = Intersect(C_curve, l_perp, 1)
N = Intersect(C_curve, l_perp, 2)
circ = Circle(Midpoint(A, B), Midpoint(A, B) + (1, 1))
l_sol = Line(F, F + (1, -1))
SetColor(C_curve, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(l_line, "Green")
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l_perp, "Blue")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(circ, 1, false)
SetColor(l_sol, "Green")
SetLineStyle(l_sol, 2)
