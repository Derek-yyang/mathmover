MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
dir_line = Line((-1, 0), (-1, 1))
C_curve = Parabola(F, dir_line)
m_l = Slider(0.5, 2.5, 0.01)
line_l = Line(F, F + (1, m_l))
A = Intersect(line_l, C_curve, 1)
B = Intersect(line_l, C_curve, 2)
P_mid = Midpoint(A, B)
line_l_prime = PerpendicularLine(P_mid, line_l)
M = Intersect(line_l_prime, C_curve, 1)
N = Intersect(line_l_prime, C_curve, 2)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetVisibleInView(dir_line, 1, false)
SetColor(C_curve, "Black")
SetColor(line_l, "Blue")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Red")
SetVisibleInView(P_mid, 1, false)
SetColor(line_l_prime, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Red")
