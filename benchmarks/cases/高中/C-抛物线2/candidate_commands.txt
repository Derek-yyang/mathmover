MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Parabola((1, 0), x = -1)
F = (1, 0)
t_l = Slider(-1.5, 1.5, 0.01)
l = Line(F, F + (1, t_l))
A = Intersect(l, C_curve, 1)
B = Intersect(l, C_curve, 2)
mid_AB = Midpoint(A, B)
l_prime = PerpendicularLine(mid_AB, l)
M = Intersect(l_prime, C_curve, 1)
N = Intersect(l_prime, C_curve, 2)
cond_circle = Circle(A, B, M)
t_sol = Slider(-1.05, 1.05, 0.01)
P_on_circle = Point(cond_circle, t_sol)
t_k = Slider(0, 20, 0.1)
k_slope = t_k
l_solution = Line(F, F + (1, k_slope))
A_sol = Intersect(l_solution, C_curve, 1)
B_sol = Intersect(l_solution, C_curve, 2)
M_sol = Intersect(PerpendicularLine(Midpoint(A_sol, B_sol), l_solution), C_curve, 1)
N_sol = Intersect(PerpendicularLine(Midpoint(A_sol, B_sol), l_solution), C_curve, 2)
circle_sol = Circle(A_sol, B_sol, M_sol)
t_check = Slider(0, 1, 0.01)
P_check = Point(circle_sol, t_check)
l_show = Line(F, F + (1, sqrt(1)))
SetColor(C_curve, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(l, "Red")
SetColor(A, "Blue")
SetColor(B, "Blue")
ShowLabel(A, true)
SetLabelMode(A, 1)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l_prime, "Green")
SetColor(M, "Orange")
SetColor(N, "Orange")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(cond_circle, "Purple")
SetFilling(cond_circle, 0.1)
SetValue(t_sol, 1)
SetColor(P_on_circle, "Gray")
SetVisibleInView(P_on_circle, 1, false)
SetValue(t_k, 3)
SetColor(l_solution, "DarkRed")
SetVisibleInView(l_solution, 1, false)
SetVisibleInView(circle_sol, 1, false)
SetValue(t_check, 0.5)
SetColor(P_check, "DarkGray")
SetVisibleInView(P_check, 1, false)
SetColor(l_show, "Purple")
ShowLabel(l_show, true)
SetLabelMode(l_show, 1)
