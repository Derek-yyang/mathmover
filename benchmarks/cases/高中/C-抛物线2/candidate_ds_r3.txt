MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Parabola((1, 0), x = -1)
F = Focus(C_curve)
directrixC = Directrix(C_curve)
t_l = Slider(0.1, 5, 0.01)
l = Line(F, (1, t_l))
A = Intersect(C_curve, l, 1)
B = Intersect(C_curve, l, 2)
midAB = Midpoint(A, B)
l_prime = PerpendicularLine(midAB, l)
M = Intersect(C_curve, l_prime, 1)
N = Intersect(C_curve, l_prime, 2)
circleABCD = Circle(midAB, midAB)
P1 = Intersect(circleABCD, l, 1)
P2 = Intersect(circleABCD, l, 2)
P3 = Intersect(circleABCD, l_prime, 1)
P4 = Intersect(circleABCD, l_prime, 2)
condition1 = If(AreCollinear(A, P1, M, N), true, false)
condition2 = If(AreCollinear(A, P2, M, N), true, false)
condition3 = If(AreCollinear(B, P1, M, N), true, false)
condition4 = If(AreCollinear(B, P2, M, N), true, false)
res1 = Intersect(condition1, condition2)
res2 = Intersect(condition3, condition4)
solX = x(res1)
l_solution = Line(F, (1, solX))
SetColor(C_curve, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(directrixC, "Gray")
SetLineStyle(directrixC, 3)
SetColor(l, "Green")
SetColor(A, "Black")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Black")
ShowLabel(B, true)
SetLabelMode(B, 1)
ShowLabel(midAB, false)
SetColor(l_prime, "Green")
SetColor(M, "Black")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Black")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(P1, 1, false)
SetVisibleInView(P2, 1, false)
SetVisibleInView(P3, 1, false)
SetVisibleInView(P4, 1, false)
SetVisibleInView(res1, 1, false)
SetVisibleInView(res2, 1, false)
SetVisibleInView(solX, 1, false)
SetColor(l_solution, "Red")
SetLineThickness(l_solution, 3)
