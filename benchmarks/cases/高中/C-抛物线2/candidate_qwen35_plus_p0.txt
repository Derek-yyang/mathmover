MODE: 2D
C: y^2 = 4x
F = (1, 0)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
SetVisibleInView(F, 1, true)
k = Slider(-2, 2, 0.1)
l = Line(F, F + (1, k))
SetColor(l, "Green")
ShowLabel(l, true)
SetLabelMode(l, 1)
SetVisibleInView(l, 1, true)
tempA = Intersect(l, C, 1)
tempB = Intersect(l, C, 2)
A = If(y(tempA) > y(tempB), tempA, tempB)
B = If(y(tempA) < y(tempB), tempA, tempB)
SetVisibleInView(tempA, 1, false)
SetVisibleInView(tempB, 1, false)
SetColor(A, "Red")
SetColor(B, "Blue")
SetPointSize(A, 4)
SetPointSize(B, 4)
ShowLabel(A, true)
ShowLabel(B, true)
SetLabelMode(A, 1)
SetLabelMode(B, 1)
SetVisibleInView(A, 1, true)
SetVisibleInView(B, 1, true)
l_prime = PerpendicularBisector(A, B)
SetColor(l_prime, "Black")
SetLineStyle(l_prime, 1)
ShowLabel(l_prime, true)
SetLabelMode(l_prime, 1)
SetVisibleInView(l_prime, 1, true)
tempM = Intersect(l_prime, C, 1)
tempN = Intersect(l_prime, C, 2)
M = If(y(tempM) > y(tempN), tempM, tempN)
N = If(y(tempM) < y(tempN), tempM, tempN)
SetVisibleInView(tempM, 1, false)
SetVisibleInView(tempN, 1, false)
SetColor(M, "Red")
SetColor(N, "Blue")
SetPointSize(M, 4)
SetPointSize(N, 4)
ShowLabel(M, true)
ShowLabel(N, true)
SetLabelMode(M, 1)
SetLabelMode(N, 1)
SetVisibleInView(M, 1, true)
SetVisibleInView(N, 1, true)
SetAxesVisible(true, true)
SetColor(C, "Black")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetVisibleInView(C, 1, true)
