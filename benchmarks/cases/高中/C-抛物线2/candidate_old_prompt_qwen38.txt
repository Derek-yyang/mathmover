MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
SetVisibleInView(F, 1, true)
d = Line((-1, 0), (-1, 1))
SetVisibleInView(d, 1, false)
C = Parabola(F, d)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
m = Slider(0.2, 5, 0.1)
SetValue(m, 1)
l = Line(F, F + (1, m))
ShowLabel(l, true)
SetColor(l, "Black")
SetVisibleInView(l, 1, true)
tempA = Intersect(l, C, 1)
tempB = Intersect(l, C, 2)
A = If(y(tempA) > y(tempB), tempA, tempB)
B = If(y(tempA) < y(tempB), tempA, tempB)
SetVisibleInView(tempA, 1, false)
SetVisibleInView(tempB, 1, false)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetPointSize(A, 6)
SetVisibleInView(A, 1, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Red")
SetPointSize(B, 6)
SetVisibleInView(B, 1, true)
lprime = PerpendicularBisector(A, B)
ShowLabel(lprime, true)
SetColor(lprime, "Gray")
SetLineStyle(lprime, 2)
SetVisibleInView(lprime, 1, true)
tempM = Intersect(lprime, C, 1)
tempN = Intersect(lprime, C, 2)
M = If(y(tempM) > y(tempN), tempM, tempN)
N = If(y(tempM) < y(tempN), tempM, tempN)
SetVisibleInView(tempM, 1, false)
SetVisibleInView(tempN, 1, false)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
SetPointSize(M, 6)
SetVisibleInView(M, 1, true)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Blue")
SetPointSize(N, 6)
SetVisibleInView(N, 1, true)
circ = Circle(A, M, B)
ShowLabel(circ, true)
SetColor(circ, "DarkBlue")
SetLineStyle(circ, 2)
SetVisibleInView(circ, 1, true)
