MODE: 2D
SetAxesVisible(true, true)
F = (1, 0)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
d: x = -1
SetVisibleInView(d, 1, true)
SetLineStyle(d, 2)
SetColor(d, "Gray")
C = Parabola(F, d)
SetColor(C, "Black")
k = Slider(-5, 5, 0.1)
l = Line(F, F + (1, k))
ShowLabel(l, true)
SetVisibleInView(l, 1, true)
tempA = Intersect(l, C, 1)
tempB = Intersect(l, C, 2)
A = If(y(tempA) > y(tempB), tempA, tempB)
B = If(y(tempA) < y(tempB), tempA, tempB)
SetVisibleInView(tempA, 1, false)
SetVisibleInView(tempB, 1, false)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Red")
l_prime = PerpendicularBisector(A, B)
ShowLabel(l_prime, true)
SetVisibleInView(l_prime, 1, true)
SetColor(l_prime, "DarkGreen")
tempM = Intersect(l_prime, C, 1)
tempN = Intersect(l_prime, C, 2)
M = If(y(tempM) > y(tempN), tempM, tempN)
N = If(y(tempM) < y(tempN), tempM, tempN)
SetVisibleInView(tempM, 1, false)
SetVisibleInView(tempN, 1, false)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Blue")
circleABMN = Circle(A, M, B)
ShowLabel(circleABMN, true)
SetVisibleInView(circleABMN, 1, true)
SetColor(circleABMN, "Purple")
SetLineStyle(circleABMN, 2)
