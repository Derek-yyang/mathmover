MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Parabola((1, 0), Line((-1, 0), (-1, 1)))
F = Focus(C_curve)
dir_C = Directrix(C_curve)
t_l = Slider(-3, 3, 0.01)
l = Line(F, F + (1, t_l))
A = Intersect(l, C_curve, 1)
B = Intersect(l, C_curve, 2)
mid_AB = Midpoint(A, B)
l_p = PerpendicularLine(mid_AB, l)
M = Intersect(l_p, C_curve, 1)
N = Intersect(l_p, C_curve, 2)
circle_MABN = Circle(mid_AB, mid_AB, M)
SetColor(C_curve, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetPointSize(F, 4)
SetVisibleInView(dir_C, 1, false)
SetColor(l, "Green")
SetLineThickness(l, 4)
SetColor(A, "Black")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Black")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l_p, "Black")
SetColor(M, "Black")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Black")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circle_MABN, "DarkGreen")
SetLineThickness(circle_MABN, 4)
SetVisibleInView(circle_MABN, 1, true)
