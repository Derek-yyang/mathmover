MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Parabola((1, 0), Line((-1, 0), (-1, 1)))
F = Focus(C)
directrixC = Directrix(C)
line_l = Line(F, F + (1, 1))
A = Intersect(C, line_l, 1)
B = Intersect(C, line_l, 2)
P = Midpoint(A, B)
line_l_prime = PerpendicularLine(P, line_l)
M = Intersect(C, line_l_prime, 1)
N = Intersect(C, line_l_prime, 2)
circleAMBN = Circle(A, M, B)
SetColor(C, "Black")
SetLineThickness(C, 2)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(F, "Black")
SetPointSize(F, 4)
SetVisibleInView(directrixC, 1, false)
SetColor(line_l, "Blue")
SetLineThickness(line_l, 2)
ShowLabel(line_l, true)
SetLabelMode(line_l, 1)
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(line_l_prime, "Green")
SetLineThickness(line_l_prime, 2)
ShowLabel(line_l_prime, true)
SetLabelMode(line_l_prime, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circleAMBN, "Gray")
SetLineStyle(circleAMBN, 2)
SetFilling(circleAMBN, 0.1)
ShowLabel(circleAMBN, false)
