MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
dirC = Line((-1, 0), (-1, 1))
C = Parabola((1, 0), dirC)
F = (1, 0)
k = Slider(-4, 4, 0.05)
l = Line(F, F + (1, k))
A = Intersect(l, C, 1)
B = Intersect(l, C, 2)
l2 = PerpendicularLine(Midpoint(A, B), l)
M = Intersect(l2, C, 1)
N = Intersect(l2, C, 2)
circ1 = Circle(Midpoint(A, B), Distance(A, B) / 2)
O1 = Intersect(PerpendicularBisector(A, M), PerpendicularBisector(A, B))
circ2 = Circle(O1, Distance(A, M) / 2)
SetVisibleInView(dirC, 1, false)
SetColor(C, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetPointSize(F, 4)
SetValue(k, 1)
SetColor(l, "Green")
SetLinethickness(l, 3)
ShowLabel(A, true)
SetLabelMode(A, 1)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l2, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(circ1, 1, false)
ShowLabel(O1, true)
SetLabelMode(O1, 1)
SetColor(O1, "DarkGreen")
SetColor(circ2, "Gray")
SetLinestyle(circ2, 2)
