MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
directrix_line = Line((-1, 0), (-1, 1))
C = Parabola(F, directrix_line)
m = Slider(-2, 2, 0.01)
line_l = Line(F, F + (1, m))
A = Intersect(line_l, C, 1)
B = Intersect(line_l, C, 2)
P = Midpoint(A, B)
line_lprime = PerpendicularLine(P, line_l)
M = Intersect(line_lprime, C, 1)
N = Intersect(line_lprime, C, 2)
circleAMBN = Circle(A, M, B)
quadAMBN = Polygon(A, M, B, N)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetVisibleInView(directrix_line, 1, false)
SetColor(C, "Black")
SetValue(m, 1)
SetColor(line_l, "Blue")
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetVisibleInView(P, 1, false)
SetColor(line_lprime, "Blue")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circleAMBN, "Gray")
SetLineStyle(circleAMBN, 2)
ShowLabel(circleAMBN, false)
SetColor(quadAMBN, "LightBlue")
SetFilling(quadAMBN, 0.2)
ShowLabel(quadAMBN, false)
