MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
directrix = Line((-1, 0), (-1, 1))
SetColor(directrix, "Gray")
SetLineStyle(directrix, 2)
ShowLabel(directrix, true)
SetLabelMode(directrix, 1)
C_curve = Parabola(F, directrix)
SetColor(C_curve, "Black")
m = Slider(0.2, 3, 0.01)
SetVisibleInView(m, 1, true)
ShowLabel(m, true)
line_l = Line(F, (2, m))
SetColor(line_l, "Blue")
ShowLabel(line_l, true)
SetLabelMode(line_l, 1)
A = Intersect(line_l, C_curve, 1)
SetColor(A, "Red")
ShowLabel(A, true)
SetLabelMode(A, 1)
B = Intersect(line_l, C_curve, 2)
SetColor(B, "Red")
ShowLabel(B, true)
SetLabelMode(B, 1)
P = Midpoint(A, B)
SetPointSize(P, 3)
SetColor(P, "DarkGreen")
SetVisibleInView(P, 1, false)
l_prime = PerpendicularLine(line_l, P)
SetColor(l_prime, "Blue")
ShowLabel(l_prime, true)
SetLabelMode(l_prime, 1)
M = Intersect(l_prime, C_curve, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
N = Intersect(l_prime, C_curve, 2)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
circAMBN = Circle(A, M, B)
SetColor(circAMBN, "Orange")
SetLineStyle(circAMBN, 2)
ShowLabel(circAMBN, false)
