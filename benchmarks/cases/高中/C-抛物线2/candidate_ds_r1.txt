MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve: y^2 = 4x
F = Focus(C_curve)
f_k(x) = x^3 + 2x^2 + 2x - 1
k = sqrt(Root(f_k, 0))
l = Line(F, F + (1, k))
A = Intersect(l, C_curve, 1)
B = Intersect(l, C_curve, 2)
midAB = Midpoint(A, B)
l_prime = PerpendicularLine(midAB, l)
M = Intersect(l_prime, C_curve, 1)
N = Intersect(l_prime, C_curve, 2)
circleAMBN = Circle(A, M, B)
SetColor(C_curve, "Black")
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetPointSize(F, 4)
SetVisibleInView(f_k, 1, false)
SetColor(l, "Green")
SetLineThickness(l, 3)
ShowLabel(A, true)
SetLabelMode(A, 1)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetVisibleInView(midAB, 1, false)
SetColor(l_prime, "Green")
SetLineThickness(l_prime, 3)
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circleAMBN, "Purple")
SetFilling(circleAMBN, 0.2)
ShowLabel(circleAMBN, false)
