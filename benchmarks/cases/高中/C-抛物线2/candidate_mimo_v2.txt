MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F = (1, 0)
C = Parabola(F, x = -1)
t = Slider(0.1, 0.9, 0.01)
M = Point(C, t)
lineMF = Line(M, F)
A = Intersect(lineMF, C, 1)
B = Intersect(lineMF, C, 2)
mid = Midpoint(A, B)
lineAB = Line(A, B)
lineL = PerpendicularLine(mid, lineAB)
M2 = Intersect(lineL, C, 1)
N = Intersect(lineL, C, 2)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(C, "Black")
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(lineMF, "Blue")
SetColor(A, "Blue")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "Blue")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(mid, "DarkGreen")
ShowLabel(mid, true)
SetLabelMode(mid, 1)
SetColor(lineAB, "Blue")
SetColor(lineL, "Orange")
ShowLabel(lineL, true)
SetLabelMode(lineL, 1)
SetColor(M2, "DarkGreen")
ShowLabel(M2, true)
SetLabelMode(M2, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
