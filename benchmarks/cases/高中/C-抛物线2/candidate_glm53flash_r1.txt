MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
dir_aux = Line((-1, 0), (-1, 1))
C = Parabola((1, 0), dir_aux)
F = Focus(C)
dirC = Directrix(C)
t_k = Slider(-4, 4, 0.1)
F_dir = F + (1, t_k)
l = Line(F, F_dir)
A = Intersect(l, C, 1)
B = Intersect(l, C, 2)
l' = PerpendicularBisector(A, B)
M = Intersect(l', C, 1)
N = Intersect(l', C, 2)
circ1 = Circle(A, M, B)
SetVisibleInView(dir_aux, 1, false)
SetColor(C, "Black")
SetPointSize(F, 4)
SetFixed(F, true)
ShowLabel(F, true)
SetLabelMode(F, 1)
SetColor(dirC, "Gray")
SetLineStyle(dirC, 1)
ShowLabel(dirC, false)
SetValue(t_k, 1)
SetVisibleInView(F_dir, 1, false)
SetColor(l, "Blue")
ShowLabel(l, true)
SetLabelMode(l, 1)
SetColor(A, "DarkGreen")
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(B, "DarkGreen")
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(l', "Green")
ShowLabel(l', true)
SetLabelMode(l', 1)
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circ1, "Orange")
ShowLabel(circ1, false)
