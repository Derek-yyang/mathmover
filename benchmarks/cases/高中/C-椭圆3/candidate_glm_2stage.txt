MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(8), 0), (sqrt(8), 0), 3)
O = (0, 0)
A = (-3, 0)
B = (3, 0)
G = (0, 1)
vecAG = Vector(A, G)
vecGB = Vector(G, B)
lP = Line((6, 0), (6, 1))
t = Slider(-5, 5, 0.1)
P = (6, t)
linePA = Line(A, P)
linePB = Line(B, P)
C = Intersect(linePA, E, 2)
D = Intersect(linePB, E, 2)
lineCD = Line(C, D)
xAxis_line = Line((0, 0), (1, 0))
Q = Intersect(lineCD, xAxis_line)
ZoomIn(-8, -5, 10, 5)
SetColor(E, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(vecAG, "Gray")
SetColor(vecGB, "Gray")
SetColor(lP, "Gray")
SetLineStyle(lP, 2)
SetValue(t, 3)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePA, "Blue")
SetColor(linePB, "Blue")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "DarkGreen")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "DarkGreen")
SetColor(lineCD, "Magenta")
SetThickness(lineCD, 3)
SetVisibleInView(xAxis_line, 1, false)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetCaption(Q, "定点")
SetLabelMode(Q, 3)
SetColor(Q, "Magenta")
SetPointSize(Q, 5)
