MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-2*sqrt(2), 0), (2*sqrt(2), 0), 3)
O = (0, 0)
A = (-3, 0)
B = (3, 0)
G = (0, 1)
lineX6 = Line((6, 0), (6, 1))
t_p = Slider(0.1, 5, 0.1)
P = (6, t_p)
linePA = Line(P, A)
linePB = Line(P, B)
CP1 = Intersect(linePA, E, 1)
CP2 = Intersect(linePA, E, 2)
C = if(Distance(CP1, A) < 0.001, CP2, CP1)
DP1 = Intersect(linePB, E, 1)
DP2 = Intersect(linePB, E, 2)
D = if(Distance(DP1, B) < 0.001, DP2, DP1)
lineCD = Line(C, D)
xAxis_line = Line((0, 0), (1, 0))
Q = Intersect(lineCD, xAxis_line)
SetColor(E, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(G, "Black")
SetColor(lineX6, "Blue")
SetValue(t_p, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePA, "Blue")
SetColor(linePB, "Blue")
SetVisibleInView(CP1, 1, false)
SetVisibleInView(CP2, 1, false)
SetColor(C, "DarkGreen")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetVisibleInView(DP1, 1, false)
SetVisibleInView(DP2, 1, false)
SetColor(D, "DarkGreen")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(lineCD, "Black")
SetVisibleInView(xAxis_line, 1, false)
SetColor(Q, "DarkGreen")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
