MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-2*sqrt(2), 0), (2*sqrt(2), 0), 3)
SetColor(E, "Black")
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
A = (-3, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
G = (0, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
lineX6 = Line((6, 0), (6, 1))
SetColor(lineX6, "Gray")
SetLineStyle(lineX6, 2)
ShowLabel(lineX6, false)
t_p = Slider(-4, 4, 0.01)
P = (6, t_p)
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
linePA = Line(P, A)
SetColor(linePA, "Blue")
ShowLabel(linePA, false)
linePB = Line(P, B)
SetColor(linePB, "Blue")
ShowLabel(linePB, false)
C_1 = Intersect(linePA, E, 1)
C_2 = Intersect(linePA, E, 2)
SetVisibleInView(C_1, 1, false)
SetVisibleInView(C_2, 1, false)
C = If(C_1 == A, C_2, C_1)
SetColor(C, "DarkGreen")
SetPointSize(C, 5)
ShowLabel(C, true)
SetLabelMode(C, 1)
D_1 = Intersect(linePB, E, 1)
D_2 = Intersect(linePB, E, 2)
SetVisibleInView(D_1, 1, false)
SetVisibleInView(D_2, 1, false)
D = If(D_1 == B, D_2, D_1)
SetColor(D, "DarkGreen")
SetPointSize(D, 5)
ShowLabel(D, true)
SetLabelMode(D, 1)
lineCD = Line(C, D)
SetColor(lineCD, "Purple")
SetLineThickness(lineCD, 3)
ShowLabel(lineCD, false)
S = (3/2, 0)
SetFixed(S, true)
ShowLabel(S, true)
SetLabelMode(S, 1)
SetColor(S, "DarkBlue")
SetPointSize(S, 5)
xAxis_line = Line((0, 0), (1, 0))
SetVisibleInView(xAxis_line, 1, false)
Q = Intersect(lineCD, xAxis_line)
SetVisibleInView(Q, 1, false)
