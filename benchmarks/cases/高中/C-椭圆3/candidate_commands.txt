MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-2*sqrt(2), 0), (2*sqrt(2), 0), 3)
A = (-3, 0)
B = (3, 0)
G = (0, 1)
lineL = Line((6, 0), (6, 1))
t_p = Slider(-5, 5, 0.1)
P = Point(lineL, t_p)
linePA = Line(A, P)
linePB = Line(B, P)
C = If(Intersect(E, linePA, 1) == A, Intersect(E, linePA, 2), Intersect(E, linePA, 1))
D = If(Intersect(E, linePB, 1) == B, Intersect(E, linePB, 2), Intersect(E, linePB, 1))
lineCD = Line(C, D)
H = (3/2, 0)
SetColor(E, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(lineL, "Gray")
SetLineStyle(lineL, 1)
SetLineThickness(lineL, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePA, "Blue")
SetColor(linePB, "Blue")
SetColor(C, "DarkGreen")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(D, "DarkGreen")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(lineCD, "Magenta")
SetLineThickness(lineCD, 2)
SetFixed(H, true)
ShowLabel(H, true)
SetLabelMode(H, 1)
SetColor(H, "Black")
