MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(8), 0), (sqrt(8), 0), 3)
O = (0, 0)
A = (-3, 0)
B = (3, 0)
G = (0, 1)
lineX6 = Line((6, 0), (6, 1))
t_p = Slider(-6, 6, 0.1)
P = (6, t_p)
linePA = Line(P, A)
linePB = Line(P, B)
iPA1 = Intersect(linePA, E, 1)
iPA2 = Intersect(linePA, E, 2)
C = If(Distance(iPA1, A) < 0.001, iPA2, iPA1)
iPB1 = Intersect(linePB, E, 1)
iPB2 = Intersect(linePB, E, 2)
D = If(Distance(iPB1, B) < 0.001, iPB2, iPB1)
lineCD = Line(C, D)
xAxis_line = Line((0, 0), (1, 0))
Q = Intersect(lineCD, xAxis_line)
SetColor(E, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
SetColor(lineX6, "Gray")
SetLineStyle(lineX6, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePA, "Blue")
SetColor(linePB, "Blue")
ShowLabel(iPA1, false)
ShowLabel(iPA2, false)
SetColor(C, "DarkGreen")
ShowLabel(C, true)
SetLabelMode(C, 1)
ShowLabel(iPB1, false)
ShowLabel(iPB2, false)
SetColor(D, "DarkGreen")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(lineCD, "Green")
SetLineThickness(lineCD, 3)
SetVisibleInView(xAxis_line, 1, false)
SetColor(Q, "DarkGreen")
SetPointSize(Q, 5)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
