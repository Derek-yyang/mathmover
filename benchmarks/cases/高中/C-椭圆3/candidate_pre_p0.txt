MODE: 2D
SetAxesVisible(true, true)
// 1. 焦点计算与椭圆构建 (a=3, b=1, c=sqrt(8)≈2.83)
F1 = (-2.83, 0)
F2 = (2.83, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetLabelMode(F1, 1)
SetVisibleInView(F1, 1, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetLabelMode(F2, 1)
SetVisibleInView(F2, 1, true)
E = Ellipse(F1, F2, 3)
SetColor(E, "Black")
SetThickness(E, 2)
// 2. 静态定点定义
A = (-3, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
G = (0, 1)
SetFixed(G, true)
ShowLabel(G, true)
SetLabelMode(G, 1)
// 3. 构建直线 x = 6 与动点 P
L_6 = Line((6, -10), (6, 10))
SetColor(L_6, "Gray")
SetLineStyle(L_6, 2)
ShowLabel(L_6, true)
t = Slider(-8, 8, 0.1)
P = (6, t)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
// 4. 构建直线 PA, PB 及其与椭圆的交点 C, D
linePA = Line(P, A)
linePB = Line(P, B)
SetVisibleInView(linePA, 1, true)
SetVisibleInView(linePB, 1, true)
// 处理交点 C (PA 与 E 的交点，排除 A)
tempC1 = Intersect(linePA, E, 1)
tempC2 = Intersect(linePA, E, 2)
C = If(Distance(tempC1, A) > 0.01, tempC1, tempC2)
SetVisibleInView(tempC1, 1, false)
SetVisibleInView(tempC2, 1, false)
SetColor(C, "Blue")
ShowLabel(C, true)
SetLabelMode(C, 1)
// 处理交点 D (PB 与 E 的交点，排除 B)
tempD1 = Intersect(linePB, E, 1)
tempD2 = Intersect(linePB, E, 2)
D = If(Distance(tempD1, B) > 0.01, tempD1, tempD2)
SetVisibleInView(tempD1, 1, false)
SetVisibleInView(tempD2, 1, false)
SetColor(D, "Blue")
ShowLabel(D, true)
SetLabelMode(D, 1)
// 5. 构建直线 CD
lineCD = Line(C, D)
SetColor(lineCD, "DarkGreen")
SetThickness(lineCD, 2)
ShowLabel(lineCD, true)
// 6. 辅助：标记原点
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetPointSize(O, 3)
