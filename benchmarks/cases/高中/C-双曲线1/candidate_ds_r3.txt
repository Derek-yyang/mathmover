MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_hyperbola = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), 1)
O = (0, 0)
A = (-1, 0)
V_right = (1, 0)
lineAsym1 = Line((0, 0), (1, sqrt(2)))
lineAsym2 = Line((0, 0), (1, -sqrt(2)))
t_k = Slider(-3, 3, 0.01)
t_m = Slider(-4, 4, 0.01)
line_l = Line((0, t_m), (1, t_k + t_m))
l_P = Intersect(line_l, C_hyperbola, 1)
l_Q = Intersect(line_l, C_hyperbola, 2)
l_M = Intersect(line_l, lineAsym1)
l_N = Intersect(line_l, lineAsym2)
mid_PQ = Midpoint(l_P, l_Q)
circle_PQ = Circle(mid_PQ, Distance(mid_PQ, l_P))
chk1 = Distance(A, mid_PQ)
chk2 = Distance(mid_PQ, l_P)
locus_l = Locus(l_P, t_k)
SetColor(C_hyperbola, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(V_right, 1, false)
SetColor(lineAsym1, "Gray")
SetLineStyle(lineAsym1, 1)
SetColor(lineAsym2, "Gray")
SetLineStyle(lineAsym2, 1)
SetColor(l_P, "Red")
SetColor(l_Q, "Red")
SetColor(l_M, "Blue")
SetColor(l_N, "Blue")
ShowLabel(l_P, true)
SetLabelMode(l_P, 1)
ShowLabel(l_Q, true)
SetLabelMode(l_Q, 1)
ShowLabel(l_M, true)
SetLabelMode(l_M, 1)
ShowLabel(l_N, true)
SetLabelMode(l_N, 1)
SetColor(circle_PQ, "LightBlue")
SetFilling(circle_PQ, 0.3)
ShowLabel(circle_PQ, false)
SetVisibleInView(chk1, 1, false)
SetVisibleInView(chk2, 1, false)
SetVisibleInView(locus_l, 1, false)
