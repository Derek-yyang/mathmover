MODE: 2D
SetAxesVisible(true, true)
F1 = (-2, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetVisibleInView(F1, 1, true)
F2 = (2, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetVisibleInView(F2, 1, true)
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetVisibleInView(A, 1, true)
C = Hyperbola(F1, F2, 1)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
asym1 = Line((0, 0), (1, sqrt(3)))
SetLineStyle(asym1, 3)
SetColor(asym1, "DarkGray")
SetVisibleInView(asym1, 1, true)
asym2 = Line((0, 0), (1, -sqrt(3)))
SetLineStyle(asym2, 3)
SetColor(asym2, "DarkGray")
SetVisibleInView(asym2, 1, true)
k = Slider(-5, 5, 0.1)
l = Line(F2, F2 + (1, k))
SetColor(l, "Blue")
SetVisibleInView(l, 1, true)
tempP = Intersect(l, C, 1)
tempQ = Intersect(l, C, 2)
P = If(x(tempP) < x(tempQ), tempP, tempQ)
Q = If(x(tempP) > x(tempQ), tempP, tempQ)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetPointSize(P, 4)
SetVisibleInView(P, 1, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
SetPointSize(Q, 4)
SetVisibleInView(Q, 1, true)
SetVisibleInView(tempP, 1, false)
SetVisibleInView(tempQ, 1, false)
O_pq = Midpoint(P, Q)
ShowLabel(O_pq, true)
SetLabelMode(O_pq, 1)
SetColor(O_pq, "Gray")
SetPointSize(O_pq, 4)
SetVisibleInView(O_pq, 1, true)
circlePQ = Circle(O_pq, P)
SetColor(circlePQ, "Gray")
SetLineStyle(circlePQ, 2)
SetVisibleInView(circlePQ, 1, true)

SetVisibleInView(M, 1, true)
N = Intersect(l, asym2, 1)

SetVisibleInView(N, 1, true)
