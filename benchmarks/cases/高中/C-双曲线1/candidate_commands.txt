MODE: 2D
SetAxesVisible(true, true)
F1 = (-2, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetVisibleInView(F1, 1, true)
F2 = (2, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetVisibleInView(F2, 1, true)
C = Hyperbola(F1, F2, 1)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
SetVisibleInView(A, 1, true)
asy1: y = sqrt(3) x
SetLineStyle(asy1, 3)
SetColor(asy1, "Gray")
SetVisibleInView(asy1, 1, true)
asy2: y = -sqrt(3) x
SetLineStyle(asy2, 3)
SetColor(asy2, "Gray")
SetVisibleInView(asy2, 1, true)
k = Slider(-1.5, 1.5, 0.01)
D = (2, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Black")
SetPointSize(D, 4)
SetVisibleInView(D, 1, true)
l = Line(D, D + (1, k))
SetColor(l, "Red")
SetVisibleInView(l, 1, true)
tempP = Intersect(l, C, 1)
tempQ = Intersect(l, C, 2)
P = If(y(tempP) > y(tempQ), tempP, tempQ)
Q = If(y(tempP) < y(tempQ), tempP, tempQ)
SetVisibleInView(tempP, 1, false)
SetVisibleInView(tempQ, 1, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetVisibleInView(P, 1, true)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Blue")
SetVisibleInView(Q, 1, true)
M = Intersect(l, asy1)
N = Intersect(l, asy2)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Green")
SetVisibleInView(M, 1, true)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Green")
SetVisibleInView(N, 1, true)
midPQ = Midpoint(P, Q)
circlePQ = Circle(midPQ, P)
SetColor(circlePQ, "Gray")
SetLineStyle(circlePQ, 2)
SetFilling(circlePQ, 0)
SetVisibleInView(circlePQ, 1, true)
segMP = Segment(M, P)
SetColor(segMP, "Orange")
SetVisibleInView(segMP, 1, true)
segPQ = Segment(P, Q)
SetColor(segPQ, "Orange")
SetVisibleInView(segPQ, 1, true)
segQN = Segment(Q, N)
SetColor(segQN, "Orange")
SetVisibleInView(segQN, 1, true)
