MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
SetColor(C_curve, "Black")
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
D = (2, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
V_right = (1, 0)
SetVisibleInView(V_right, 1, false)
asym1 = Line(O, (1, sqrt(3)))
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
ShowLabel(asym1, false)
asym2 = Line(O, (1, -sqrt(3)))
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
ShowLabel(asym2, false)
k_slider = Slider(-1.5, 1.5, 0.01)
SetCaption(k_slider, "k")
SetValue(k_slider, 0.5)
l_line = Line(D, (3, k_slider))
SetColor(l_line, "Blue")
P = Intersect(l_line, C_curve, 1)
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
Q = Intersect(l_line, C_curve, 2)
SetColor(Q, "Red")
SetPointSize(Q, 5)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
M = Intersect(l_line, asym1)
SetColor(M, "Red")
SetPointSize(M, 5)
ShowLabel(M, true)
SetLabelMode(M, 1)
N = Intersect(l_line, asym2)
SetColor(N, "Red")
SetPointSize(N, 5)
ShowLabel(N, true)
SetLabelMode(N, 1)
midPQ = Midpoint(P, Q)
SetVisibleInView(midPQ, 1, false)
circlePQ = Circle(midPQ, P)
SetColor(circlePQ, "LightBlue")
SetLineStyle(circlePQ, 2)
ShowLabel(circlePQ, false)
segAP = Segment(A, P)
SetColor(segAP, "DarkGreen")
ShowLabel(segAP, false)
segAQ = Segment(A, Q)
SetColor(segAQ, "DarkGreen")
ShowLabel(segAQ, false)
