MODE: 2D
SetAxesVisible(true, true)
a = 1
b = 1.732
c = 2
F1 = (-2, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F1, true)
F2 = (2, 0)
SetColor(F2, "Black")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetFixed(F2, true)
C = Hyperbola(F1, F2, a)
SetColor(C, "Black")
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
D = (2, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
k = Slider(-5, 5, 0.1)
l = Line(D, D + (1, k))
ShowLabel(l, true)
SetColor(l, "Blue")
tempP = Intersect(l, C, 1)
tempQ = Intersect(l, C, 2)
P = If(y(tempP) > y(tempQ), tempP, tempQ)
Q = If(y(tempP) < y(tempQ), tempP, tempQ)
SetVisibleInView(tempP, 1, false)
SetVisibleInView(tempQ, 1, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
center = Midpoint(P, Q)
circ = Circle(center, P)
SetLineStyle(circ, 2)
SetColor(circ, "Gray")
ShowLabel(circ, true)
asym1 = Line((0,0), (1, 1.732))
asym2 = Line((0,0), (1, -1.732))
SetLineStyle(asym1, 3)
SetLineStyle(asym2, 3)
SetColor(asym1, "DarkGray")
SetColor(asym2, "DarkGray")
M = Intersect(l, asym1)
N = Intersect(l, asym2)
ShowLabel(M, true)
