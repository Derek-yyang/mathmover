MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
SetColor(C_curve, "Black")
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
S = (2, 0)
SetFixed(S, true)
ShowLabel(S, true)
SetLabelMode(S, 1)
SetColor(S, "DarkGreen")
lineAsym1 = Line(O, (1, sqrt(3)))
lineAsym2 = Line(O, (1, -sqrt(3)))
SetColor(lineAsym1, "Gray")
SetColor(lineAsym2, "Gray")
SetLineStyle(lineAsym1, 2)
SetLineStyle(lineAsym2, 2)
ShowLabel(lineAsym1, false)
ShowLabel(lineAsym2, false)
k = Slider(-1.6, 1.6, 0.05)
SetCaption(k, "k")
line_l = Line(S, S + (1, k))
SetColor(line_l, "Blue")
P = Intersect(C_curve, line_l, 1)
Q = Intersect(C_curve, line_l, 2)
SetColor(P, "Red")
SetColor(Q, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
centerPQ = Midpoint(P, Q)
ShowLabel(centerPQ, false)
circlePQ = Circle(centerPQ, Distance(P, Q) / 2)
SetColor(circlePQ, "Purple")
ShowLabel(circlePQ, false)
M = Intersect(line_l, lineAsym1)
N = Intersect(line_l, lineAsym2)
SetColor(M, "DarkGreen")
SetColor(N, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
segMN = Segment(M, N)
SetColor(segMN, "Gray")
ShowLabel(segMN, false)
segPQ = Segment(P, Q)
SetColor(segPQ, "Blue")
ShowLabel(segPQ, false)
