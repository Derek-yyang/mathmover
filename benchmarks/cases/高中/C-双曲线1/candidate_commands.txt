MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-2, 0), (2, 0), 1)
O = (0, 0)
A = (-1, 0)
F1 = (-2, 0)
F2 = (2, 0)
asym1 = Line((0, 0), (1, sqrt(3)))
asym2 = Line((0, 0), (1, -sqrt(3)))
k = Slider(-2, 2, 0.01)
m = Slider(-8, 8, 0.01)
l = Line((0, m), (1, k + m))
P = Intersect(l, C, 1)
Q = Intersect(l, C, 2)
M = Intersect(l, asym1)
N = Intersect(l, asym2)
midPQ = Midpoint(P, Q)
circlePQ = Circle(midPQ, Distance(P, Q) / 2)
SetColor(C, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
SetValue(k, 0.5)
SetValue(m, 3)
SetColor(l, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(midPQ, 1, false)
SetColor(circlePQ, "Purple")
SetLineStyle(circlePQ, 2)
ShowLabel(circlePQ, false)
