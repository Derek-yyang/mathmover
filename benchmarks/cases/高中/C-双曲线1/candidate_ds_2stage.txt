MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2*sqrt(3)/3, 0), (2*sqrt(3)/3, 0), sqrt(3))
O = (0, 0)
A = (-sqrt(3), 0)
asym1 = Line((-1, -sqrt(3)), (0, 0))
asym2 = Line((-1, sqrt(3)), (0, 0))
k_slider = Slider(-3, 3, 0.01)
m_slider = Slider(-5, 5, 0.01)
l_line = Line(Point((0, 0) + (0, 1)), Point((0, 0) + (1, 0)))
l_param = Line((0, 0), (1, k_slider))
l = Line((0, m_slider), (1, m_slider + k_slider))
P = Intersect(l, C_curve, 1)
Q = Intersect(l, C_curve, 2)
M = Intersect(l, asym1)
N = Intersect(l, asym2)
circlePQ = Circle(Midpoint(P, Q), Distance(Midpoint(P, Q), P))
V1 = (sqrt(3), 0)
V2 = (-sqrt(3), 0)
SetColor(C_curve, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
SetVisibleInView(asym1, 1, true)
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
SetVisibleInView(asym2, 1, true)
SetValue(k_slider, 1)
SetValue(m_slider, 1)
SetVisibleInView(l_line, 1, false)
SetVisibleInView(l_param, 1, false)
SetColor(l, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(circlePQ, "LightBlue")
SetFilling(circlePQ, 0.2)
ShowLabel(circlePQ, false)
SetVisibleInView(V1, 1, false)
SetVisibleInView(V2, 1, false)
