MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
A = (-1, 0)
V_right = (1, 0)
P0 = (2, 3)
T = (2, 0)
asym1 = Line((0, 0), (1, sqrt(3)))
asym2 = Line((0, 0), (1, -sqrt(3)))
k = Slider(-1.5, 1.5, 0.05)
D = (3, k)
lineL = Line(T, D)
P = Intersect(lineL, C_curve, 1)
Q = Intersect(lineL, C_curve, 2)
M = Intersect(lineL, asym1)
N = Intersect(lineL, asym2)
midPQ = Midpoint(P, Q)
circlePQ = Circle(midPQ, P)
segAP = Segment(A, P)
segAQ = Segment(A, Q)
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(V_right, 1, false)
SetFixed(P0, true)
ShowLabel(P0, true)
SetLabelMode(P0, 1)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
SetValue(k, 1)
SetColor(D, "Red")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetVisibleInView(D, 1, false)
SetColor(lineL, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(M, "DarkGreen")

SetColor(N, "DarkGreen")
