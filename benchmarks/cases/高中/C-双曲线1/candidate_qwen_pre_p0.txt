MODE: 2D
// 1. 基础设置与坐标系
SetAxesVisible(true, true)
SetGridVisible(true)
// 2. 定义双曲线参数 (根据题目推导: a=1, b=sqrt(3), c=2)
// 焦点
F1 = (-2, 0)
F2 = (2, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
// 双曲线 C: x^2 - y^2/3 = 1
// 使用 Hyperbola(F1, F2, a) 构建，其中 a=1
C = Hyperbola(F1, F2, 1)
SetColor(C, "Black")
// 3. 定义关键点与渐近线
// 左顶点 A
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
// 渐近线 y = ±sqrt(3)x
asy1: y = sqrt(3) * x
asy2: y = -sqrt(3) * x
SetColor(asy1, "Gray")
SetLineStyle(asy1, 3)
SetColor(asy2, "Gray")
SetLineStyle(asy2, 3)
// 4. 构建动直线 l (基于问题2结论：恒过定点 F2(2,0))
// 定义斜率 k 的滑块
k = Slider(-5, 5, 0.1)
// 直线 l 过点 F2(2,0)，斜率为 k
// 使用两点式构建直线，避免斜率无穷大问题（虽然 Slider 限制了范围，但两点式更稳健）
// 注意：当 k 接近渐近线斜率时，交点可能趋向无穷远
l = Line(F2, F2 + (1, k))
SetColor(l, "Blue")
ShowLabel(l, true)
SetLabelMode(l, 1)
// 5. 定义交点 P, Q (直线 l 与双曲线 C)
// 获取两个交点
tempP = Intersect(l, C, 1)
tempQ = Intersect(l, C, 2)
// 根据 y 坐标大小分配 P 和 Q (确保 P 在上方，Q 在下方，或者反之，保持一致性)
P = If(y(tempP) > y(tempQ), tempP, tempQ)
Q = If(y(tempP) < y(tempQ), tempP, tempQ)
// 隐藏临时点
SetVisibleInView(tempP, 1, false)
SetVisibleInView(tempQ, 1, false)
// 设置 P, Q 样式
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
// 6. 定义交点 M, N (直线 l 与渐近线)
M = Intersect(l, asy1)
N = Intersect(l, asy2)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Green")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Green")
// 7. 验证问题 (2): 以 PQ 为直径的圆经过点 A
// 构建以 PQ 为直径的圆
circle_PQ = Circle(P, Q)
SetColor(circle_PQ, "Orange")
SetFilling(circle_PQ, 0.1)
SetLineStyle(circle_PQ, 2)
// 验证角 PAQ 是否为 90 度 (显示角度值)
angle_PAQ = Angle(P, A, Q)
// 创建文本显示角度
text_angle = Text("∠PAQ = " + angle_PAQ + "°", (0, 6))
SetColor(text_angle, "Blue")
// 8. 验证问题 (3): P, Q 是否为 MN 的三等分点
// 计算距离
d_MP = Distance(M, P)
d_PQ = Distance(P, Q)
d_QN = Distance(Q, N)
// 创建文本显示距离比
text_ratio1 = Text("MP/PQ = " + (d_MP / d_PQ), (0, 5.5))
text_ratio2 = Text("PQ/QN = " + (d_PQ / d_QN), (0, 5))
SetColor(text_ratio1, "Black")
SetColor(text_ratio2, "Black")
// 9. 辅助线段 (可视化 MN)
seg_MN = Segment(M, N)
SetColor(seg_MN, "Green")
SetLineStyle(seg_MN, 2)
