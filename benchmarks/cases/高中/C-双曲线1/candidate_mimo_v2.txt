MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
a_val = sqrt(3)
b_val = sqrt(12)
c_val = sqrt(a_val^2 + b_val^2)
O = (0, 0)
A_hyperbola = (-a_val, 0)
C_curve = Hyperbola((-c_val, 0), (c_val, 0), a_val)
asym1 = Line(O, O + (a_val, b_val))
asym2 = Line(O, O + (a_val, -b_val))
t_k = Slider(-5, 5, 0.01)
P = Point(C_curve, 0.5)
Q = Point(C_curve, 0.75)
line_l = Line(P, Q)
M_point = Intersect(line_l, asym1)
N_point = Intersect(line_l, asym2)
center_PQ = Midpoint(P, Q)
radius_PQ = Distance(center_PQ, A_hyperbola)
circle_PQ = Circle(center_PQ, radius_PQ)
V1 = Vertex(C_curve, 1)
V2 = Vertex(C_curve, 2)
arc_right = Arc(C_curve, V1, V2)
P_dynamic = Point(arc_right, t_k)
Q_dynamic = Point(arc_right, t_k + 0.1)
line_l_dynamic = Line(P_dynamic, Q_dynamic)
M_dynamic = Intersect(line_l_dynamic, asym1)
N_dynamic = Intersect(line_l_dynamic, asym2)
center_PQ_dynamic = Midpoint(P_dynamic, Q_dynamic)
radius_PQ_dynamic = Distance(center_PQ_dynamic, A_hyperbola)
circle_PQ_dynamic = Circle(center_PQ_dynamic, radius_PQ_dynamic)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A_hyperbola, true)
ShowLabel(A_hyperbola, true)
SetLabelMode(A_hyperbola, 1)
SetColor(C_curve, "Black")
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
ShowLabel(t_k, true)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(line_l, "Blue")
SetColor(M_point, "DarkGreen")
ShowLabel(M_point, true)
SetLabelMode(M_point, 1)
SetColor(N_point, "DarkGreen")
ShowLabel(N_point, true)
SetLabelMode(N_point, 1)
SetColor(circle_PQ, "Orange")
SetLineStyle(circle_PQ, 2)
SetVisibleInView(V1, 1, false)
SetVisibleInView(V2, 1, false)
SetColor(P_dynamic, "Red")
ShowLabel(P_dynamic, true)
SetLabelMode(P_dynamic, 1)
SetColor(Q_dynamic, "Red")
ShowLabel(Q_dynamic, true)
SetLabelMode(Q_dynamic, 1)
SetColor(line_l_dynamic, "Blue")
SetColor(M_dynamic, "DarkGreen")
ShowLabel(M_dynamic, true)
SetLabelMode(M_dynamic, 1)
SetColor(N_dynamic, "DarkGreen")
ShowLabel(N_dynamic, true)
SetLabelMode(N_dynamic, 1)
SetColor(circle_PQ_dynamic, "Orange")
SetLineStyle(circle_PQ_dynamic, 2)
