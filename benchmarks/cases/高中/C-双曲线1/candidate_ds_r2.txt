MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
O = (0, 0)
D = (2, 3)
A = (-1, 0)
T = (2, 0)
asym1 = Line((0, 0), (1, sqrt(3)))
asym2 = Line((0, 0), (1, -sqrt(3)))
k = Slider(-2.5, 2.5, 0.01)
l = Line(T, T + (1, k))
P = Intersect(l, C_curve, 1)
Q = Intersect(l, C_curve, 2)
circlePQ = Circle(Midpoint(P, Q), P)
M = Intersect(l, asym1)
N = Intersect(l, asym2)
SetColor(C_curve, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
SetColor(asym1, "Gray")
SetColor(asym2, "Gray")
SetLineStyle(asym1, 1)
SetLineStyle(asym2, 1)
SetValue(k, 0.5)
SetColor(l, "Blue")
SetColor(P, "Red")
SetColor(Q, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(circlePQ, "LightBlue")
SetColor(M, "DarkGreen")
SetColor(N, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
