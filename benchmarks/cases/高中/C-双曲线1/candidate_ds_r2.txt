MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
A = (-1, 0)
O = (0, 0)
asym1 = Line((0, 0), (1, sqrt(3)))
asym2 = Line((0, 0), (1, -sqrt(3)))
t_line = Slider(-3, 3, 0.01)
m_line = Slider(-3, 3, 0.01)
l_line = Line((0, m_line), (1, t_line + m_line))
P = Intersect(l_line, C_curve, 1)
Q = Intersect(l_line, C_curve, 2)
M = Intersect(l_line, asym1)
N = Intersect(l_line, asym2)
set1_circle = Circle(A, M)
set1_circled = Circle(A, N)
cond1 = (Distance(A, M) == Distance(A, N))
cond2 = (Vector(A, P) == Vector(M, Q))
cond3 = (Vector(A, Q) == Vector(N, P))
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(asym1, "Gray")
SetLineStyle(asym1, 2)
SetColor(asym2, "Gray")
SetLineStyle(asym2, 2)
SetColor(l_line, "Blue")
SetColor(P, "Red")
SetColor(Q, "Red")
SetColor(M, "DarkGreen")
SetColor(N, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(set1_circle, 1, false)
SetVisibleInView(set1_circled, 1, false)
SetVisibleInView(cond1, 1, false)
SetVisibleInView(cond2, 1, false)
SetVisibleInView(cond3, 1, false)
