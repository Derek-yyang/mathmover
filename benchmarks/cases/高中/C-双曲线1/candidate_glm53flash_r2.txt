MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-2, 0), (2, 0), 1)
F1 = (-2, 0)
F2 = (2, 0)
B = (2, 3)
A = (-1, 0)
T = (2, 0)
asymp1 = Line((0, 0), (1, sqrt(3)))
asymp2 = Line((0, 0), (1, -sqrt(3)))
k = Slider(-1.5, 1.5, 0.05)
l = Line(T, T + (1, k))
P = Intersect(l, C, 1)
Q = Intersect(l, C, 2)
circPQ = Circle(P, Q)
M = Intersect(l, asymp1)
N = Intersect(l, asymp2)
E = M + (1 / 3) * (N - M)
H = M + (2 / 3) * (N - M)
segMN = Segment(M, N)
SetColor(C, "Black")
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetPointSize(F1, 4)
SetColor(F1, "DarkGreen")
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetPointSize(F2, 4)
SetColor(F2, "DarkGreen")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetPointSize(B, 4)
SetColor(B, "Red")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
SetColor(T, "DarkGreen")
SetColor(asymp1, "Gray")
SetLineStyle(asymp1, 1)
SetColor(asymp2, "Gray")
SetLineStyle(asymp2, 1)
SetValue(k, 0.5)
SetColor(l, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
SetColor(circPQ, "Green")
SetFilling(circPQ, 0.15)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "DarkGreen")
SetVisibleInView(E, 1, false)
SetVisibleInView(H, 1, false)
SetColor(segMN, "Purple")
SetLineStyle(segMN, 2)
