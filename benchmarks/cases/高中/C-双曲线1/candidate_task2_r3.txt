SetAxesVisible(true, true)
SetGridVisible(true)

O = (0, 0)
A = (-1, 0)
B = (1, 0)
lineAsym1 = Line((0, 0), (1, sqrt(3)))
lineAsym2 = Line((0, 0), (1, -sqrt(3)))
T = (2, 0)
k = Slider(-1.5, 1.5, 0.01)
lineL = Line(T, T+(1, k))
P = Intersect(lineL, C_curve, 1)
Q = Intersect(lineL, C_curve, 2)
M = Intersect(lineL, lineAsym1)
N = Intersect(lineL, lineAsym2)
midPQ = Midpoint(P, Q)
circlePQ = Circle(midPQ, P)
segAP = Segment(A, P)
segAQ = Segment(A, Q)
segMN = Segment(M, N)
SetColor(C_curve, "Black")
SetFixed(O, true)

SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(lineAsym1, "Gray")

SetColor(lineAsym2, "Gray")
SetLineStyle(lineAsym2, 2)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
SetColor(lineL, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
ShowLabel(N, true)

ShowLabel(midPQ, false)
