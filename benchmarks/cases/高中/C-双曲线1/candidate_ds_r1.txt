MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-2, 0), (2, 0), 1)
O = (0, 0)
A = (-1, 0)
T = (2, 0)
line_asym1 = Line((0, 0), (1, sqrt(3)))
line_asym2 = Line((0, 0), (1, -sqrt(3)))
k = Slider(-1.5, 1.5, 0.01)
line_l = Line(T, T + (1, k))
P = Intersect(line_l, C_curve, 1)
Q = Intersect(line_l, C_curve, 2)
circlePQ = Circle(Midpoint(P, Q), P)
M = Intersect(line_l, line_asym1)
N = Intersect(line_l, line_asym2)
segmentMN = Segment(M, N)
SetColor(C_curve, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
SetColor(line_asym1, "Gray")
SetLineStyle(line_asym1, 2)
ShowLabel(line_asym1, false)
SetColor(line_asym2, "Gray")
SetLineStyle(line_asym2, 2)
ShowLabel(line_asym2, false)
SetValue(k, 1)
SetColor(line_l, "Blue")
ShowLabel(line_l, true)
SetLabelMode(line_l, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(circlePQ, "Purple")
ShowLabel(circlePQ, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "Red")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segmentMN, "Blue")
ShowLabel(segmentMN, false)
