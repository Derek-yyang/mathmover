MODE: 2D
SetAxesVisible(true, true)
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
F1 = (-2, 0)
SetFixed(F1, true)
SetColor(F1, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
F2 = (2, 0)
SetFixed(F2, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
C = Hyperbola(F1, F2, 1)
SetColor(C, "Black")
A = (-1, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (2, 3)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
T = (2, 0)
SetFixed(T, true)
ShowLabel(T, true)
SetLabelMode(T, 1)
asym1 = Line(O, O + (1, 1.732))
SetLineStyle(asym1, 3)
SetColor(asym1, "DarkGray")
asym2 = Line(O, O + (1, -1.732))
SetLineStyle(asym2, 3)
SetColor(asym2, "DarkGray")
k = Slider(-5, 5, 0.1)
l = Line(T, T + (1, k))
SetColor(l, "Blue")
ShowLabel(l, true)
M = Intersect(l, asym1)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "DarkGreen")
N = Intersect(l, asym2)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "DarkGreen")
tempP = Intersect(l, C, 1)
tempQ = Intersect(l, C, 2)
P = If(y(tempP) > y(tempQ), tempP, tempQ)
Q = If(y(tempP) < y(tempQ), tempP, tempQ)
SetVisibleInView(tempP, 1, false)
SetVisibleInView(tempQ, 1, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Blue")
circlePQ = Circle(Midpoint(P, Q), P)
SetLineStyle(circlePQ, 2)
SetColor(circlePQ, "Gray")
segAP = Segment(A, P)
SetLineStyle(segAP, 2)
SetColor(segAP, "Red")
segAQ = Segment(A, Q)
SetLineStyle(segAQ, 2)
SetColor(segAQ, "Blue")
