MODE: 2D
SetAxesVisible(true, true)
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
c_val = sqrt(5)
F1 = (-c_val, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetLabelMode(F1, 1)
SetVisibleInView(F1, 1, true)
F2 = (c_val, 0)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetLabelMode(F2, 1)
SetVisibleInView(F2, 1, true)
E = Ellipse(F1, F2, 3)
SetColor(E, "Black")
arc = Arc(E, D, A)
t = Slider(0, 0.99, 0.01)
P = Point(arc, t)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
l_BC = Line(B, C)
SetVisibleInView(l_BC, 1, true)
l_PD = Line(P, D)
SetVisibleInView(l_PD, 1, true)
M = Intersect(l_PD, l_BC)
SetColor(M, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetVisibleInView(M, 1, true)
l_PA = Line(P, A)
SetVisibleInView(l_PA, 1, true)
l_bottom = Line(C, C + (1, 0))
SetLineStyle(l_bottom, 2)
SetColor(l_bottom, "Gray")
SetVisibleInView(l_bottom, 1, true)
N = Intersect(l_PA, l_bottom)
SetColor(N, "Blue")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(N, 1, true)
seg_CD = Segment(C, D)
SetColor(seg_CD, "Green")
SetLineStyle(seg_CD, 3)
SetVisibleInView(seg_CD, 1, true)
seg_MN = Segment(M, N)
SetColor(seg_MN, "Green")
SetLineStyle(seg_MN, 3)
SetVisibleInView(seg_MN, 1, true)
