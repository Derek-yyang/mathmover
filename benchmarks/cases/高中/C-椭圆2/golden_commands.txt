MODE: 2D
SetAxesVisible(true, true)
F1 = (-2.24, 0)
F2 = (2.24, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetVisibleInView(F1, 1, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetVisibleInView(F2, 1, true)
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
E = Ellipse(F1, F2, 3)
SetColor(E, "Black")
ShowLabel(E, true)
arcP = Arc(E, D, A)
SetVisibleInView(arcP, 1, false)
t = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
l_BC = Line(B, C)
SetColor(l_BC, "Gray")
SetLineStyle(l_BC, 2)
l_PD = Line(P, D)
SetColor(l_PD, "Blue")
M = Intersect(l_PD, l_BC)
SetColor(M, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
l_PA = Line(P, A)
SetColor(l_PA, "Blue")
l_y_neg2 = Line(C, C + (1, 0))
SetColor(l_y_neg2, "Gray")
SetLineStyle(l_y_neg2, 2)
N = Intersect(l_PA, l_y_neg2)
SetColor(N, "Blue")
ShowLabel(N, true)
SetLabelMode(N, 1)
seg_MN = Segment(M, N)
SetColor(seg_MN, "DarkGreen")
SetLineStyle(seg_MN, 1)
ShowLabel(seg_MN, true)
seg_CD = Segment(C, D)
SetColor(seg_CD, "DarkGreen")
SetLineStyle(seg_CD, 1)
ShowLabel(seg_CD, true)
SetVisibleInView(l_PD, 1, true)
SetVisibleInView(l_PA, 1, true)
