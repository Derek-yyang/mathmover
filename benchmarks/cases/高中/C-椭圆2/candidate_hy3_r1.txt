MODE: 2D
setaxesvisible(true, true)
setgridvisible(true)
f(x) = -x^2 + 2x + 3
A = (0, 3)
B = (-1, 0)
D = (2, 3)
E = (root(f, 2, 4), 0)
C = B + D - A
polyABCD = polygon(A, B, C, D)
M = (A + C) / 2
l = line(E, M)
F = intersect(l, f, 2)
t = slider(-0.5, 3, 0.05)
P = (t, f(t))
polyPFE = polygon(P, F, E)
setcolor(f, "Black")
setfixed(A, true)
showlabel(A, true)
setlabelmode(A, 1)
setfixed(B, true)
showlabel(B, true)
setlabelmode(B, 1)
setfixed(D, true)
showlabel(D, true)
setlabelmode(D, 1)
setcolor(E, "Black")
setpointsize(E, 4)
setlabelmode(E, 1)
showlabel(E, true)
setfixed(C, true)
showlabel(C, true)
setlabelmode(C, 1)
setfilling(polyABCD, 0.2)
setcolor(polyABCD, "Blue")
setfixed(M, true)
showlabel(M, true)
setlabelmode(M, 1)
setcolor(l, "Green")
setlinestyle(l, 2)
setcolor(F, "Black")
setpointsize(F, 4)
setlabelmode(F, 1)
showlabel(F, true)
setcolor(P, "Red")
showlabel(P, true)
setlabelmode(P, 1)
setfilling(polyPFE, 0.3)
setcolor(polyPFE, "Red")
