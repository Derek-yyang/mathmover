MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
ellipse = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
A = (0, 2)
C = (0, -2)
B = (-3, 0)
D = (3, 0)
arcP = Arc(ellipse, D, A)
t_p = Slider(0, 1, 0.01)
P = Point(arcP, t_p)
lineBC = Line(B, C)
linePD = Line(P, D)
linePA = Line(P, A)
lineYeq2 = Line(C, (1, -2))
M = Intersect(linePD, lineBC)
N = Intersect(linePA, lineYeq2)
segmentMN = Segment(M, N)
segmentCD = Segment(C, D)
SetColor(ellipse, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineBC, "Blue")
SetColor(linePD, "Blue")
SetColor(linePA, "Blue")
SetColor(lineYeq2, "Gray")
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segmentMN, "Black")
SetColor(segmentCD, "Black")
