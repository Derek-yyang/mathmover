MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_ellipse = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
A = (0, 2)
C = (0, -2)
B = (-3, 0)
D = (3, 0)
V_top = A
V_right = D
arcP = Arc(C_ellipse, A, D)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
linePD = Line(P, D)
lineBC = Line(B, C)
M = Intersect(linePD, lineBC)
linePA = Line(P, A)
line_y_eq = Line((0, -2), (1, -2))
N = Intersect(linePA, line_y_eq)
segMN = Segment(M, N)
segCD = Segment(C, D)
SetColor(C_ellipse, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePD, "Blue")
SetColor(lineBC, "Blue")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Red")
SetColor(linePA, "Blue")
SetColor(line_y_eq, "Gray")
SetLineStyle(line_y_eq, 2)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Red")
SetColor(segMN, "Green")
SetLineThickness(segMN, 4)
SetColor(segCD, "Green")
SetLineThickness(segCD, 4)
