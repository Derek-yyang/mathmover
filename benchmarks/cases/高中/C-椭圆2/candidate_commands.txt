MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Ellipse((0, 0), (0, 0), 3, 2)
A = (0, 2)
C_pt = (0, -2)
B = (-3, 0)
D = (3, 0)
V_top = (0, 2)
V_right = (3, 0)
arcP = Arc(C, V_right, V_top)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
linePA = Line(P, A)
linePD = Line(P, D)
lineBC = Line(B, C_pt)
M = Intersect(linePD, lineBC)
line_y_neg2 = Line((0, -2), (1, -2))
N = Intersect(linePA, line_y_neg2)
lineMN = Line(M, N)
lineCD = Line(C_pt, D)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(C_pt, true)
ShowLabel(C_pt, true)
SetLabelMode(C_pt, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetVisibleInView(V_top, 1, false)
SetVisibleInView(V_right, 1, false)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePA, "Blue")
SetColor(linePD, "Blue")
SetColor(lineBC, "DarkGreen")
SetColor(M, "Purple")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetVisibleInView(line_y_neg2, 1, false)
SetColor(N, "Orange")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(lineMN, "Red")
SetLineStyle(lineMN, 1)
SetColor(lineCD, "Blue")
SetLineStyle(lineCD, 1)
