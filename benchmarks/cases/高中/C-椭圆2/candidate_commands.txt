MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
A = (0, 2)
C = (0, -2)
B = (-3, 0)
D = (3, 0)
arcP = Arc(E, D, A)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
linePD = Line(P, D)
lineBC = Line(B, C)
linePA = Line(P, A)
line_y_neg2 = Line((0, -2), (1, -2))
M = Intersect(linePD, lineBC)
N = Intersect(linePA, line_y_neg2)
segMN = Segment(M, N)
segCD = Segment(C, D)
SetColor(E, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetVisibleInView(t_p, 1, false)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePD, "Blue")
SetColor(lineBC, "Blue")
SetColor(linePA, "Blue")
SetColor(line_y_neg2, "Gray")
SetLineStyle(line_y_neg2, 2)
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segMN, "Red")
SetColor(segCD, "Black")
