MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
A = (0, 2)
B = (-3, 0)
C = (0, -2)
D = (3, 0)
arcP = Arc(E, D, A)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
lineBC = Line(B, C)
linePD = Line(P, D)
linePA = Line(P, A)
lineH = Line((0, -2), (1, -2))
M = Intersect(linePD, lineBC)
N = Intersect(linePA, lineH)
lineCD = Line(C, D)
lineMN = Line(M, N)
SetColor(E, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineBC, "Blue")
SetColor(linePD, "Blue")
SetColor(linePA, "Blue")
SetColor(lineH, "Gray")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "DarkGreen")
SetColor(lineCD, "Purple")
SetColor(lineMN, "Orange")
