MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
SetColor(E, "Black")
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
arcP = Arc(E, D, A)
SetColor(arcP, "Red")
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
linePD = Line(P, D)
SetColor(linePD, "Blue")
lineBC = Line(B, C)
SetColor(lineBC, "Blue")
linePA = Line(P, A)
SetColor(linePA, "Blue")
lineY = Line((0, -2), (1, -2))
SetColor(lineY, "Blue")
SetLinestyle(lineY, 2)
M = Intersect(linePD, lineBC)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "DarkGreen")
N = Intersect(linePA, lineY)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "DarkGreen")
lineCD = Line(C, D)
SetColor(lineCD, "Orange")
SetThickness(lineCD, 3)
lineMN = Line(M, N)
SetColor(lineMN, "Purple")
SetThickness(lineMN, 3)
