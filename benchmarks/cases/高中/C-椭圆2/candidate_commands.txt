lineBC = Line(B, C)
M = Intersect(linePD, lineBC)
linePA = Line(P, A)
line_y_neg2 = Line((0, -2), (1, -2))
N = Intersect(linePA, line_y_neg2)
lineMN = Line(M, N)
lineCD = Line(C, D)

SetLabelMode(A, 1)
