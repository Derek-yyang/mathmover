MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
A = (0, 2)
C = (0, -2)
B = (-3, 0)
D = (3, 0)
arcP = Arc(E, D, A)
t_p = Slider(0.001, 0.999, 0.001)
P = Point(arcP, t_p)
linePD = Line(P, D)
lineBC = Line(B, C)
M = Intersect(linePD, lineBC)
linePA = Line(P, A)
line_h = Line((-5, -2), (5, -2))
N = Intersect(linePA, line_h)
segMN = Segment(M, N)
segCD = Segment(C, D)
SetColor(E, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(linePD, "Blue")
SetColor(lineBC, "Blue")
SetColor(M, "DarkGreen")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(linePA, "Blue")
SetVisibleInView(line_h, 1, false)
SetColor(N, "DarkGreen")
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(segMN, "Red")
SetLineStyle(segMN, 1)
SetColor(segCD, "Gray")
SetLineStyle(segCD, 2)
