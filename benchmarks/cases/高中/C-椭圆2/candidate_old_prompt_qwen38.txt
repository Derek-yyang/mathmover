MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
F1 = (-sqrt(5), 0)
SetFixed(F1, true)
ShowLabel(F1, false)
F2 = (sqrt(5), 0)
SetFixed(F2, true)
ShowLabel(F2, false)
E = Ellipse(F1, F2, 3)
SetColor(E, "Black")
SetVisibleInView(E, 1, true)
arcP = Arc(E, A, D)
t = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
l_BC = Line(B, C)
SetVisibleInView(l_BC, 1, true)
l_PD = Line(P, D)
SetVisibleInView(l_PD, 1, true)
l_PA = Line(P, A)
SetVisibleInView(l_PA, 1, true)
l_y2 = Line(C, C + (1, 0))
SetVisibleInView(l_y2, 1, true)
l_CD = Line(C, D)
SetVisibleInView(l_CD, 1, true)
M = Intersect(l_PD, l_BC)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetVisibleInView(M, 1, true)
N = Intersect(l_PA, l_y2)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetVisibleInView(N, 1, true)
l_MN = Line(M, N)
SetVisibleInView(l_MN, 1, true)
