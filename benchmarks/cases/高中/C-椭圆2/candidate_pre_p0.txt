MODE: 2D

SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
F2 = (2.236, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
E = Ellipse(F1, F2, 3)
SetColor(E, "Black")
SetVisibleInView(E, 1, true)
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
arcP = Arc(E, D, A)
t = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
linePD = Line(P, D)
SetColor(linePD, "Black")
SetVisibleInView(linePD, 1, true)
lineBC = Line(B, C)
SetColor(lineBC, "Black")
SetVisibleInView(lineBC, 1, true)
M = Intersect(linePD, lineBC)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
linePA = Line(P, A)
SetColor(linePA, "Black")
SetVisibleInView(linePA, 1, true)
lineY2 = Line(C, C + (1, 0))
SetColor(lineY2, "Gray")
SetLineStyle(lineY2, 2)
SetVisibleInView(lineY2, 1, true)
N = Intersect(linePA, lineY2)
ShowLabel(N, true)
SetLabelMode(N, 1)
SetColor(N, "Blue")

SetColor(lineCD, "Gray")

SetVisibleInView(lineCD, 1, true)
