MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
E_curve = Ellipse((-sqrt(5), 0), (sqrt(5), 0), 3)
SetColor(E_curve, "Black")
A = (0, 2)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (-3, 0)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
C = (0, -2)
SetFixed(C, true)
ShowLabel(C, true)
SetLabelMode(C, 1)
D = (3, 0)
SetFixed(D, true)
ShowLabel(D, true)
SetLabelMode(D, 1)
arcP = Arc(E_curve, D, A)
SetVisibleInView(arcP, 1, false)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(arcP, t_p)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
linePD = Line(P, D)
SetColor(linePD, "Black")
lineBC = Line(B, C)
SetColor(lineBC, "Black")
M = Intersect(linePD, lineBC)
SetColor(M, "Black")
ShowLabel(M, true)
SetLabelMode(M, 1)
linePA = Line(P, A)
SetColor(linePA, "Black")
line_y_neg2 = Line((0, -2), (1, -2))
SetColor(line_y_neg2, "Black")
N = Intersect(linePA, line_y_neg2)
SetColor(N, "Black")
ShowLabel(N, true)
SetLabelMode(N, 1)
lineMN = Line(M, N)
SetColor(lineMN, "Black")
lineCD = Line(C, D)
SetColor(lineCD, "Black")
