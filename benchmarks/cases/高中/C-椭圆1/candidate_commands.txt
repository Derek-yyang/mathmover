MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Ellipse((-sqrt(3), 0), (sqrt(3), 0), 2)
P = (-sqrt(3), 1/2)
O = (0, 0)
A = (-2, 0)
B = (0, -1)
V_right = (2, 0)
V_top = (0, 1)
arcM = Arc(C_curve, V_right, V_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
lineMB = Line(M, B)
lineMA = Line(M, A)
segAB = Segment(A, B)
xAxis_line = Line((0, 0), (1, 0))
yAxis_line = Line((0, 0), (0, 1))
C = Intersect(lineMB, xAxis_line)
D = Intersect(lineMA, yAxis_line)
triOCD = Polygon(O, C, D)
SetColor(C_curve, "Black")
SetLinethickness(C_curve, 2)
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Orange")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(O, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
SetColor(B, "Black")
SetFixed(V_right, true)
ShowLabel(V_right, false)
SetVisibleInView(V_right, 1, false)
SetFixed(V_top, true)
ShowLabel(V_top, false)
SetVisibleInView(V_top, 1, false)
ShowLabel(arcM, false)
SetVisibleInView(arcM, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(lineMB, "Blue")
SetLinethickness(lineMB, 2)
SetColor(lineMA, "Blue")
SetLinethickness(lineMA, 2)
SetColor(segAB, "Black")
SetLinethickness(segAB, 1)
SetVisibleInView(xAxis_line, 1, false)
SetVisibleInView(yAxis_line, 1, false)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "DarkGreen")
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "DarkGreen")
SetColor(triOCD, "LightBlue")
SetFilling(triOCD, 0.3)
ShowLabel(triOCD, false)
