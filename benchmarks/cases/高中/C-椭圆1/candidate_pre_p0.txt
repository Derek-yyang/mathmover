MODE: 2D
SetAxesVisible(true, true)
// 1. 定义椭圆参数与静态定点
// 椭圆 C: x^2/4 + y^2 = 1 => a=2, b=1, c=sqrt(3) ≈ 1.732
F1 = (-1.732, 0)
F2 = (1.732, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetLabelMode(F1, 1)
ShowLabel(F1, true)
SetFixed(F1, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetLabelMode(F2, 1)
ShowLabel(F2, true)
SetFixed(F2, true)
CurveC = Ellipse(F1, F2, 2)
SetColor(CurveC, "Black")
// 椭圆中心 O
O = (0, 0)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
// 左顶点 A, 下顶点 B
A = (-2, 0)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
B = (0, -1)
SetFixed(B, true)
ShowLabel(B, true)
SetLabelMode(B, 1)
// 文本中提到的已知点 P
P = (-1.732, 0.5)
SetFixed(P, true)
ShowLabel(P, true)
SetLabelMode(P, 1)
// 2. 构建第一象限内的动点 M
// 第一象限对应右顶点到上顶点的弧
V_Right = (2, 0)
V_Top = (0, 1)
arcM = Arc(CurveC, V_Right, V_Top)
SetVisibleInView(arcM, 1, false)
t = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
// 3. 构建直线与交点
// 直线 MB 与 x 轴交于点 C
lineMB = Line(M, B)
xAxis = Line((0,0), (1,0))
SetVisibleInView(xAxis, 1, true)
C = Intersect(lineMB, xAxis)
ShowLabel(C, true)
SetLabelMode(C, 1)
SetColor(C, "Blue")
// 直线 MA 与 y 轴交于点 D
lineMA = Line(M, A)
yAxis = Line((0,0), (0,1))
SetVisibleInView(yAxis, 1, true)
D = Intersect(lineMA, yAxis)
ShowLabel(D, true)
SetLabelMode(D, 1)
SetColor(D, "Blue")
// 4. 构建目标三角形 OCD
triOCD = Polygon(O, C, D)
SetColor(triOCD, "LightBlue")
SetFilling(triOCD, 0.3)
ShowLabel(triOCD, true)
// 5. 辅助线样式
SetLineStyle(lineMB, 2)
SetColor(lineMB, "Gray")
SetLineStyle(lineMA, 2)
SetColor(lineMA, "Gray")
// 确保所有点可见
SetVisibleInView(O, 1, true)
SetVisibleInView(A, 1, true)
SetVisibleInView(B, 1, true)
SetVisibleInView(M, 1, true)
SetVisibleInView(C, 1, true)
SetVisibleInView(D, 1, true)
SetVisibleInView(P, 1, true)
SetVisibleInView(CurveC, 1, true)
