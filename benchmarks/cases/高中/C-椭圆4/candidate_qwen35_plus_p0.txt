MODE: 2D
SetAxesVisible(true, true)
F1 = (-4, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetVisibleInView(F1, 1, true)
F2 = (4, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetVisibleInView(F2, 1, true)
C = Ellipse(F1, F2, 6)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
A = (6, 0)
SetFixed(A, true)
ShowLabel(A, false)
SetVisibleInView(A, 1, false)
B = (0, sqrt(20))
SetFixed(B, true)
ShowLabel(B, false)
SetVisibleInView(B, 1, false)
arc = Arc(C, A, B)
SetVisibleInView(arc, 1, false)
t = Slider(0, 1, 0.01)
M = Point(arc, t)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
SetVisibleInView(M, 1, true)
tri = Polygon(M, F1, F2)
SetFilling(tri, 0.2)
SetColor(tri, "LightBlue")
SetVisibleInView(tri, 1, true)
M_target = (3, sqrt(15))
ShowLabel(M_target, true)
SetLabelMode(M_target, 1)
SetColor(M_target, "Red")
SetPointSize(M_target, 5)
SetVisibleInView(M_target, 1, true)
