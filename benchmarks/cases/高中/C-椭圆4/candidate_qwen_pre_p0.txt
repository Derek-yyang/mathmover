MODE: 2D
F1 = (-4, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetVisibleInView(F1, 1, true)
F2 = (4, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetVisibleInView(F2, 1, true)
C = Ellipse(F1, F2, 6)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
arc1 = Arc(C, (6, 0), (0, sqrt(20)))
SetColor(arc1, "Gray")
SetLineStyle(arc1, 2)
SetVisibleInView(arc1, 1, true)
t = Slider(0, 1, 0.01)
P = Point(arc1, t)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(P, 1, true)
M = (3, sqrt(15))
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(M, "Blue")
SetPointSize(M, 5)
SetVisibleInView(M, 1, true)
tri = Polygon(F1, F2, M)
SetColor(tri, "LightBlue")
SetFilling(tri, 0.3)
SetVisibleInView(tri, 1, true)
SetAxesVisible(true, true)
