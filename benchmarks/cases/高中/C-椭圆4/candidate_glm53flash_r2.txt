MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Ellipse((-4, 0), (4, 0), 6)
F1 = Focus(C, 1)
F2 = Focus(C, 2)
V_top = (0, sqrt(20))
V_right = (6, 0)
arcM = Arc(C, V_right, V_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
triMF = Polygon(M, F1, F2)
SetColor(C, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "DarkGreen")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "DarkGreen")
SetVisibleInView(V_top, 1, false)
SetVisibleInView(V_right, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(triMF, "LightBlue")
SetFilling(triMF, 0.3)
ShowLabel(triMF, false)
