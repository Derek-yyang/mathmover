MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Ellipse((-4, 0), (4, 0), 6)
F1 = Element(Focus(C), 1)
F2 = Element(Focus(C), 2)
V1 = (6, 0)
V2 = (0, 2*sqrt(5))
arcM = Arc(C, V1, V2)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
triMF1F2 = Polygon(M, F1, F2)
circF1 = Circle(F1, Distance(F1, F2))
circF2 = Circle(F2, Distance(F1, F2))
SetColor(C, "Black")
SetPointSize(F1, 4)
SetPointSize(F2, 4)
SetFixed(F1, true)
SetFixed(F2, true)
ShowLabel(F1, true)
ShowLabel(F2, true)
SetLabelMode(F1, 1)
SetLabelMode(F2, 1)
SetVisibleInView(V1, 1, false)
SetVisibleInView(V2, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.3)
ShowLabel(triMF1F2, false)
SetColor(circF1, "Gray")
SetLineStyle(circF1, 2)
SetColor(circF2, "Gray")
SetLineStyle(circF2, 2)
