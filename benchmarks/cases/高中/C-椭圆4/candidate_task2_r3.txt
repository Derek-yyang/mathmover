MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Ellipse((-4, 0), (4, 0), 6)
O = (0, 0)
F1 = (-4, 0)
F2 = (4, 0)
V_right = (6, 0)
V_top = (0, 2*sqrt(5))
arcM = Arc(C_curve, V_right, V_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
lineMF1 = Line(M, F1)
lineMF2 = Line(M, F2)
lineF1F2 = Line(F1, F2)
triMF1F2 = Polygon(M, F1, F2)
SetColor(C_curve, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetPointSize(F1, 4)
SetColor(F1, "DarkBlue")
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetPointSize(F2, 4)
SetColor(F2, "DarkBlue")
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(lineMF1, "Blue")
SetColor(lineMF2, "Blue")
SetColor(lineF1F2, "Blue")
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.3)
ShowLabel(triMF1F2, false)
