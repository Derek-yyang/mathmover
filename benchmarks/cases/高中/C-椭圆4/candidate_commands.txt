MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-4, 0)
F2 = (4, 0)
C = Ellipse(F1, F2, 6)
cF1 = Circle(F1, Distance(F1, F2))
M_pos = Intersect(C, cF1, 1)
M_neg = Intersect(C, cF1, 2)
M = If(y(M_pos) > 0, M_pos, M_neg)
tri = Polygon(M, F1, F2)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(C, "Black")
SetVisibleInView(cF1, 1, false)
SetVisibleInView(M_pos, 1, false)
SetVisibleInView(M_neg, 1, false)
SetColor(M, "Red")
SetFixed(M, true)
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(tri, "LightBlue")
SetFilling(tri, 0.3)
ShowLabel(tri, false)
