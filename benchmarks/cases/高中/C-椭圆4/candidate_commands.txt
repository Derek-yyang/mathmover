MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-4, 0)
F2 = (4, 0)
C = Ellipse(F1, F2, 6)
V_right = (6, 0)
V_top = (0, sqrt(20))
arc1 = Arc(C, V_right, V_top)
t = Slider(0, 1, 0.01)
M = Point(arc1, t)
tri = Polygon(M, F1, F2)
line_bisector = Line((0, 0), (0, 1))
M1 = Intersect(line_bisector, C, 1)
circle1 = Circle(F1, 8)
intersections2 = Intersect(circle1, C)
M2_candidate1 = Element(intersections2, 1)
M2_candidate2 = Element(intersections2, 2)
M2 = If(y(M2_candidate1) > 0, M2_candidate1, M2_candidate2)
circle2 = Circle(F2, 8)
O = (0, 0)
A_left = (-6, 0)
A_right = (6, 0)
B_top = (0, sqrt(20))
B_bottom = (0, -sqrt(20))
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(C, "Black")
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(tri, "LightBlue")
SetFilling(tri, 0.3)
ShowLabel(tri, false)
SetVisibleInView(line_bisector, 1, false)
SetColor(M1, "Green")
ShowLabel(M1, true)
SetLabelMode(M1, 1)
SetColor(circle1, "Gray")
SetLineStyle(circle1, 2)
SetColor(M2, "Green")
ShowLabel(M2, true)
SetLabelMode(M2, 1)
SetColor(circle2, "Gray")
SetLineStyle(circle2, 2)
SetVisibleInView(circle2, 1, false)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A_left, true)
ShowLabel(A_left, true)
SetLabelMode(A_left, 1)
SetFixed(A_right, true)
ShowLabel(A_right, true)
SetLabelMode(A_right, 1)
SetFixed(B_top, true)
ShowLabel(B_top, true)
SetLabelMode(B_top, 1)
SetFixed(B_bottom, true)
ShowLabel(B_bottom, true)
SetLabelMode(B_bottom, 1)
