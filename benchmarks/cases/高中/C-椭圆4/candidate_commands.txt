MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Ellipse((-4, 0), (4, 0), 6)
SetColor(C_curve, "Black")
F1 = (-4, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
F2 = (4, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
V_right = (6, 0)
V_top = (0, 2*sqrt(5))
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
arcM = Arc(C_curve, V_right, V_top)
circleF1 = Circle(F1, 8)
SetColor(circleF1, "Gray")
SetLineStyle(circleF1, 2)
SetVisibleInView(circleF1, 1, false)
circleF2 = Circle(F2, 8)
SetColor(circleF2, "Gray")
SetLineStyle(circleF2, 2)
SetVisibleInView(circleF2, 1, false)
M1 = Intersect(circleF1, arcM)
SetColor(M1, "Red")
ShowLabel(M1, true)
SetLabelMode(M1, 1)
M2 = Intersect(circleF2, arcM)
SetColor(M2, "Red")
ShowLabel(M2, true)
SetLabelMode(M2, 1)
lineM1F1 = Segment(M1, F1)
SetColor(lineM1F1, "Blue")
lineM1F2 = Segment(M1, F2)
SetColor(lineM1F2, "Blue")
lineM2F1 = Segment(M2, F1)
SetColor(lineM2F1, "DarkGreen")
lineM2F2 = Segment(M2, F2)
SetColor(lineM2F2, "DarkGreen")
tri1 = Polygon(M1, F1, F2)
SetColor(tri1, "LightBlue")
SetFilling(tri1, 0.2)
ShowLabel(tri1, false)
tri2 = Polygon(M2, F1, F2)
SetColor(tri2, "LightYellow")
SetFilling(tri2, 0.2)
ShowLabel(tri2, false)
