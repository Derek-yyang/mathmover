MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Ellipse((-4, 0), (4, 0), 6)
A_left = (-6, 0)
A_right = (6, 0)
B_top = (0, sqrt(20))
B_bottom = (0, -sqrt(20))
F1 = Focus(C_curve, 1)
F2 = Focus(C_curve, 2)
arcM = Arc(C_curve, A_right, B_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
l1 = Segment(M, F1)
l2 = Segment(M, F2)
l3 = Segment(F1, F2)
triMF1F2 = Polygon(M, F1, F2)
SetColor(C_curve, "Black")
SetVisibleInView(A_left, 1, false)
SetVisibleInView(A_right, 1, false)
SetVisibleInView(B_top, 1, false)
SetVisibleInView(B_bottom, 1, false)
SetColor(F1, "DarkGreen")
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F2, "DarkGreen")
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(l1, "Blue")
SetColor(l2, "Blue")
SetColor(l3, "Blue")
SetLineStyle(l3, 3)
SetLineStyle(l1, 0)
SetLineStyle(l2, 0)
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.15)
ShowLabel(triMF1F2, false)
SetVisibleInView(l1, 1, false)
SetVisibleInView(l2, 1, false)
SetVisibleInView(l3, 1, false)
