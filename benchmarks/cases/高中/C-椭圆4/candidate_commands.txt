MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-sqrt(36 - 20), 0)
F2 = (sqrt(36 - 20), 0)
C = Ellipse(F1, F2, 6)
V_right = (6, 0)
V_top = (0, sqrt(20))
arcM = Arc(C, V_right, V_top)
base = Distance(F1, F2)
circleF1 = Circle(F1, base)
M = Intersect(arcM, circleF1)
triMF1F2 = Polygon(M, F1, F2)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(C, "Black")
ShowLabel(C, true)
SetLabelMode(C, 1)
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
SetVisibleInView(arcM, 1, false)
SetVisibleInView(base, 1, false)
SetColor(circleF1, "Gray")
SetVisibleInView(circleF1, 1, false)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.3)
ShowLabel(triMF1F2, false)
