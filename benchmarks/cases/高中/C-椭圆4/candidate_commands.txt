MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Ellipse((-4, 0), (4, 0), 6)
SetColor(C_curve, "Black")
F1 = (-4, 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetPointSize(F1, 5)
SetColor(F1, "Black")
F2 = (4, 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetPointSize(F2, 5)
SetColor(F2, "Black")
V_right = (6, 0)
V_top = (0, sqrt(20))
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
arcM = Arc(C_curve, V_right, V_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetPointSize(M, 5)
triMF1F2 = Polygon(M, F1, F2)
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.3)
ShowLabel(triMF1F2, false)
