MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-4, 0)
F2 = (4, 0)

V_right = (6, 0)
V_top = (0, 2*sqrt(5))
arcM = Arc(C_curve, V_right, V_top)
t_m = Slider(0.01, 0.99, 0.01)
M = Point(arcM, t_m)
triMF = Polygon(F1, F2, M)
dMF1 = Distance(M, F1)
dMF2 = Distance(M, F2)
dF1F2 = Distance(F1, F2)

SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetPointSize(F1, 4)
SetColor(F1, "Blue")
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetPointSize(F2, 4)
SetColor(F2, "Blue")
SetColor(C_curve, "Black")
SetVisibleInView(V_right, 1, false)
SetVisibleInView(V_top, 1, false)
SetColor(M, "Red")

ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(triMF, "LightBlue")
