MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-4, 0)
F2 = (4, 0)
C = Ellipse(F1, F2, 6)
t_m = Slider(0.01, 5.99, 0.01)
M = Point(t_m, sqrt(20*(1-(t_m/6)^2)))
circleF1 = Circle(F1, F2)
triMF1F2 = Polygon(M, F1, F2)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Blue")
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Blue")
SetColor(C, "Black")
SetValue(t_m, 3)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(circleF1, "Gray")
SetLineStyle(circleF1, 1)
ShowLabel(circleF1, false)
SetColor(triMF1F2, "LightBlue")
SetFilling(triMF1F2, 0.3)
ShowLabel(triMF1F2, false)
