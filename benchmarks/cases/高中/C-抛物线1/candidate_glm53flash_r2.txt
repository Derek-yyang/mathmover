MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_p = Slider(0.05, 0.95, 0.01)
P = Point(C_curve, t_p)
lineAP = Line(A, P)
axHelp = Line((0, 1), (1, 1))
lineAQ = Reflect(lineAP, axHelp)
Q = Intersect(lineAQ, C_curve, 2)
linePQ = Line(P, Q)
k_AP = Slope(lineAP)
k_AQ = Slope(lineAQ)
k_l = Slope(linePQ)
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineAP, "Blue")
SetVisibleInView(axHelp, 1, false)
SetColor(lineAQ, "Blue")
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(linePQ, "Red")
SetLineThickness(linePQ, 5)
ShowLabel(k_l, true)
SetLabelMode(k_l, 1)
