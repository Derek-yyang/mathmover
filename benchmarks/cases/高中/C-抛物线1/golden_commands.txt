MODE: 2D
SetAxesVisible(true, true)
F1 = (-1.732, 0)
SetFixed(F1, true)
SetColor(F1, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetVisibleInView(F1, 1, true)
F2 = (1.732, 0)
SetFixed(F2, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetVisibleInView(F2, 1, true)
C = Hyperbola(F1, F2, 1.414)
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
A = (2, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(A, 1, true)
m = Slider(-1.5, 1.5, 0.1)
lineAP = Line(A, A + (1, m))
SetVisibleInView(lineAP, 1, true)
SetLineStyle(lineAP, 2)
SetColor(lineAP, "Gray")
lineAQ = Line(A, A + (1, -m))
SetVisibleInView(lineAQ, 1, true)
SetLineStyle(lineAQ, 2)
SetColor(lineAQ, "Gray")
tempP1 = Intersect(lineAP, C, 1)
tempP2 = Intersect(lineAP, C, 2)
P = If(Distance(tempP1, A) > 0.01, tempP1, tempP2)
SetVisibleInView(tempP1, 1, false)
SetVisibleInView(tempP2, 1, false)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(P, 1, true)
tempQ1 = Intersect(lineAQ, C, 1)
tempQ2 = Intersect(lineAQ, C, 2)
Q = If(Distance(tempQ1, A) > 0.01, tempQ1, tempQ2)
SetVisibleInView(tempQ1, 1, false)
SetVisibleInView(tempQ2, 1, false)
SetColor(Q, "Blue")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetVisibleInView(Q, 1, true)
l = Line(P, Q)
SetColor(l, "Black")
SetThickness(l, 2)
ShowLabel(l, true)
SetVisibleInView(l, 1, true)
asymp1 = Line((0,0), (1.414, 1))
asymp2 = Line((0,0), (1.414, -1))
SetLineStyle(asymp1, 3)
SetLineStyle(asymp2, 3)
SetColor(asymp1, "DarkGray")
SetColor(asymp2, "DarkGray")
SetVisibleInView(asymp1, 1, true)
SetVisibleInView(asymp2, 1, true)
