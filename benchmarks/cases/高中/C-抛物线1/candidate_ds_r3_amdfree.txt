MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
O = (0, 0)
t_m = Slider(0.1, 0.9, 0.01)
M = Point(C_curve, t_m)
P = Point(C_curve, -t_m)
lineAP = Line(A, P)
lineAQ = Line(A, M)
l = Line(P, M)
slope_l = Slope(l)
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetColor(M, "Red")
ShowLabel(M, true)
SetLabelMode(M, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Blue")
SetColor(l, "Red")
SetLineThickness(l, 3)
ShowLabel(slope_l, true)
SetLabelMode(slope_l, 2)
