MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
O = (0, 0)
A = (2, 1)
m = Slider(1.05, 2.5, 0.01)
line_l = Line((0, m), (1, m - 1))
P = Intersect(line_l, C, 1)
Q = Intersect(line_l, C, 2)
lineAP = Line(A, P)
lineAQ = Line(A, Q)
triAPQ = Polygon(A, P, Q)
SetColor(C, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetValue(m, 1.5)
SetColor(line_l, "Red")
SetThickness(line_l, 4)
ShowLabel(line_l, true)
SetLabelMode(line_l, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(lineAP, "Black")
SetColor(lineAQ, "Black")
SetColor(triAPQ, "LightBlue")
SetFilling(triAPQ, 0.2)
ShowLabel(triAPQ, false)
