MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
O = (0, 0)
A = (2, 1)
V1 = (sqrt(2), -1)
arcP = Arc(C_curve, (sqrt(2), 1), V1)
t_p = Slider(0.05, 0.95, 0.01)
P = Point(arcP, t_p)
hA = Line(A, A + (1, 0))
lineAP = Line(A, P)
reflAP = Reflect(lineAP, hA)
Q = Intersect(reflAP, C_curve, 2)
linePQ = Line(P, Q)
mAP = Slope(lineAP)
mPQ = Slope(linePQ)
SetColor(C_curve, "Black")
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Blue")
SetVisibleInView(V1, 1, false)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(hA, 1, false)
SetColor(lineAP, "Blue")
SetColor(reflAP, "Blue")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Red")
SetColor(linePQ, "Red")
SetLinethickness(linePQ, 3)
