MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_l = Slider(-2, 2, 0.01)
k_target = Slider(-1, 1, 0.01)
P = Point(C_curve, t_l)
lineAP = Line(A, P)
Q = Intersect(lineAP, C_curve, 2)
m_slope = Slider(-3, 3, 0.01)
line_l = Line(P, P + (1, m_slope))
slope_AP = Slope(lineAP)
slope_AQ = Slope(Line(A, Q))
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(line_l, "Blue")
SetLineStyle(line_l, 0)
SetLineThickness(line_l, 3)
