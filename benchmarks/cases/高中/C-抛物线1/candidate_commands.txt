MODE: 2D
SetAxesVisible(true, true)
// 1. 定义双曲线 C: x^2/2 - y^2 = 1
// 计算参数: a^2=2 => a=1.414, b^2=1 => b=1, c^2=a^2+b^2=3 => c=1.732
F1 = (-1.732, 0)
F2 = (1.732, 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetFixed(F1, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetFixed(F2, true)
C = Hyperbola(F1, F2, 1.4142)
SetColor(C, "Black")
SetThickness(C, 2)
// 2. 定义定点 A(2, 1)
A = (2, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Black")
SetPointSize(A, 4)
// 3. 定义主控参数斜率 m (控制直线 AP 的斜率)
// 根据图示，AP 与 AQ 关于水平方向对称，斜率之和为 0，即 k_AP = m, k_AQ = -m
m = Slider(0.1, 2, 0.05)
// 4. 构建直线 AP 和 AQ
// 使用方向向量 (1, m) 和 (1, -m)
lineAP = Line(A, A + (1, m))
lineAQ = Line(A, A + (1, -m))
SetLineStyle(lineAP, 2)
SetLineStyle(lineAQ, 2)
SetColor(lineAP, "Gray")
SetColor(lineAQ, "Gray")
// 5. 获取交点 P 和 Q (排除点 A)
tempP1 = Intersect(lineAP, C, 1)
tempP2 = Intersect(lineAP, C, 2)
P = If(Distance(tempP1, A) > 0.01, tempP1, tempP2)
SetVisibleInView(tempP1, 1, false)
SetVisibleInView(tempP2, 1, false)
tempQ1 = Intersect(lineAQ, C, 1)
tempQ2 = Intersect(lineAQ, C, 2)
Q = If(Distance(tempQ1, A) > 0.01, tempQ1, tempQ2)
SetVisibleInView(tempQ1, 1, false)
SetVisibleInView(tempQ2, 1, false)
// 6. 设置动点 P, Q 样式
SetColor(P, "Red")
SetPointSize(P, 5)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Blue")
SetPointSize(Q, 5)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
// 7. 构建直线 l (PQ)
l = Line(P, Q)
SetColor(l, "DarkRed")
SetThickness(l, 2)
ShowLabel(l, true)
// 8. 辅助线：连接 AP, AQ 以增强视觉表达
segAP = Segment(A, P)
segAQ = Segment(A, Q)
SetColor(segAP, "Red")
SetColor(segAQ, "Blue")
SetLineStyle(segAP, 1)
SetLineStyle(segAQ, 1)
