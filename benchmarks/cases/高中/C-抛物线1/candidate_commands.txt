MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_P = Slider(-2, 0.8, 0.01)
P = Point(C, t_P)
lineAP = Line(A, P)
lineV = Line(A, A + (0, 1))
lineAQ = Reflect(lineAP, lineV)
Q = Intersect(lineAQ, C, 2)
linePQ = Line(P, Q)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineAP, "Black")
SetVisibleInView(lineV, 1, false)
SetColor(lineAQ, "Black")
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(linePQ, "Red")
