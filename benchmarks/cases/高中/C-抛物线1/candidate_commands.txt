MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_p = Slider(0.01, 0.99, 0.01)
P = Point(C_curve, t_p)
t_q = Slider(0.01, 0.99, 0.01)
Q = Point(C_curve, t_q)
lineAP = Line(A, P)
lineAQ = Line(A, Q)
k_AP = Slope(lineAP)
k_AQ = Slope(lineAQ)
k_sum = k_AP + k_AQ
cond1 = If(k_sum == 0, 1, 0)
P_on_C = IsOnPath(P, C_curve)
Q_on_C = IsOnPath(Q, C_curve)
line_l = Line(P, Q)
k_l = Slope(line_l)
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Blue")
SetColor(line_l, "Green")
SetValue(k_l, 0)
SetFixed(line_l, false)
ShowLabel(line_l, false)
