MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t = Slider(0.05, 0.95, 0.01)
P = Point(C, t)
axisx2 = Line((2, 0), (2, 1))
Q = Reflect(P, axisx2)
l = Line(P, Q)
s_l = Slope(l)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "DarkGreen")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(axisx2, 1, false)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(l, "Red")
ShowLabel(s_l, true)
