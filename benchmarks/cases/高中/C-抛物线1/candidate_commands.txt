MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_p = Slider(0.05, 0.95, 0.01)
P = Point(C, t_p)
axisQ = Line((0, 1), (1, 1))
Q = Reflect(P, axisQ)
lineAP = Line(A, P)
lineAQ = Line(A, Q)
l = Line(P, Q)
m_l = Slope(l)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(A, "Red")
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetVisibleInView(axisQ, 1, false)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Blue")
SetColor(l, "Red")
SetLinethickness(l, 3)
ShowLabel(m_l, true)
