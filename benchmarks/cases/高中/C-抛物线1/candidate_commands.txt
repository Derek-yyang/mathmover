MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
F1 = (-sqrt(3), 0)
F2 = (sqrt(3), 0)
a = sqrt(2)
C = Hyperbola(F1, F2, a)
A = (2, 1)
s = Slider(-2, 2, 0.01)
P = (-sqrt(2)*cosh(s), sinh(s))
lineAP = Line(A, P)
slopeAP = Slope(lineAP)
slopeAQ = -slopeAP
lineAQ = Line(A, A + (1, slopeAQ))
Qlist = Intersect(lineAQ, C)
Q = Element(Qlist, 2)
line_l = Line(P, Q)
k_l = Slope(line_l)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Orange")
SetColor(Q, "Blue")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(line_l, "Green")
SetLineStyle(line_l, 1)
ShowLabel(k_l, true)
SetLabelMode(k_l, 1)
