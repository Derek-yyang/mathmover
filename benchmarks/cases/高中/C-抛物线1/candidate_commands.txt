MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
O = (0, 0)
A = (2, 1)
t_P = Slider(0.28, 0.47, 0.001, 1, 140)
P = Point(C_curve, t_P)
lineAP = Line(A, P)
kAP = Slope(lineAP)
lineAQ = Line(A, A + (1, -kAP))
kAQ = Slope(lineAQ)
Q1 = Intersect(lineAQ, C_curve, 1)
Q2 = Intersect(lineAQ, C_curve, 2)
Q = If(Distance(A, Q1) > Distance(A, Q2), Q1, Q2)
lineL = Line(P, Q)
slopeL = Slope(lineL)
Text("k_{AP} + k_{AQ} = " + (kAP + kAQ), (3.1, 2.3), true, true)
Text("k_l = " + slopeL, (3.1, 1.9), true, true)
SetColor(C_curve, "Black")
SetLineThickness(C_curve, 2)
SetFixed(O, true)
ShowLabel(O, true)
SetLabelMode(O, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetPointSize(A, 4)
SetValue(t_P, 0.38)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetPointSize(P, 4)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Blue")
SetVisibleInView(Q1, 1, false)
SetVisibleInView(Q2, 1, false)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetPointSize(Q, 4)
SetColor(lineL, "Red")
SetLineThickness(lineL, 2)
SetCaption(lineL, "l")
ShowLabel(lineL, true)
SetLabelMode(lineL, 3)
