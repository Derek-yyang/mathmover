MODE: 2D
SetAxesVisible(true, true)
F1 = (-sqrt(3), 0)
SetFixed(F1, true)
ShowLabel(F1, true)
SetLabelMode(F1, 1)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetVisibleInView(F1, 1, true)
F2 = (sqrt(3), 0)
SetFixed(F2, true)
ShowLabel(F2, true)
SetLabelMode(F2, 1)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetVisibleInView(F2, 1, true)
C = Hyperbola(F1, F2, sqrt(2))
SetColor(C, "Black")
A = (2, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(A, 1, true)
k = Slider(1.8, 4, 0.01)
lineAP = Line(A, A + (1, k))
lineAQ = Line(A, A + (1, -k))
tempP1 = Intersect(lineAP, C, 1)
tempP2 = Intersect(lineAP, C, 2)
P = If(Distance(tempP1, A) > 0.1, tempP1, tempP2)
SetVisibleInView(tempP1, 1, false)
SetVisibleInView(tempP2, 1, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Red")
SetVisibleInView(P, 1, true)
tempQ1 = Intersect(lineAQ, C, 1)
tempQ2 = Intersect(lineAQ, C, 2)
Q = If(Distance(tempQ1, A) > 0.1, tempQ1, tempQ2)
SetVisibleInView(tempQ1, 1, false)
SetVisibleInView(tempQ2, 1, false)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Blue")
SetVisibleInView(Q, 1, true)
l = Line(P, Q)
SetColor(l, "Red")
SetVisibleInView(l, 1, true)
