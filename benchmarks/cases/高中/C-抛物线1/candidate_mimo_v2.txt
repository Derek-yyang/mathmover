MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C = Hyperbola((-sqrt(6), 0), (sqrt(6), 0), sqrt(2))
A = (2, 1)
V_right = (sqrt(2), 0)
t_p = Slider(0.01, 0.99, 0.01)
arcP = Arc(C, V_right, (0, -1))
P = Point(arcP, t_p)
lineAP = Line(A, P)
k_AP = Slope(lineAP)
k_AQ = -k_AP
lineAQ = Line(A, A + (1, k_AQ))
Q = Intersect(lineAQ, C)
line_l = Line(P, Q)
k_l = Slope(line_l)
SetColor(C, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(V_right, 1, false)
SetColor(P, "Blue")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(lineAP, "Green")
SetColor(lineAQ, "Orange")
SetColor(Q, "Blue")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(line_l, "Red")
SetColor(k_l, "DarkGreen")
ShowLabel(k_l, true)
SetLabelMode(k_l, 1)
