b = Slider(1.2, 2.8, 0.01)

Q = Intersect(line_l, C_curve, 2)
lineAP = Line(A, P)
lineAQ = Line(A, Q)

SetPointSize(O, 5)

ShowLabel(A, true)
SetLabelMode(A, 1)
SetPointSize(A, 5)
SetVisibleInView(b, 1, true)
SetColor(line_l, "Red")
