MODE: 2D
SetAxesVisible(true, true)
SetGridVisible(true)
C_curve = Hyperbola((-sqrt(3), 0), (sqrt(3), 0), sqrt(2))
A = (2, 1)
t_p = Slider(0.5, 1.5, 0.01)
P = Point(C_curve, t_p)
t_q = Slider(-1.5, -0.5, 0.01)
Q = Point(C_curve, t_q)
line_l = Line(P, Q)
lineAP = Line(A, P)
lineAQ = Line(A, Q)
kAP = Slope(lineAP)
kAQ = Slope(lineAQ)
sum_k = kAP + kAQ
kl = Slope(line_l)
SetColor(C_curve, "Black")
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetColor(P, "Red")
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(Q, "Red")
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(line_l, "Red")
SetLineThickness(line_l, 3)
SetColor(lineAP, "Blue")
SetColor(lineAQ, "Blue")
ShowLabel(sum_k, true)
SetLabelMode(sum_k, 1)
ShowLabel(kl, true)
SetLabelMode(kl, 1)
