MODE: 2D
F1 = (-sqrt(3), 0)
F2 = (sqrt(3), 0)
SetColor(F1, "Black")
SetPointSize(F1, 4)
SetLabelMode(F1, 1)
SetVisibleInView(F1, 1, true)
SetColor(F2, "Black")
SetPointSize(F2, 4)
SetLabelMode(F2, 1)
SetVisibleInView(F2, 1, true)
C = Hyperbola(F1, F2, sqrt(2))
SetColor(C, "Black")
SetVisibleInView(C, 1, true)
A = (2, 1)
SetFixed(A, true)
ShowLabel(A, true)
SetLabelMode(A, 1)
SetVisibleInView(A, 1, true)
k = Slider(-1.5, 1.5, 0.01)
lineAP = Line(A, A + (1, k))
SetVisibleInView(lineAP, 1, true)
lineAQ = Line(A, A + (1, -k))
SetVisibleInView(lineAQ, 1, true)
tempP1 = Intersect(lineAP, C, 1)
tempP2 = Intersect(lineAP, C, 2)
P = If(Distance(tempP1, A) < 0.1, tempP2, tempP1)
SetVisibleInView(tempP1, 1, false)
SetVisibleInView(tempP2, 1, false)
ShowLabel(P, true)
SetLabelMode(P, 1)
SetColor(P, "Blue")
SetVisibleInView(P, 1, true)
tempQ1 = Intersect(lineAQ, C, 1)
tempQ2 = Intersect(lineAQ, C, 2)
Q = If(Distance(tempQ1, A) < 0.1, tempQ2, tempQ1)
SetVisibleInView(tempQ1, 1, false)
SetVisibleInView(tempQ2, 1, false)
ShowLabel(Q, true)
SetLabelMode(Q, 1)
SetColor(Q, "Blue")
SetVisibleInView(Q, 1, true)
l = Line(P, Q)
SetColor(l, "Red")
SetLineStyle(l, 1)
SetVisibleInView(l, 1, true)
ShowLabel(l, true)
SetLabelMode(l, 1)
SetAxesVisible(true, true)
